    -- Auto Open Locked Doors v2: room-scoped targeting, bounded scans, honest verification.
    do
        local Players = game:GetService("Players")
        local localPlayer = Players.LocalPlayer
        local env = (getgenv and getgenv()) or _G
        if env._FairwellAutoOpenLockedDoorsState then
            env._FairwellAutoOpenLockedDoorsState.running = false
        end
        local state = {running = true, enabled = false}
        env._FairwellAutoOpenLockedDoorsState = state

        local box = env._FairwellHotelGroupbox
        if not box then
            local ok, result = pcall(function()
                return tab:AddLeftGroupbox("Hotel", "building-2")
            end)
            if ok then box = result; env._FairwellHotelGroupbox = result end
        end

        if box then
            local statusLabel
            local lastStatus = ""
            local lastActionAt = 0
            local function status(message)
                if message == lastStatus then return end
                lastStatus = message
                if statusLabel then pcall(function() statusLabel:SetText("Status: " .. message) end) end
            end
            pcall(function()
                statusLabel = box:AddLabel({Text = "Status: off", DoesWrap = true})
                box:AddLabel({
                    Text = "Targets the locked door in your current room. Teleport and prompt interactions are best-effort client actions; the server must accept the unlock.",
                    DoesWrap = true
                })
            end)

            local function norm(value)
                return string.lower(tostring(value or "")):gsub("[%s%p_%-]", "")
            end

            local function getCharacter()
                local character = localPlayer.Character
                local root = character and character:FindFirstChild("HumanoidRootPart")
                local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                if root and humanoid and humanoid.Health > 0 then return character, root end
                return character, nil
            end

            local function inventory()
                local character = localPlayer.Character
                local backpack = localPlayer:FindFirstChildOfClass("Backpack")
                local key, lockpick = false, false
                for _, container in ipairs({character, backpack}) do
                    if container then
                        for _, item in ipairs(container:GetChildren()) do
                            if item:IsA("Tool") then
                                local name = norm(item.Name)
                                if name:find("key", 1, true) then key = true end
                                if name:find("lockpick", 1, true) then lockpick = true end
                            end
                        end
                    end
                end
                return key, lockpick
            end

            local function nearestRoom(root)
                local rooms = workspace:FindFirstChild("CurrentRooms")
                if not rooms or not root then return nil end
                local roomNumber = tonumber(root:GetAttribute("CurrentRoom"))
                if roomNumber then
                    local exact = rooms:FindFirstChild(tostring(roomNumber))
                    if exact then return exact end
                end
                -- Prefer a room whose bounding box contains the player.
                for _, room in ipairs(rooms:GetChildren()) do
                    if room:IsA("Model") then
                        local ok, cf, size = pcall(function() return room:GetBoundingBox() end)
                        if ok and cf and size then
                            local p = cf:PointToObjectSpace(root.Position)
                            if math.abs(p.X) <= size.X / 2 + 8
                                and math.abs(p.Y) <= size.Y / 2 + 12
                                and math.abs(p.Z) <= size.Z / 2 + 8 then
                                return room
                            end
                        end
                    end
                end
                -- Fallback: nearest room part, but only within a small radius.
                local best, bestDistance
                for _, room in ipairs(rooms:GetChildren()) do
                    local distance
                    for _, item in ipairs(room:GetDescendants()) do
                        if item:IsA("BasePart") then
                            local d = (root.Position - item.Position).Magnitude
                            if not distance or d < distance then distance = d end
                        end
                    end
                    if distance and distance <= 65 and (not bestDistance or distance < bestDistance) then
                        best, bestDistance = room, distance
                    end
                end
                return best
            end

            local function getPromptPart(prompt)
                local node = prompt and prompt.Parent
                while node do
                    if node:IsA("BasePart") then return node end
                    node = node.Parent
                end
                return nil
            end

            local function isLockedDoorPrompt(prompt, hasKey, hasLockpick)
                if not prompt:IsA("ProximityPrompt") or not prompt.Enabled then return false end
                local action, objectText = norm(prompt.ActionText), norm(prompt.ObjectText)
                local context, node = "", prompt.Parent
                local door, locked = false, false
                for _ = 1, 8 do
                    if not node then break end
                    local name = norm(node.Name)
                    context = context .. " " .. name
                    if name:find("door", 1, true) or name:find("gate", 1, true) then door = true end
                    if name:find("lock", 1, true) or name:find("padlock", 1, true) then locked = true end
                    node = node.Parent
                end
                local unlockAction = action:find("unlock", 1, true)
                    or action:find("lockpick", 1, true)
                    or action:find("usekey", 1, true)
                local lockText = objectText:find("lock", 1, true)
                    or context:find("lockeddoor", 1, true)
                    or context:find("doorlock", 1, true)
                    or locked
                if not ((unlockAction and (door or lockText)) or (door and lockText)) then return false end
                if action:find("lockpick", 1, true) then return hasLockpick end
                return hasKey or hasLockpick
            end

            local function teleportToPrompt(root, prompt)
                local part = getPromptPart(prompt)
                if not part then return false end
                local character = root.Parent
                -- Approach from both sides based on the player's current position;
                -- avoids assuming LookVector always points toward the usable side.
                local forward = part.CFrame.LookVector
                local a = part.Position + forward * 3 + Vector3.new(0, 1.5, 0)
                local b = part.Position - forward * 3 + Vector3.new(0, 1.5, 0)
                local destination = ((root.Position - a).Magnitude <= (root.Position - b).Magnitude) and a or b
                local ok = pcall(function()
                    character:PivotTo(CFrame.new(destination, part.Position))
                end)
                if not ok then
                    ok = pcall(function() root.CFrame = CFrame.new(destination, part.Position) end)
                end
                return ok
            end

            local function activate(prompt)
                local fire = (type(fireproximityprompt) == "function" and fireproximityprompt)
                    or env.fireproximityprompt
                if type(fire) == "function" then
                    local ok = pcall(fire, prompt)
                    if ok then return true end
                end
                return pcall(function()
                    prompt:InputHoldBegin()
                    task.wait(math.clamp(prompt.HoldDuration, 0.1, 1.5))
                    prompt:InputHoldEnd()
                end)
            end

            pcall(function()
                box:AddToggle("FairwellAutoOpenLockedDoors", {
                    Text = "Auto Open Locked Doors",
                    Tooltip = "Finds the locked door in your current room, approaches it, and attempts the unlock interaction.",
                    Default = false,
                    Callback = function(value)
                        state.enabled = value == true
                        lastActionAt = 0
                        status(state.enabled and "enabled; detecting your room" or "off")
                    end
                })
            end)

            task.spawn(function()
                while state.running do
                    task.wait(0.35)
                    if state.enabled then
                        local character, root = getCharacter()
                        if not root then
                            status("waiting for your character")
                        else
                            local room = nearestRoom(root)
                            if not room then
                                status("could not confidently detect your current room")
                            else
                                local hasKey, hasLockpick = inventory()
                                if not hasKey and not hasLockpick then
                                    status("in room " .. room.Name .. "; need a key or lockpick before unlocking")
                                else
                                    local target, targetDistance
                                    for _, item in ipairs(room:GetDescendants()) do
                                        if item:IsA("ProximityPrompt") and isLockedDoorPrompt(item, hasKey, hasLockpick) then
                                            local part = getPromptPart(item)
                                            if part then
                                                local distance = (part.Position - root.Position).Magnitude
                                                if not targetDistance or distance < targetDistance then
                                                    target, targetDistance = item, distance
                                                end
                                            end
                                        end
                                    end
                                    if not target then
                                        status("in room " .. room.Name .. "; no matching locked-door prompt found")
                                    elseif os.clock() - lastActionAt >= 3 then
                                        lastActionAt = os.clock()
                                        status("room " .. room.Name .. "; approaching locked door")
                                        if teleportToPrompt(root, target) then
                                            task.wait(0.25)
                                            if target.Parent and target.Enabled then
                                                status("at door in room " .. room.Name .. "; attempting unlock")
                                                activate(target)
                                                task.wait(0.65)
                                            end
                                            if not target.Parent or not target.Enabled then
                                                status("door prompt changed after interaction; checking next scan")
                                            elseif target.Parent and target.Enabled then
                                                status("interaction sent, but door still appears locked")
                                            end
                                        else
                                            status("found door, but could not approach its prompt")
                                        end
                                    else
                                        status("room " .. room.Name .. "; locked door identified")
                                    end
                                end
                            end
                        end
                    end
                end
            end)
        end
    end
