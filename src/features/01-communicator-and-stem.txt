-- Fairwell Heaven DOORS features for MSDoors
-- Remote content file. Only gameplay features are shown in this tab.

return function(tab, reloadContent, hotelTab)
    if not tab then return end

    -- Fairwell Pet: just Fairwell's head, with a custom Motor6D hold pose.
    -- Replaces the old Communicator. The right-arm pose is applied after Roblox's
    -- Animator update so this does not depend on DOORS' default Tool-hold animation.
    do
        local Players = game:GetService("Players")
        local TweenService = game:GetService("TweenService")
        local RunService = game:GetService("RunService")
        local UserInputService = game:GetService("UserInputService")
        local ReplicatedStorage = game:GetService("ReplicatedStorage")
        local player = Players.LocalPlayer
        local allowedUserId = 11412884109
        local ITEM_NAME = "Fairwell Pet"

        local function grantFairwellPet()
            if not player or player.UserId ~= allowedUserId then
                if player then warn("[Fairwell Pet] Auto-grant skipped: UserId does not match.") end
                return
            end
            local backpack = player:FindFirstChildOfClass("Backpack") or player:WaitForChild("Backpack", 10)
            if not backpack then warn("[Fairwell Pet] Backpack not ready."); return end
            local character = player.Character
            if backpack:FindFirstChild(ITEM_NAME) or (character and character:FindFirstChild(ITEM_NAME)) then return end

            local tool = Instance.new("Tool")
            tool.Name = ITEM_NAME
            tool.ToolTip = "Fairwell's head. Use for a happy bounce; inspect for a curious tilt."
            tool.RequiresHandle = true
            tool.CanBeDropped = false
            tool.Grip = CFrame.new(0, 0.10, -1.05) * CFrame.Angles(0, math.rad(180), 0)

            -- Square Fairwell head based on the supplied reference drawing.
            local head = Instance.new("Part")
            head.Name = "Handle"
            head.Shape = Enum.PartType.Block
            head.Size = Vector3.new(0.92, 0.88, 0.30)
            head.Color = Color3.fromRGB(0, 0, 0)
            head.Material = Enum.Material.SmoothPlastic
            head.CanCollide = false
            head.CanTouch = false
            head.CanQuery = false
            head.Massless = true
            head.CastShadow = false
            head.Parent = tool

            local function addFacePart(name, size, color, offset, shape)
                local part = Instance.new("Part")
                part.Name = name
                part.Shape = shape or Enum.PartType.Block
                part.Size = size
                part.Color = color
                part.Material = Enum.Material.SmoothPlastic
                part.CanCollide = false
                part.CanTouch = false
                part.CanQuery = false
                part.Massless = true
                part.CastShadow = false
                part.CFrame = head.CFrame * offset
                part.Parent = tool
                local weld = Instance.new("WeldConstraint")
                weld.Name = name .. "Weld"
                weld.Part0 = head
                weld.Part1 = part
                weld.Parent = part
                return part
            end

            local cyan = Color3.fromRGB(0, 174, 245)
            local red = Color3.fromRGB(255, 0, 0)
            local black = Color3.fromRGB(0, 0, 0)
            addFacePart("BlueFace", Vector3.new(0.76, 0.70, 0.035), cyan, CFrame.new(0, 0, -0.163))
            addFacePart("LeftEyeRim", Vector3.new(0.245, 0.275, 0.045), black, CFrame.new(-0.205, 0.105, -0.192), Enum.PartType.Ball)
            addFacePart("RightEyeRim", Vector3.new(0.245, 0.275, 0.045), black, CFrame.new(0.205, 0.105, -0.192), Enum.PartType.Ball)
            addFacePart("LeftEye", Vector3.new(0.205, 0.23, 0.04), red, CFrame.new(-0.205, 0.105, -0.219), Enum.PartType.Ball)
            addFacePart("RightEye", Vector3.new(0.205, 0.23, 0.04), red, CFrame.new(0.205, 0.105, -0.219), Enum.PartType.Ball)
            local mouth = addFacePart("RedMouth", Vector3.new(0.22, 0.035, 0.025), red, CFrame.new(0.055, -0.205, -0.192))

            local equipped = false
            local inspectDebounce = 0
            local poseConnection
            local characterJoints = {}
            local poseOffsets = {}
            local originalMouthColor = mouth.Color
            local activeTweens = {}

            local function findMotor(characterNow, names)
                for _, object in ipairs(characterNow:GetDescendants()) do
                    if object:IsA("Motor6D") then
                        for _, name in ipairs(names) do
                            if object.Name == name then return object end
                        end
                    end
                end
                return nil
            end

            local function resetHoldPose()
                if poseConnection then poseConnection:Disconnect(); poseConnection = nil end
                for joint, original in pairs(characterJoints) do
                    if joint and joint.Parent then joint.Transform = original end
                end
                characterJoints = {}
                poseOffsets = {}
            end

            local function applyHoldPose()
                resetHoldPose()
                local characterNow = player.Character
                if not characterNow then return end
                local humanoid = characterNow:FindFirstChildOfClass("Humanoid")
                if not humanoid then return end

                -- R6 uses Right Shoulder/Right Hip; R15 uses shoulder, elbow and wrist.
                local shoulderNames = {"RightShoulder", "Right Shoulder"}
                local elbowNames = {"RightElbow", "Right Elbow"}
                local wristNames = {"RightWrist", "Right Wrist"}
                local shoulder = findMotor(characterNow, shoulderNames)
                local elbow = findMotor(characterNow, elbowNames)
                local wrist = findMotor(characterNow, wristNames)
                if shoulder then
                    characterJoints[shoulder] = shoulder.Transform
                    poseOffsets[shoulder] = CFrame.Angles(math.rad(-72), math.rad(8), math.rad(20))
                end
                if elbow then
                    characterJoints[elbow] = elbow.Transform
                    poseOffsets[elbow] = CFrame.Angles(math.rad(-42), 0, math.rad(-12))
                end
                if wrist then
                    characterJoints[wrist] = wrist.Transform
                    poseOffsets[wrist] = CFrame.Angles(math.rad(5), 0, math.rad(-10))
                end

                -- Apply after the Animator writes Motor6D.Transform each frame.
                poseConnection = RunService.PreSimulation:Connect(function()
                    if not equipped or not player.Character then return end
                    for joint, offset in pairs(poseOffsets) do
                        if joint and joint.Parent then
                            joint.Transform = offset
                        end
                    end
                end)
            end

            local function playFaceBounce()
                if not equipped then return end
                for _, tween in ipairs(activeTweens) do pcall(function() tween:Cancel() end) end
                activeTweens = {}
                local baseGrip = tool.Grip
                local up = TweenService:Create(head, TweenInfo.new(0.12, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
                    Size = Vector3.new(0.98, 0.82, 0.30)
                })
                local settle = TweenService:Create(head, TweenInfo.new(0.2, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {
                    Size = Vector3.new(0.92, 0.88, 0.30)
                })
                table.insert(activeTweens, up)
                table.insert(activeTweens, settle)
                up:Play()
                up.Completed:Once(function()
                    if head and head.Parent then settle:Play() end
                end)
                mouth.Color = paleBlue
                task.delay(0.25, function()
                    if mouth and mouth.Parent then mouth.Color = originalMouthColor end
                    if tool.Parent and equipped then tool.Grip = baseGrip end
                end)
            end

            local function playInspectPose()
                if not equipped or os.clock() - inspectDebounce < 0.25 then return end
                inspectDebounce = os.clock()
                -- Brief curious tilt; the head remains attached to the hand as one item.
                local base = tool.Grip
                local tilt = TweenService:Create(tool, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
                    Grip = base * CFrame.Angles(math.rad(-12), 0, math.rad(16))
                })
                local settle = TweenService:Create(tool, TweenInfo.new(0.22, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {
                    Grip = base
                })
                tilt:Play()
                tilt.Completed:Once(function()
                    if tool and tool.Parent then settle:Play() end
                end)
                mouth.Color = paleBlue
                task.delay(0.55, function()
                    if mouth and mouth.Parent then mouth.Color = originalMouthColor end
                end)
            end

            tool.Equipped:Connect(function()
                equipped = true
                applyHoldPose()
                print("[Fairwell Pet] Equipped with custom Motor6D hold pose.")
            end)
            tool.Unequipped:Connect(function()
                equipped = false
                resetHoldPose()
                for _, tween in ipairs(activeTweens) do pcall(function() tween:Cancel() end) end
                activeTweens = {}
                head.Size = Vector3.new(0.92, 0.88, 0.30)
                mouth.Color = originalMouthColor
            end)
            tool.Activated:Connect(playFaceBounce)

            -- Listen for the inspect signals exposed by this DOORS/MSDoors build.
            local inspectConnections = {}
            local remotes = ReplicatedStorage:FindFirstChild("RemotesFolder")
            local toolAnimate = remotes and remotes:FindFirstChild("ToolAnimate")
            if toolAnimate and toolAnimate:IsA("RemoteEvent") then
                table.insert(inspectConnections, toolAnimate.OnClientEvent:Connect(function(...)
                    if not equipped then return end
                    for _, value in ipairs({...}) do
                        if type(value) == "string" and string.lower(value) == "inspect" then
                            playInspectPose()
                            return
                        end
                    end
                end))
            end
            local coinRoll = workspace:FindFirstChild("FIXROBLOX_015")
            coinRoll = coinRoll and coinRoll:FindFirstChild("CoinRoll")
            local remoteInspect = coinRoll and coinRoll:FindFirstChild("RemoteInspect")
            if remoteInspect and remoteInspect:IsA("RemoteEvent") then
                table.insert(inspectConnections, remoteInspect.OnClientEvent:Connect(playInspectPose))
            end
            local inputConnection = UserInputService.InputBegan:Connect(function(input, processed)
                if not processed and equipped and input.KeyCode == Enum.KeyCode.R then
                    playInspectPose()
                end
            end)
            tool.Destroying:Connect(function()
                equipped = false
                resetHoldPose()
                if inputConnection then inputConnection:Disconnect() end
                for _, connection in ipairs(inspectConnections) do
                    if connection then connection:Disconnect() end
                end
            end)

            tool.Parent = backpack
            print("[Fairwell Pet] Granted Fairwell's head with a custom hold pose.")
        end

        task.spawn(function()
            grantFairwellPet()
            task.wait(2)
            local backpack = player and player:FindFirstChildOfClass("Backpack")
            local character = player and player.Character
            if player and player.UserId == allowedUserId and backpack
                and not backpack:FindFirstChild(ITEM_NAME)
                and not (character and character:FindFirstChild(ITEM_NAME)) then
                grantFairwellPet()
            end
        end)
    end

    local stairwellBox
    pcall(function()
        stairwellBox = tab:AddLeftGroupbox("Stairwell", "eye")
    end)

    local espOk, espBox = pcall(function()
        return stairwellBox
    end)

    if espOk and espBox then
        local espEnabled = false
        local espFolder = Instance.new("Folder")
        espFolder.Name = "FairwellStemESP"
        espFolder.Parent = game:GetService("CoreGui")

        local function clearESP()
            for _, child in ipairs(espFolder:GetChildren()) do
                child:Destroy()
            end
        end

        local function stemRelated(instance)
            local node = instance
            for _ = 1, 5 do
                if not node then break end
                if string.find(string.lower(node.Name), "stem", 1, true) then
                    return true
                end
                node = node.Parent
            end
            return false
        end

        local function scan()
            clearESP()
            local seen = {}
            for _, instance in ipairs(workspace:GetDescendants()) do
                if stemRelated(instance) then
                    local target = instance:IsA("Model") and instance or instance:FindFirstAncestorOfClass("Model")
                    target = target or (instance:IsA("BasePart") and instance or nil)
                    if target and not seen[target] then
                        seen[target] = true
                        local holder = Instance.new("Folder")
                        holder.Name = "StemESPItem"
                        holder.Parent = espFolder

                        local highlight = Instance.new("Highlight")
                        highlight.Adornee = target
                        highlight.FillColor = Color3.fromRGB(255, 70, 70)
                        highlight.OutlineColor = Color3.fromRGB(255, 230, 70)
                        highlight.FillTransparency = 0.55
                        highlight.OutlineTransparency = 0
                        highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        highlight.Parent = holder

                        local part = target:IsA("BasePart") and target or target:FindFirstChildWhichIsA("BasePart", true)
                        if part then
                            local labelGui = Instance.new("BillboardGui")
                            labelGui.Name = "StemLabel"
                            labelGui.Adornee = part
                            labelGui.AlwaysOnTop = true
                            labelGui.Size = UDim2.fromOffset(110, 30)

                            labelGui.Parent = holder

                            local label = Instance.new("TextLabel")
                            label.BackgroundTransparency = 1
                            label.Size = UDim2.fromScale(1, 1)
                            label.Text = "STEM"
                            label.TextColor3 = Color3.fromRGB(255, 230, 70)
                            label.TextStrokeTransparency = 0
                            label.TextScaled = true
                            label.Font = Enum.Font.GothamBold
                            label.Parent = labelGui
                        end
                    end
                end
            end
        end

        pcall(function()
            espBox:AddToggle("FairwellStemESPToggle", {
                Text = "STEM ESP",
                Default = false,

                    espEnabled = enabled == true
                    clearESP()

    local stemOk, stemBox = pcall(function()
        return stairwellBox

    if stemOk and stemBox then
        local Players = game:GetService("Players")
        local ReplicatedStorage = game:GetService("ReplicatedStorage")
        local player = Players.LocalPlayer
        local env = (getgenv and getgenv()) or _G
        env._FairwellAutoKillStem = false
        local statusLabel

        local function setStatus(message)
            print("[Fairwell Auto kill Stem] " .. message)
            if statusLabel then pcall(function() statusLabel:SetText("Status: " .. message) end) end
        end
        pcall(function() statusLabel = stemBox:AddLabel({Text = "Status: idle", DoesWrap = true}) end)

        local function findHeldStem()
            local character = player.Character
            if not character then return nil end
            local tool = character:FindFirstChild("Stem")
            if not tool or not tool:IsA("Tool") then return nil end
            local handle = tool:FindFirstChild("Handle")
            if not handle or not handle:IsA("BasePart") then return tool, nil, nil end
            for _, item in ipairs(handle:GetDescendants()) do
                if item.Name:match("^Stem_") then return tool, handle, item.Name end
            end
            return tool, handle, nil
        end

        local function findVariant(tool, handle, identifier)
            -- Check the held item, its identifier, attributes, and exposed string values.
            local function inspect(value)
                local lower = string.lower(tostring(value or ""))
                for _, variant in ipairs({"Mouth", "Ear", "Eye", "Blank"}) do
                    if string.find(lower, string.lower(variant), 1, true) then return variant end
                end
            end
            local found = inspect(identifier)
            if found then return found end
            local roots = {tool}
            if handle then table.insert(roots, handle) end
            for _, root in ipairs(roots) do
                for key, value in pairs(root:GetAttributes()) do
                    found = inspect(key) or inspect(value)
                    if found then return found end
                end
                for _, item in ipairs(root:GetDescendants()) do
                    found = inspect(item.Name)
                    if not found and item:IsA("StringValue") then found = inspect(item.Value) end
                    if not found then
                        for key, value in pairs(item:GetAttributes()) do
                            found = inspect(key) or inspect(value)
                            if found then break end
                        end
                    end
                    if found then return found end
                end
            end
            return "Blank"
        end

        local function findNearbyStemPrompt()
            local character = player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not root then return nil, "character root not found" end
            local bestPrompt, bestDistance
            for _, prompt in ipairs(workspace:GetDescendants()) do
                if prompt:IsA("ProximityPrompt") and prompt.Enabled then
                    local parent = prompt.Parent
                    local part = parent and (parent:IsA("BasePart") and parent or parent:FindFirstChildWhichIsA("BasePart", true))
                    if part then
                        local distance = (root.Position - part.Position).Magnitude
                        if distance <= math.max(prompt.MaxActivationDistance + 5, 12) then
                            local node, isStem = prompt, false
                            for _ = 1, 8 do
                                if not node then break end
                                if string.find(string.lower(node.Name), "stem", 1, true) then isStem = true break end
                                node = node.Parent
                            end
                            if isStem and (not bestDistance or distance < bestDistance) then
                                bestPrompt, bestDistance = prompt, distance
                            end
                        end
                    end
                end
            end
            return bestPrompt, bestPrompt and "nearby Stem pickup found" or "searching for a nearby Stem"
        end

        local function activatePrompt(prompt)
            if not prompt or not prompt.Parent then return false, "prompt disappeared" end
            local ok, err = pcall(function()
                prompt:InputHoldBegin()
                task.wait(math.max(prompt.HoldDuration, 0.02) + 0.02)
                prompt:InputHoldEnd()
            end)
            if not ok then return false, tostring(err) end
            return true, "pickup attempted"
        end

        local function fireHeldStem(tool, handle, identifier)
            if not handle then return false, "held Stem Handle not found" end
            if not identifier then return false, "Stem_ identifier not found under Handle" end
            local remotes = ReplicatedStorage:FindFirstChild("RemotesFolder")
            local remote = remotes and remotes:FindFirstChild("FireProjectile")
            if not remote or not remote:IsA("RemoteEvent") then return false, "FireProjectile RemoteEvent not found" end
            local camera = workspace.CurrentCamera
            if not camera then return false, "camera unavailable" end
            local direction = camera.CFrame.LookVector
            local velocity = direction * 100
            local exclusions = {}
            local props = workspace:FindFirstChild("Props")
            local character = player.Character
            if props then table.insert(exclusions, props) end
            if character then table.insert(exclusions, character) end
            local variant = findVariant(tool, handle, identifier)
            local ok, err = pcall(function()
                remote:FireServer(identifier, handle.Position, direction, velocity, "Stem", variant, exclusions)
            end)
            if not ok then return false, tostring(err) end
            return true, "threw " .. variant .. " Stem"
        end

        pcall(function()
            stemBox:AddToggle("FairwellAutoKillStemToggle", {
                Text = "Auto kill Stem",
                Default = false,
                Callback = function(enabled)
                    env._FairwellAutoKillStem = enabled == true
                    if not enabled then setStatus("off") return end
                    task.spawn(function()
                        setStatus("enabled; auto-picking up and throwing Stems")
                        local lastIdentifier
                        local lastInfo
                        while env._FairwellAutoKillStem do
                            local character = player.Character
                            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                            if not humanoid or humanoid.Health <= 0 then
                                setStatus("paused: character unavailable/dead")
                                task.wait(0.25)
                            else
                                local tool, handle, identifier = findHeldStem()
                                if tool and handle and identifier and identifier ~= lastIdentifier then
                                    local ok, message = fireHeldStem(tool, handle, identifier)
                                    setStatus(ok and message or ("throw failed: " .. message))
                                    if ok then lastIdentifier = identifier end
                                    task.wait(ok and 0.08 or 0.15)
                                else
                                    -- If no usable held Stem is available, seek and pick one up.
                                    if not tool or not handle or not identifier then lastIdentifier = nil end
                                    local prompt, info = findNearbyStemPrompt()
                                    if info ~= lastInfo then setStatus(info); lastInfo = info end
                                    if prompt then
                                        local ok, message = activatePrompt(prompt)
                                        setStatus(ok and message or ("pickup failed: " .. message))
                                        task.wait(0.04)
                                    else
                                        task.wait(0.08)
                                    end
                                end
                            end
                        end
                        setStatus("off")
                    end)
                end
            })
        end)
    end

        local function landingPart(object)
            if object:IsA("BasePart") then return object end
            if object:IsA("Model") then
                return object.PrimaryPart or object:FindFirstChildWhichIsA("BasePart", true)
            end
            return object:FindFirstChildWhichIsA("BasePart", true)
        end

        local function getLandings(logic)
            local result = {}
            if not logic then return result end
            for _, object in ipairs(logic:GetChildren()) do
                if object:IsA("Model") or object:IsA("BasePart") then
                    local part = landingPart(object)
                    if part and part:IsDescendantOf(workspace) then
                        table.insert(result, {object = object, part = part})
                    end
                end
            end
            table.sort(result, function(a, b)
                return a.part.Position.Y < b.part.Position.Y
            end)
            return result
        end

        local function teleportTo(part)
            if not part or not part.Parent then
                setFarmStatus("teleport failed: target part missing")
                return false
            end
            local character = player and player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not character or not root then
                setFarmStatus("teleport failed: character/root missing")
                return false
            end

            -- Use a direct CFrame assignment to the HumanoidRootPart; PivotTo can
            -- silently appear ineffective for this client-side automation.
            local destination = part.CFrame + Vector3.new(0, 3, 0)
            local ok, err = pcall(function()
                root.CFrame = destination
                root.AssemblyLinearVelocity = Vector3.zero
                root.AssemblyAngularVelocity = Vector3.zero
            end)
            if not ok then
                setFarmStatus("CFrame teleport failed: " .. tostring(err))
                return false
            end
            return true
        end

        local function isInsideCurrentRoom(object)

            return room ~= nil and object:IsDescendantOf(room)

        -- Return closed?, reason. Resolve all explicit state markers before using
        -- prompts: an explicit OPEN marker anywhere in the door model wins over a
        -- stale false marker or an enabled prompt left behind by the game.
        local function getDoorState(door)
            if not door or not door:IsA("Model") then return false, "not a Door model" end

            local function containsOpenedWord(value)
                return type(value) == "string"
                    and string.find(string.lower(value), "opened", 1, true) ~= nil
            end

            if containsOpenedWord(door.Name) then return false, "model name says opened" end

            -- An object merely named "Opened" is not proof the door is open:
            -- some maps keep marker objects around regardless of their state.
            -- Only Boolean values/attributes can establish state; prompt is fallback.
            local foundExplicitClosed = false
            local keys = {"Opened", "IsOpen", "Open", "IsOpened"}
            local function inspectState(container)
                for _, key in ipairs(keys) do
                    local value = container:GetAttribute(key)
                    if value == true then return "open" end
                    if value == false then foundExplicitClosed = true end
                end
                if container:IsA("BoolValue") then
                    local key = string.lower(container.Name)
                    if key == "opened" or key == "isopen" or key == "open" or key == "isopened" then
                        if container.Value == true then return "open" end
                        foundExplicitClosed = true
                    end
                end
                return nil
            end

            local rootState = inspectState(door)
            if rootState == "open" then return false, "explicit open flag" end
            for _, object in ipairs(door:GetDescendants()) do
                if inspectState(object) == "open" then return false, "explicit open flag: " .. object.Name end
            end
            if foundExplicitClosed then return true, "explicit closed flag" end

            for _, object in ipairs(door:GetDescendants()) do
                if object:IsA("ProximityPrompt") and object.Enabled then
                    return true, "enabled prompt (fallback)"
                end
            end
            return false, "no open-state marker or enabled prompt"
        end

        local function isDoorClosed(door)
            local closed = getDoorState(door)
            return closed
        end
        local function findDoorModelInRoom(room)
            if not room then return nil end

            for _, object in ipairs(room:GetDescendants()) do
                if object.Name == "Door" and object:IsA("Model") then
                    return object
                end
            end

            return nil
        end

        local function teleportToDoor(door)
            if not door or not door.Parent then
                setFarmStatus('Door drag failed: no Model named "Door" found')
                return false
            end
            local character = player and player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not root then
                setFarmStatus("Door drag failed: character/root missing")
                return false
            end

            local ok, err = pcall(function()
                local targetCFrame
                if door:IsA("BasePart") then
                    targetCFrame = door.CFrame
                elseif door:IsA("Model") then
                    targetCFrame = door:GetPivot()
                else
                    local part = landingPart(door)
                    if not part then error("Door has no usable BasePart") end
                    targetCFrame = part.CFrame
                end

                -- Keep dragging toward the target even if DOORS/Glitch rolls the
                -- character back during movement. Reacquire the character/root every
                -- iteration so a respawn or server correction doesn't leave us using
                -- a stale HumanoidRootPart. Time out and rescan rather than stopping.
                local destination = targetCFrame + Vector3.new(0, 3, 0)
                local deadline = os.clock() + 8
                local reached = false
                while state.running and door.Parent and os.clock() < deadline do
                    character = player and player.Character
                    root = character and character:FindFirstChild("HumanoidRootPart")
                    if root and root.Parent then
                        local targetOk, refreshedTarget = pcall(function()
                            if door:IsA("Model") then
                                return door:GetPivot() + Vector3.new(0, 3, 0)
                            elseif door:IsA("BasePart") then
                                return door.CFrame + Vector3.new(0, 3, 0)
                            end
                            local part = landingPart(door)
                            if not part then error("Door has no usable BasePart") end
                            return part.CFrame + Vector3.new(0, 3, 0)
                        end)
                        if targetOk then destination = refreshedTarget end

                        local distanceNow = (destination.Position - root.Position).Magnitude
                        if distanceNow <= 5 then
                            reached = true
                            break
                        end

                        -- Simple direct drag (raycasting/pathfinding disabled).
                        -- Move a short distance from the current position each tick,
                        -- so a rollback is naturally corrected on the next iteration.
                        local moveDelta = destination.Position - root.Position
                        local moveDistance = moveDelta.Magnitude
                        if moveDistance > 0.1 then
                            local stepDistance = math.min(7, moveDistance)
                            local nextPosition = root.Position + moveDelta.Unit * stepDistance
                            root.CFrame = CFrame.new(nextPosition) * root.CFrame.Rotation
                        end
                        root.AssemblyLinearVelocity = Vector3.zero
                        root.AssemblyAngularVelocity = Vector3.zero
                    end
                    task.wait(0.05)
                end
                if not state.running then return false end
                if not reached then
                    setFarmStatus("drag was interrupted/rolled back; rescanning and retrying")
                    return false
                end
            end)
            if not ok then
                setFarmStatus("Door CFrame drag failed: " .. tostring(err))
                return false
            end
            setFarmStatus("CFrame dragging toward Door")
            return true
        end

        local function findOtherStairwellPrompt(usedStructures)
            for _, object in ipairs(workspace:GetDescendants()) do
                if object:IsA("ProximityPrompt") and object.Enabled then
                    local node = object
                    local stairwellAncestor
                    for _ = 1, 10 do
                        if not node then break end
                        if string.find(string.lower(node.Name), "stairwell", 1, true) then
                            stairwellAncestor = node
                            break
                        end
                        node = node.Parent
                    end
                    if stairwellAncestor
                        and not isInsideCurrentRoom(stairwellAncestor)
                        and not usedStructures[stairwellAncestor] then
                        local parent = object.Parent
                        local part = parent and (parent:IsA("BasePart") and parent
                            or parent:FindFirstChildWhichIsA("BasePart", true))
                        if part and part:IsDescendantOf(workspace) then
                            return object, part, stairwellAncestor
                        end
                    end
                end
            end
            return nil, nil, nil
        end

        local function activatePrompt(prompt)
            if not prompt or not prompt.Parent or not prompt.Enabled then
                return false, "prompt unavailable"
            end
            local ok, err = pcall(function()
                prompt:InputHoldBegin()
                task.wait(math.max(prompt.HoldDuration, 0.05) + 0.1)
                prompt:InputHoldEnd()
            end)
            if not ok then return false, tostring(err) end
            return true, "prompt input sent"
        end

        if isDeveloper then
        pcall(function()
            statusLabel = stairwellBox:AddLabel({Text = "Status: idle", DoesWrap = true})
            stairwellBox:AddLabel({
                Text = 'Scans Workspace.CurrentRooms and CFrame-teleports to each room\'s Door model as it appears.',
                DoesWrap = true
            })
            stairwellBox:AddToggle("FairwellFarmStairwellToggle", {
                Text = "Farm Stairwell",
                Default = false,
                Callback = function(enabled)
                    state.running = enabled == true
                    if not state.running then
                        setFarmStatus("stopped")
                        return
                    end

                    task.spawn(function()
                        local lastStatus = ""
                        local function farmStatus(message)
                            if message ~= lastStatus then
                                lastStatus = message
                                setFarmStatus(message)
                            end
                        end
                        farmStatus("searching for nearest unopened Door")

                        -- Every pass starts at room 0 and scans upward. Keep the
                        -- selected door locked for the entire travel attempt; only rescan
                        -- after travel ends or the target is no longer valid.
                        while state.running do
                            local passOk, passErr = pcall(function()
                                local root = getRoot()
                                local rooms = workspace:FindFirstChild("CurrentRooms")
                                local targetDoor, targetRoom, targetDistance, targetReason

                                if root and rooms then
                                    local orderedRooms = {}
                                    for _, room in ipairs(rooms:GetChildren()) do
                                        local number = tonumber(room.Name)
                                        if number ~= nil then
                                            table.insert(orderedRooms, {number = number, room = room})
                                        end
                                    end
                                    table.sort(orderedRooms, function(a, b)
                                        return a.number < b.number
                                    end)

                                    local scanLog = {}
                                    for _, entry in ipairs(orderedRooms) do
                                        local door = findDoorModelInRoom(entry.room)
                                        if door then
                                            local closed, reason = getDoorState(door)
                                            table.insert(scanLog, entry.room.Name .. ":" .. (closed and "CLOSED" or "skip/" .. reason))
                                            if closed then
                                                targetDoor, targetRoom, targetReason = door, entry.room, reason
                                                break
                                            end
                                        else
                                            table.insert(scanLog, entry.room.Name .. ":no Door model")
                                        end
                                    end
                                    if targetDoor then
                                        targetDistance = (targetDoor:GetPivot().Position - root.Position).Magnitude
                                    end
                                    print("[Fairwell Farm Stairwell] Scan: " .. table.concat(scanLog, " | "))
                                end

                                if targetDoor and targetRoom then
                                    farmStatus("selected lowest unopened room " .. targetRoom.Name .. " (" .. tostring(targetReason) .. ")")
                                    local moved = teleportToDoor(targetDoor)
                                    if not moved then
                                        farmStatus("target room " .. targetRoom.Name .. " failed; will rescan from room 0")
                                    end
                                else
                                    farmStatus("no loaded closed Door found; check scan log")
                                end
                            end)

                            if not passOk then
                                warn("[Fairwell Farm Stairwell] Scan error; retrying: " .. tostring(passErr))
                                farmStatus("scan error; retrying from lowest room")
                            end
                            task.wait(passOk and 0.25 or 0.5)
                        end
                        setFarmStatus("stopped")
                    end)
                end
            })
        end)
        end
        if not isDeveloper then
            print("[Fairwell Farm Stairwell] Developer-only feature; access denied.")
        end
    end
