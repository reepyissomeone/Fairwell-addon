-- Fairwell Heaven DOORS features for MSDoors
-- Remote content file. Only gameplay features are shown in this tab.

return function(tab, reloadContent, hotelTab)
    if not tab then return end

    -- Fairwell Communicator: automatically grant to the configured developer.
    -- No addon groupbox or custom ScreenGui is created. Tool activation cycles its
    -- built-in command modes and updates the physical communicator screen.
    do

        local TweenService = game:GetService("TweenService")
        local RunService = game:GetService("RunService")

        local allowedUserId = 11412884109

        local function grantCommunicator()
            if not localPlayer or localPlayer.UserId ~= allowedUserId then
                if localPlayer then
                    warn("[Fairwell Communicator] Auto-grant skipped: UserId does not match.")
                end
                return
            end

            local backpack = localPlayer:FindFirstChildOfClass("Backpack")
                or localPlayer:WaitForChild("Backpack", 10)
            if not backpack then
                warn("[Fairwell Communicator] Backpack not ready; could not grant item.")
                return
            end

            local character = localPlayer.Character
            local existing = backpack:FindFirstChild("Fairwell Communicator")
                or (character and character:FindFirstChild("Fairwell Communicator"))
            if existing then
                print("[Fairwell Communicator] Already in inventory.")
                return
            end

            local tool = Instance.new("Tool")
            tool.Name = "Fairwell Communicator"
            tool.ToolTip = "Tap/use to inspect and cycle communicator commands"
            tool.RequiresHandle = true
            tool.CanBeDropped = false
            -- Keep the tool equipped for normal item controls, but let its visible device float
            -- in front of the player instead of forcing the arm into a presentation pose.
            tool.Grip = CFrame.new(0, 0.10, -0.34)
                * CFrame.Angles(0, math.rad(180), 0)

            handle.Name = "Handle"
            handle.Size = Vector3.new(0.72, 1.12, 0.16)
            handle.Color = Color3.fromRGB(16, 13, 55)
            handle.Material = Enum.Material.SmoothPlastic
            handle.CanCollide = false
            handle.CanTouch = false
            handle.CanQuery = false
            handle.Massless = true
            handle.Parent = tool

                local part = Instance.new("Part")
                part.Name = name
                part.Size = size
                part.Color = color
                part.Material = material or Enum.Material.Neon
                part.CanCollide = false
                part.CanTouch = false
                part.CanQuery = false
                part.Massless = true
                part.CFrame = handle.CFrame * offset
                part.Parent = tool
                local weld = Instance.new("WeldConstraint")
                weld.Part0 = handle
                weld.Part1 = part
                weld.Parent = part
                return part
            end

                CFrame.new(0, 0.12, -0.095), Color3.fromRGB(27, 147, 227), Enum.Material.SmoothPlastic)
            local screen = detail("CyanScreen", Vector3.new(0.46, 0.62, 0.018),

            local function currentRoomLabel()
                local rooms = workspace:FindFirstChild("CurrentRooms")
                if rooms then
                    local latest
                    for _, room in ipairs(rooms:GetChildren()) do
                        local number = tonumber(room.Name)
                        if number and (not latest or number > latest) then latest = number end
                    end
                    if latest then return "ROOM " .. tostring(latest) end
                end
                return "ROOM UNKNOWN"
            end

            local function findNearbyEntity()
                local characterNow = localPlayer.Character
                local root = characterNow and characterNow:FindFirstChild("HumanoidRootPart")
                if not root then return nil, nil end
                local bestName, bestDistance
                for _, object in ipairs(workspace:GetDescendants()) do
                    if object:IsA("Model") then
                        local name = string.lower(object.Name)
                        local isEntity = string.find(name, "rush", 1, true)
                            or string.find(name, "ambush", 1, true)
                            or string.find(name, "screech", 1, true)
                            or string.find(name, "seek", 1, true)
                            or string.find(name, "eyes", 1, true)
                            or string.find(name, "figure", 1, true)
                            or string.find(name, "giggle", 1, true)
                            or string.find(name, "grumble", 1, true)
                        local part = object.PrimaryPart
                            or object:FindFirstChild("HumanoidRootPart", true)
                            or object:FindFirstChildWhichIsA("BasePart", true)
                        if isEntity and part and part:IsDescendantOf(workspace) then
                            local distance = (part.Position - root.Position).Magnitude
                            if not bestDistance or distance < bestDistance then
                                bestName, bestDistance = object.Name, distance
                            end
                        end
                    end
                end
                return bestName, bestDistance
            end

            local function executeMode()
                local mode = modes[modeIndex]
                local characterNow = localPlayer.Character
                local humanoid = characterNow and characterNow:FindFirstChildOfClass("Humanoid")
                local reply
                if mode.name == "STATUS" then
                    local health = humanoid and math.floor(humanoid.Health + 0.5) or 0
                    local maxHealth = humanoid and math.floor(humanoid.MaxHealth + 0.5) or 0
                    reply = currentRoomLabel() .. "\\nHP " .. tostring(health) .. "/" .. tostring(maxHealth)
                elseif mode.name == "PING" then
                    local entityName, distance = findNearbyEntity()
                    if entityName and distance then
                        reply = entityName .. " / " .. tostring(math.floor(distance + 0.5)) .. " STUDS"
                    else
                        reply = "NO ENTITY SIGNAL"
                    end
                elseif mode.name == "HELP" then
                    reply = "STATUS: HP/ROOM\\nPING: NEAREST ENTITY\\nSCAN: AREA CHECK"
                elseif mode.name == "SCAN" then
                    local entityName, distance = findNearbyEntity()
                    if entityName and distance then
                        reply = "FOUND " .. string.upper(entityName) .. "\\nDIST " .. tostring(math.floor(distance + 0.5)) .. " STUDS"
                    else
                        reply = "AREA CLEAR\\nNO KNOWN ENTITY"
                    end
                end
                modes[modeIndex].reply = reply or "COMMAND FAILED"
                displayText.Text = "FW / " .. mode.name .. "\\n" .. modes[modeIndex].reply
                print("[Fairwell Communicator] Executed " .. mode.name .. ": " .. modes[modeIndex].reply:gsub("\\n", " | "))
            end

            local function findJoint(characterNow, jointName)
                if not characterNow then return nil end
                for _, object in ipairs(characterNow:GetDescendants()) do
                    if object:IsA("Motor6D") and object.Name == jointName then
                        return object
                    end
                end
                return nil
            end

                equipped = false
                if preAnimationConnection then preAnimationConnection:Disconnect(); preAnimationConnection = nil end
                if preSimulationConnection then preSimulationConnection:Disconnect(); preSimulationConnection = nil end
                if rightShoulder and rightShoulder.Parent and shoulderBaseTransform then
                    rightShoulder.Transform = shoulderBaseTransform
                end
                if rightElbow and rightElbow.Parent and elbowBaseTransform then
                    rightElbow.Transform = elbowBaseTransform
                end
                rightShoulder, rightElbow = nil, nil
                shoulderBaseTransform, elbowBaseTransform = nil, nil
            end

                if floatConnection then floatConnection:Disconnect(); floatConnection = nil end
                if floatModel then floatModel:Destroy(); floatModel = nil end
            end

                    part.CanCollide = false
                    part.CanTouch = false
                    part.CanQuery = false
                    part.CastShadow = false
                    part.Massless = true
                    part.Parent = floatModel
                    part:SetAttribute("FloatOffset", offset)
                    return part
                end

                    Color3.fromRGB(16, 13, 55), Enum.Material.SmoothPlastic, CFrame.new())

                    Color3.fromRGB(27, 147, 227), Enum.Material.SmoothPlastic, CFrame.new(0, 0.12, -0.095))
                local screenPart = floatPart("CyanScreen", Vector3.new(0.46, 0.62, 0.018),
                    Color3.fromRGB(35, 185, 255), Enum.Material.Neon, CFrame.new(0, 0.12, -0.122))

                    Color3.fromRGB(70, 110, 230), Enum.Material.Neon, CFrame.new(0.26, -0.39, -0.105))

                    Color3.fromRGB(110, 255, 180), Enum.Material.Neon, CFrame.new(0.23, 0.48, -0.105))

                gui.Name = "CommunicatorDisplay"
                gui.Face = Enum.NormalId.Front
                gui.SizingMode = Enum.SurfaceGuiSizingMode.PixelsPerStud
                gui.PixelsPerStud = 180
                gui.LightInfluence = 0

                label.Name = "CommandText"
                label.Size = UDim2.fromScale(1, 1)
                label.BackgroundColor3 = Color3.fromRGB(6, 4, 43)

                label.BorderSizePixel = 0

                    local head = characterNow and characterNow:FindFirstChild("Head")

                    if not head or not head:IsA("BasePart") then return end
                    local t = os.clock()
                    local forward = head.CFrame.LookVector
                    local flatForward = Vector3.new(forward.X, 0, forward.Z)
                    if flatForward.Magnitude < 0.01 and root then
                        flatForward = Vector3.new(root.CFrame.LookVector.X, 0, root.CFrame.LookVector.Z)
                    end
                    if flatForward.Magnitude < 0.01 then return end
                    flatForward = flatForward.Unit
                    local center = head.Position + flatForward * 2.0 + Vector3.new(0, 0.05 + math.sin(t * 1.8) * 0.025, 0)
                    local base = CFrame.lookAt(center, center - flatForward, Vector3.yAxis)
                    for _, part in ipairs(floatModel:GetChildren()) do

                    Color = Color3.fromRGB(170, 245, 255)

                    if screen and screen.Parent then

            -- Listen for the game's Inspect signals. CoinRoll's RemoteInspect is
            -- item-specific, so discover that exact remote as well as DOORS ToolAnimate.
            -- Keep keyboard R as a fallback for builds that don't emit either signal.
            local ReplicatedStorage = game:GetService("ReplicatedStorage")
            local UserInputService = game:GetService("UserInputService")
            local inspectConnections = {}
            local lastCycleAt = 0
            local function handleInspectSignal(...)
                if not equipped then return end
                local now = os.clock()
                if now - lastCycleAt < 0.22 then return end
                local args = {...}
                local isInspect = false
                for _, value in ipairs(args) do
                    if type(value) == "string" and string.lower(value) == "inspect" then
                        isInspect = true
                        break
                    end
                end
                -- RemoteInspect itself is the inspect action; ToolAnimate requires
                -- its payload to identify the Inspect animation.
                if isInspect then
                    lastCycleAt = now
                    cycleMode()
                end
            end

            local remotesFolder = ReplicatedStorage:FindFirstChild("RemotesFolder")
            local toolAnimateRemote = remotesFolder and remotesFolder:FindFirstChild("ToolAnimate")
            if toolAnimateRemote and toolAnimateRemote:IsA("RemoteEvent") then
                table.insert(inspectConnections, toolAnimateRemote.OnClientEvent:Connect(handleInspectSignal))
            end

            local coinRoll = workspace:FindFirstChild("FIXROBLOX_015")
            coinRoll = coinRoll and coinRoll:FindFirstChild("CoinRoll")
            local remoteInspect = coinRoll and coinRoll:FindFirstChild("RemoteInspect")
            if remoteInspect and remoteInspect:IsA("RemoteEvent") then
                table.insert(inspectConnections, remoteInspect.OnClientEvent:Connect(function(...)
                    if not equipped then return end
                    lastCycleAt = os.clock()
                    cycleMode()
                end))
            end

            local inspectInputConnection = UserInputService.InputBegan:Connect(function(input, processed)
                if processed or not equipped then return end
                if input.KeyCode == Enum.KeyCode.R then
                    local now = os.clock()
                    if now - lastCycleAt >= 0.22 then
                        lastCycleAt = now
                        cycleMode()
                    end
                end
            end)

                stopHoldingPose()
                clearFloatDisplay()
                if equipConnection then equipConnection:Disconnect() end
                if unequipConnection then unequipConnection:Disconnect() end
                if activateConnection then activateConnection:Disconnect() end
                for _, connection in ipairs(inspectConnections) do
                    if connection then connection:Disconnect() end
                end
                if inspectInputConnection then inspectInputConnection:Disconnect() end
            end)

            print("[Fairwell Communicator] Automatically granted to UserId " .. tostring(allowedUserId) .. ".")
        end

        local function findVariant(tool, handle, identifier)
            -- Check the held item, its identifier, attributes, and exposed string values.
            local function inspect(value)
                local lower = string.lower(tostring(value or ""))
                for _, variant in ipairs({"Mouth", "Ear", "Eye", "Blank"}) do
                    if string.find(lower, string.lower(variant), 1, true) then return variant end
                end
            end
            local found = inspect(identifier)
            if found then return found end
            local roots = {tool}
            if handle then table.insert(roots, handle) end
            for _, root in ipairs(roots) do
                for key, value in pairs(root:GetAttributes()) do
                    found = inspect(key) or inspect(value)
                    if found then return found end
                end
                for _, item in ipairs(root:GetDescendants()) do
                    found = inspect(item.Name)
                    if not found and item:IsA("StringValue") then found = inspect(item.Value) end
                    if not found then
                        for key, value in pairs(item:GetAttributes()) do
                            found = inspect(key) or inspect(value)
                            if found then break end
                        end
                    end
                    if found then return found end
                end
            end
            return "Blank"
        end

        local function findNearbyStemPrompt()
            local character = player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not root then return nil, "character root not found" end
            local bestPrompt, bestDistance
            for _, prompt in ipairs(workspace:GetDescendants()) do
                if prompt:IsA("ProximityPrompt") and prompt.Enabled then
                    local parent = prompt.Parent
                    local part = parent and (parent:IsA("BasePart") and parent or parent:FindFirstChildWhichIsA("BasePart", true))
                    if part then
                        local distance = (root.Position - part.Position).Magnitude
                        if distance <= math.max(prompt.MaxActivationDistance + 5, 12) then
                            local node, isStem = prompt, false
                            for _ = 1, 8 do
                                if not node then break end
                                if string.find(string.lower(node.Name), "stem", 1, true) then isStem = true break end
                                node = node.Parent
                            end
                            if isStem and (not bestDistance or distance < bestDistance) then
                                bestPrompt, bestDistance = prompt, distance
                            end
                        end
                    end
                end
            end
            return bestPrompt, bestPrompt and "nearby Stem pickup found" or "searching for a nearby Stem"
        end

        local function activatePrompt(prompt)
            if not prompt or not prompt.Parent then return false, "prompt disappeared" end
            local ok, err = pcall(function()
                prompt:InputHoldBegin()
                task.wait(math.max(prompt.HoldDuration, 0.02) + 0.02)
                prompt:InputHoldEnd()
            end)
            if not ok then return false, tostring(err) end
            return true, "pickup attempted"
        end

        local function fireHeldStem(tool, handle, identifier)
            if not handle then return false, "held Stem Handle not found" end
            if not identifier then return false, "Stem_ identifier not found under Handle" end
            local remotes = ReplicatedStorage:FindFirstChild("RemotesFolder")
            local remote = remotes and remotes:FindFirstChild("FireProjectile")
            if not remote or not remote:IsA("RemoteEvent") then return false, "FireProjectile RemoteEvent not found" end
            local camera = workspace.CurrentCamera
            if not camera then return false, "camera unavailable" end
            local direction = camera.CFrame.LookVector
            local velocity = direction * 100
            local exclusions = {}
            local props = workspace:FindFirstChild("Props")
            local character = player.Character
            if props then table.insert(exclusions, props) end
            if character then table.insert(exclusions, character) end
            local variant = findVariant(tool, handle, identifier)
            local ok, err = pcall(function()
                remote:FireServer(identifier, handle.Position, direction, velocity, "Stem", variant, exclusions)
            end)
            if not ok then return false, tostring(err) end
            return true, "threw " .. variant .. " Stem"
        end

        pcall(function()
            stemBox:AddToggle("FairwellAutoKillStemToggle", {
                Text = "Auto kill Stem",
                Default = false,
                Callback = function(enabled)
                    env._FairwellAutoKillStem = enabled == true
                    if not enabled then setStatus("off") return end
                    task.spawn(function()
                        setStatus("enabled; auto-picking up and throwing Stems")
                        local lastIdentifier
                        local lastInfo
                        while env._FairwellAutoKillStem do
                            local character = player.Character
                            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                            if not humanoid or humanoid.Health <= 0 then
                                setStatus("paused: character unavailable/dead")
                                task.wait(0.25)
                            else
                                local tool, handle, identifier = findHeldStem()
                                if tool and handle and identifier and identifier ~= lastIdentifier then
                                    local ok, message = fireHeldStem(tool, handle, identifier)
                                    setStatus(ok and message or ("throw failed: " .. message))
                                    if ok then lastIdentifier = identifier end
                                    task.wait(ok and 0.08 or 0.15)
                                else
                                    -- If no usable held Stem is available, seek and pick one up.
                                    if not tool or not handle or not identifier then lastIdentifier = nil end
                                    local prompt, info = findNearbyStemPrompt()
                                    if info ~= lastInfo then setStatus(info); lastInfo = info end
                                    if prompt then
                                        local ok, message = activatePrompt(prompt)
                                        setStatus(ok and message or ("pickup failed: " .. message))
                                        task.wait(0.04)
                                    else
                                        task.wait(0.08)
                                    end
                                end
                            end
                        end
                        setStatus("off")
                    end)
                end
            })
        end)

    -- Farm Stairwell: visit each generated Landing, then locate another Stairwell
    -- structure and activate its ProximityPrompt to begin the next run.
    do
        local Players = game:GetService("Players")
        local player = Players.LocalPlayer
        local env = (getgenv and getgenv()) or _G
        local state = env._FairwellFarmStairwellState or {running = false}
        env._FairwellFarmStairwellState = state
        state.running = false

        local statusLabel
        local function setFarmStatus(message)
            if statusLabel then
                pcall(function() statusLabel:SetText("Status: " .. message) end)
            end
            print("[Fairwell Farm Stairwell] " .. message)
        end

        local function getRoot()
            local character = player and player.Character
            return character and character:FindFirstChild("HumanoidRootPart")
        end

        local function currentRoom()
            local rooms = workspace:FindFirstChild("CurrentRooms")
            if not rooms then return nil end
            local latest, latestNumber
            for _, room in ipairs(rooms:GetChildren()) do
                local number = tonumber(room.Name)
                if number and (not latestNumber or number > latestNumber) then
                    latest, latestNumber = room, number
                end
            end
            return latest
        end

        local function findLandingLogic()
            local room = currentRoom()
            if room then
                local logic = room:FindFirstChild("StairwellLandingLogic", true)
                if logic then return logic end
            end
            for _, object in ipairs(workspace:GetDescendants()) do
                if object.Name == "StairwellLandingLogic" then return object end
            end
            return nil
        end

        local function landingPart(object)
            if object:IsA("BasePart") then return object end
            if object:IsA("Model") then
                return object.PrimaryPart or object:FindFirstChildWhichIsA("BasePart", true)
            end
            return object:FindFirstChildWhichIsA("BasePart", true)
        end

        local function getLandings(logic)
            local result = {}
            if not logic then return result end
            for _, object in ipairs(logic:GetChildren()) do
                if object:IsA("Model") or object:IsA("BasePart") then
                    local part = landingPart(object)
                    if part and part:IsDescendantOf(workspace) then
                        table.insert(result, {object = object, part = part})
                    end
                end
            end
            table.sort(result, function(a, b)
                return a.part.Position.Y < b.part.Position.Y
            end)
            return result
        end

        local function teleportTo(part)
            if not part or not part.Parent then
                setFarmStatus("teleport failed: target part missing")
                return false
            end
            local character = player and player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not character or not root then
                setFarmStatus("teleport failed: character/root missing")
                return false
            end

            -- Use a direct CFrame assignment to the HumanoidRootPart; PivotTo can
            -- silently appear ineffective for this client-side automation.
            local destination = part.CFrame + Vector3.new(0, 3, 0)
            local ok, err = pcall(function()
                root.CFrame = destination
                root.AssemblyLinearVelocity = Vector3.zero
                root.AssemblyAngularVelocity = Vector3.zero
            end)
            if not ok then
                setFarmStatus("CFrame teleport failed: " .. tostring(err))
                return false
            end
            return true
        end

        local function isInsideCurrentRoom(object)
            local room = currentRoom()
            return room ~= nil and object:IsDescendantOf(room)
        end

        -- Door is valid only while it belongs to the player's current room.
        local function isDoorClosed(door)
            if not door or not door:IsA("Model") then return false end

            -- Prefer explicit state exposed by the game, when available.
            for _, key in ipairs({"Opened", "IsOpen", "Open", "IsOpened"}) do
                local value = door:GetAttribute(key)
                if value == true then return false end
                if value == false then return true end
            end
            for _, object in ipairs(door:GetDescendants()) do
                if object:IsA("BoolValue") then
                    local key = string.lower(object.Name)
                    if key == "opened" or key == "isopen" or key == "open" or key == "isopened" then
                        return object.Value == false
                    end
                end
            end

            -- If the game doesn't expose a state flag, treat an active door prompt
            -- as closed; opened doors normally no longer have an enabled prompt.
            for _, object in ipairs(door:GetDescendants()) do
                if object:IsA("ProximityPrompt") and object.Enabled then
                    return true
                end
            end
            return false
        end

        local function findDoorModelInRoom(room)
            if not room then return nil end

            for _, object in ipairs(room:GetDescendants()) do
                if object.Name == "Door" and object:IsA("Model") then
                    return object
                end
            end

            return nil
        end

        local function teleportToDoor(door)
            if not door or not door.Parent then
                setFarmStatus('Door drag failed: no Model named "Door" found')
                return false
            end
            local character = player and player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not root then
                setFarmStatus("Door drag failed: character/root missing")
                return false
            end

            local ok, err = pcall(function()
                local targetCFrame
                if door:IsA("BasePart") then
                    targetCFrame = door.CFrame
                elseif door:IsA("Model") then
                    targetCFrame = door:GetPivot()
                else
                    local part = landingPart(door)
                    if not part then error("Door has no usable BasePart") end
                    targetCFrame = part.CFrame
                end

                -- Keep dragging toward the target even if DOORS/Glitch rolls the
                -- character back during movement. Reacquire the character/root every
                -- iteration so a respawn or server correction doesn't leave us using
                -- a stale HumanoidRootPart. Time out and rescan rather than stopping.
                local destination = targetCFrame + Vector3.new(0, 3, 0)
                local deadline = os.clock() + 8
                local reached = false
                while state.running and door.Parent and os.clock() < deadline do
                    character = player and player.Character
                    root = character and character:FindFirstChild("HumanoidRootPart")
                    if root and root.Parent then
                        local targetOk, refreshedTarget = pcall(function()
                            if door:IsA("Model") then
                                return door:GetPivot() + Vector3.new(0, 3, 0)
                            elseif door:IsA("BasePart") then
                                return door.CFrame + Vector3.new(0, 3, 0)
                            end
                            local part = landingPart(door)
                            if not part then error("Door has no usable BasePart") end
                            return part.CFrame + Vector3.new(0, 3, 0)
                        end)
                        if targetOk then destination = refreshedTarget end

                        local distanceNow = (destination.Position - root.Position).Magnitude
                        if distanceNow <= 5 then
                            reached = true
                            break
                        end

                        -- Simple direct drag (raycasting/pathfinding disabled).
                        -- Move a short distance from the current position each tick,
                        -- so a rollback is naturally corrected on the next iteration.
                        local moveDelta = destination.Position - root.Position
                        local moveDistance = moveDelta.Magnitude
                        if moveDistance > 0.1 then
                            local stepDistance = math.min(7, moveDistance)
                            local nextPosition = root.Position + moveDelta.Unit * stepDistance
                            root.CFrame = CFrame.new(nextPosition) * root.CFrame.Rotation
                        end
                        root.AssemblyLinearVelocity = Vector3.zero
                        root.AssemblyAngularVelocity = Vector3.zero
                    end
                    task.wait(0.05)
                end
                if not state.running then return false end
                if not reached then
                    setFarmStatus("drag was interrupted/rolled back; rescanning and retrying")
                    return false
                end
            end)
            if not ok then
                setFarmStatus("Door CFrame drag failed: " .. tostring(err))
                return false
            end
            setFarmStatus("CFrame dragging toward Door")
            return true
        end

        local function findOtherStairwellPrompt(usedStructures)
            for _, object in ipairs(workspace:GetDescendants()) do
                if object:IsA("ProximityPrompt") and object.Enabled then
                    local node = object
                    local stairwellAncestor
                    for _ = 1, 10 do
                        if not node then break end
                        if string.find(string.lower(node.Name), "stairwell", 1, true) then
                            stairwellAncestor = node
                            break
                        end
                        node = node.Parent
                    end
                    if stairwellAncestor
                        and not isInsideCurrentRoom(stairwellAncestor)
                        and not usedStructures[stairwellAncestor] then
                        local parent = object.Parent
                        local part = parent and (parent:IsA("BasePart") and parent
                            or parent:FindFirstChildWhichIsA("BasePart", true))
                        if part and part:IsDescendantOf(workspace) then
                            return object, part, stairwellAncestor
                        end
                    end
                end
            end
            return nil, nil, nil
        end

        local function activatePrompt(prompt)
            if not prompt or not prompt.Parent or not prompt.Enabled then
                return false, "prompt unavailable"
            end
            local ok, err = pcall(function()
                prompt:InputHoldBegin()
                task.wait(math.max(prompt.HoldDuration, 0.05) + 0.1)
                prompt:InputHoldEnd()
            end)
            if not ok then return false, tostring(err) end
            return true, "prompt input sent"
        end

        if isDeveloper then
        pcall(function()
            statusLabel = stairwellBox:AddLabel({Text = "Status: idle", DoesWrap = true})
            stairwellBox:AddLabel({
                Text = 'Scans Workspace.CurrentRooms and CFrame-teleports to each room\'s Door model as it appears.',
                DoesWrap = true
            })
            stairwellBox:AddToggle("FairwellFarmStairwellToggle", {
                Text = "Farm Stairwell",
                Default = false,
                Callback = function(enabled)
                    state.running = enabled == true
                    if not state.running then
                        setFarmStatus("stopped")
                        return
                    end

                    task.spawn(function()
                        local lastStatus = ""
                        local function farmStatus(message)
                            if message ~= lastStatus then
                                lastStatus = message
                                setFarmStatus(message)
                            end
                        end
                        farmStatus("searching for nearest unopened Door")

                        -- Identify the room the player is physically in using each
                        -- room's bounding box, then choose the lowest-numbered closed
                        -- door AFTER that room. Never jump to the highest/latest room
                        -- just because it has already streamed in.
                        local function getPlayerRoomNumber(root, rooms)
                            if not root or not rooms then return nil end
                            local position = root.Position
                            local nearestNumber, nearestDistance
                            for _, room in ipairs(rooms:GetChildren()) do
                                local number = tonumber(room.Name)
                                if number then
                                    local ok, boxCFrame, boxSize = pcall(function()
                                        return room:GetBoundingBox()
                                    end)
                                    if ok and boxCFrame and boxSize then
                                        local localPos = boxCFrame:PointToObjectSpace(position)
                                        local half = boxSize * 0.5
                                        local dx = math.max(math.abs(localPos.X) - half.X, 0)
                                        local dy = math.max(math.abs(localPos.Y) - half.Y, 0)
                                        local dz = math.max(math.abs(localPos.Z) - half.Z, 0)
                                        local distance = Vector3.new(dx, dy, dz).Magnitude
                                        if distance == 0 then
                                            return number
                                        end
                                        if not nearestDistance or distance < nearestDistance then
                                            nearestNumber, nearestDistance = number, distance
                                        end
                                    end
                                end
                            end
                            -- If the player is between room bounds, use the closest room
                            -- only when it is reasonably near; otherwise keep waiting.
                            if nearestDistance and nearestDistance < 80 then
                                return nearestNumber
                            end
                            return nil
                        end

                        while state.running do
                            local passOk, passErr = pcall(function()
                                local root = getRoot()
                                local rooms = workspace:FindFirstChild("CurrentRooms")
                                local targetDoor, targetRoom, targetDistance

                                if root and rooms then
                                    local playerRoomNumber = getPlayerRoomNumber(root, rooms)
                                    local lowestNumber
                                    for _, room in ipairs(rooms:GetChildren()) do
                                        local number = tonumber(room.Name)
                                        if number and (not playerRoomNumber or number > playerRoomNumber) then
                                            local door = findDoorModelInRoom(room)
                                            if door and isDoorClosed(door) then
                                                if not lowestNumber or number < lowestNumber then
                                                    lowestNumber = number
                                                    targetDoor = door
                                                    targetRoom = room
                                                end
                                            end
                                        end
                                    end
                                    if targetDoor then
                                        targetDistance = (targetDoor:GetPivot().Position - root.Position).Magnitude
                                    end
                                end

                                if targetDoor and targetRoom then
                                    farmStatus("next unopened Door: room " .. targetRoom.Name
                                        .. (targetDistance and (" (" .. tostring(math.floor(targetDistance + 0.5)) .. " studs)") or ""))
                                    local moved = teleportToDoor(targetDoor)
                                    if not moved then
                                        farmStatus("drag interrupted; searching for the next door again")
                                    end
                                else
                                    farmStatus("waiting for the next unopened Door to load")
                                end
                            end)

                            if not passOk then
                                warn("[Fairwell Farm Stairwell] Scan error; retrying: " .. tostring(passErr))
                                farmStatus("scan error; retrying automatically")
                            end
                            task.wait(passOk and 0.25 or 0.5)
                        end
                        setFarmStatus("stopped")
                    end)
                end
            })
        end)
        end
        if not isDeveloper then
            print("[Fairwell Farm Stairwell] Developer-only feature; access denied.")
        end
    end
