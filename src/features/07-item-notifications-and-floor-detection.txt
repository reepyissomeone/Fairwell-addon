        -- Detect current floor/subfloor using active workspace names and room structure.
        -- Order matters: more specific subfloors must win over generic room folders.
        local function currentFloorInfo()
            local allNames = {}
            local function addName(obj)
                if obj then allNames[string.lower(obj.Name)] = true end
            end
            for _, obj in ipairs(workspace:GetChildren()) do addName(obj) end

            -- Inspect only names, not arbitrary text or item descriptions.
            -- Include nested room containers because some MSDoors versions place
            -- the active floor marker below Workspace rather than at its root.
            local activeRooms = workspace:FindFirstChild("CurrentRooms")
                or workspace:FindFirstChild("Rooms")
            local activeRoomNumber, activeRoom
            if activeRooms then
                for _, room in ipairs(activeRooms:GetChildren()) do
                    local number = tonumber(room.Name:match("%d+"))
                    if number and (not activeRoomNumber or number > activeRoomNumber) then
                        activeRoomNumber, activeRoom = number, room
                    end
                end
            end

            local roomNames = {}
            if activeRoom then
                table.insert(roomNames, string.lower(activeRoom.Name))
                for _, obj in ipairs(activeRoom:GetDescendants()) do
                    table.insert(roomNames, string.lower(obj.Name))
                end
            end

            local function anyNameContains(needles, includeRoomNames)
                for name in pairs(allNames) do
                    for _, needle in ipairs(needles) do
                        if string.find(name, needle, 1, true) then return true end
                    end
                end
                if includeRoomNames then
                    for _, name in ipairs(roomNames) do
                        for _, needle in ipairs(needles) do
                            if string.find(name, needle, 1, true) then return true end
                        end
                    end
                end
                return false
            end

            local floor, subfloor = "Unknown", "Unknown"
            local roomLabel = activeRoomNumber and string.format("Room %03d", activeRoomNumber) or nil

            if anyNameContains({"stairwell", "stairwellrooms"}, false) then
                floor, subfloor = "The Stairwell", "The Stairwell"
            elseif anyNameContains({"archives", "sectora", "sectorz", "a-000", "a000"}, true) then
                floor, subfloor = "The Archives", "The Archives"
            elseif anyNameContains({"backdoor", "backdoors"}, false) then
                floor, subfloor = "The Backdoor", "The Backdoor"
            elseif anyNameContains({"outdoors", "groundskeeper", "bramble", "eyestalk"}, false) then
                floor, subfloor = "The Outdoors", "The Outdoors"
            elseif anyNameContains({"mines", "minecart", "the nest", "nestroom", "damroom"}, false) then
                floor, subfloor = "The Mines", "The Mines"
            elseif anyNameContains({"hotel-", "hotelminus", "hotelminusvision"}, false) then
                floor, subfloor = "Hotel-", "Hotel- Vision"
            elseif anyNameContains({"lobby", "lobbyfloor", "lobbyroom", "lobbyrooms"}, false) then
                -- Lobby markers must be checked before the generic CurrentRooms fallback;
                -- MSDoors can keep a CurrentRooms folder present while the player is in the lobby.
                floor, subfloor = "Lobby", "Lobby"
            elseif anyNameContains({"hotel-", "hotelminus", "hotelminusvision"}, false) then
                floor, subfloor = "Hotel-", "Hotel- Vision"
            elseif anyNameContains({"hotel"}, false) then
                floor, subfloor = "The Hotel", "The Hotel"
            elseif anyNameContains({"rooms", "a-000", "a000"}, false) then
                floor, subfloor = "The Rooms", "The Rooms Vision"
            elseif activeRooms then
                floor, subfloor = "The Hotel", "The Hotel"
            end

            -- The Mines contains named regions; only label a subfloor when an
            -- actual region marker is found in the active room's descendants.
            if floor == "The Mines" then
                if anyNameContains({"stairwell"}, true) then
                    subfloor = "The Stairwell"
                elseif anyNameContains({"nest", "grumble"}, true) then
                    subfloor = "The Mines — The Nest"
                elseif anyNameContains({"dam"}, true) then
                    subfloor = "The Mines — The Dam"
                end
            end

            -- Report the active room number when the current folder uses numeric names.
            return floor, subfloor, roomLabel
        end

            vitaminspickup="Vitamins", vitaminsmodel="Vitamins",
            glowstickpickup="Glowstick", glowstickmodel="Glowstick"
        }
        local function candidate(obj)
            if not obj or not obj.Parent then return false end

            -- Reject room mechanisms and scenery even if a nested child happens to
            -- share an item name (e.g. TimeLever/Lever, Candle Medium, Ripped Wall).
            local node = obj
            for _ = 1, 8 do
                if not node then break end
                local ancestorName = norm(node.Name)
                if string.find(ancestorName, "timelever", 1, true)
                    or string.find(ancestorName, "candlemedium", 1, true)
                    or string.find(ancestorName, "rippedwall", 1, true) then
                    return false
                end
                node = node.Parent
            end

            -- Exact names only. Prefix/suffix matching and generic "item"/"pickup"
            -- prompt heuristics caused scenery and room mechanisms to trigger alerts.
            local n = norm(obj.Name)
            if not known[n] and not itemAliases[n] then return false end
            return obj:IsA("Model") or obj:IsA("Tool")
        end
        pcall(function()
            settingsBox:AddDropdown("FairwellItemNotificationChoices", {
                Text = "Notify for these items",
                Values = itemChoices,
                Default = (function()
                    local defaults = {}
                    for _, choice in ipairs(itemChoices) do
                        if importantItems[norm(choice)] then table.insert(defaults, choice) end
                    end
                    return defaults
                end)(),
                Multi = true,
                AllowNull = true,
                Callback = function(values)
                    selectedItemNames = {}
                    if type(values) == "table" then
                        for key, value in pairs(values) do
                            if type(key) == "number" and type(value) == "string" then
                                selectedItemNames[value] = true
                            elseif type(key) == "string" and value == true then
                                selectedItemNames[key] = true
                            end
                        end
                    end
                    setStatus("selection updated: " .. tostring((function()
                        local n = 0
                        for _ in pairs(selectedItemNames) do n += 1 end
                        return n
                    end)()) .. " item types")
                end
            })
        end)
        pcall(function()
            settingsBox:AddToggle("FairwellAllFloorItemNotifications", {
                Text="Item Notifications", Default=false,
                Callback=function(enabled)
                    itemNotifyState.enabled = enabled == true
                    seen = {}
                    startedAt = os.clock()
                    setStatus(itemNotifyState.enabled and "scanning all floors" or "off")
                end
            })
        end)
        task.spawn(function()
            while itemNotifyState.running do
                if itemNotifyState.enabled then
                    local count = 0
                    for _, obj in ipairs(workspace:GetDescendants()) do
                        if not itemNotifyState.enabled then break end
                        if (obj:IsA("Model") or obj:IsA("Tool")) and candidate(obj) then
                            count += 1
                            local key = obj:GetFullName()
                            if not seen[key] then
                                seen[key] = true
                                local rawDetectedName = obj.Name:gsub("_", " ")
                                local detectedName = itemAliases[norm(rawDetectedName)] or rawDetectedName
                                local detectedNorm = norm(detectedName)
                                local wantsNotification = false
                                for choice in pairs(selectedItemNames) do
                                    local choiceNorm = norm(choice)
                                    if detectedNorm == choiceNorm then
                                        wantsNotification = true
                                        break
                                    end
                                end
                                local floor = currentFloor()
                                local requiredFloor
                                for choice, onlyFloor in pairs(itemFloor) do
                                    if norm(choice) == detectedNorm then requiredFloor = onlyFloor break end
                                end
                                local correctFloor = not requiredFloor or floor == requiredFloor
                                local isImportant = importantItems[detectedNorm] == true
                                if wantsNotification and correctFloor and os.clock() - startedAt > 2 then
                                    local detectedObject = obj
                                    local waitBeforePopup = popupSettings.delay
                                    task.delay(waitBeforePopup, function()
                                        if itemNotifyState.running and itemNotifyState.enabled
                                            and detectedObject and detectedObject.Parent
                                            and type(showItemPopup) == "function" then
                                            showItemPopup(detectedName)
                                        end
                                    end)
                                end
                            end
                        end
                    end
                    local floor, subfloor, roomLabel = currentFloorInfo()
                    setStatus("active: " .. floor .. " / " .. subfloor .. (roomLabel and (" / " .. roomLabel) or "") .. "; " .. tostring(count) .. " item candidates")
                end
                -- Keep scanning responsive without tying the scan loop to popup delay.
                task.wait(0.5)
            end
        end)
    end
