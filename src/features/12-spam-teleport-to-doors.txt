return function(tab, context)
    -- Spam Teleport to Doors. Only exact-name "Doors" instances are accepted.
    local Players = game:GetService("Players")
    local localPlayer = Players.LocalPlayer
    local env = context and context.Environment or ((getgenv and getgenv()) or _G)

    local previous = env._FairwellSpamTeleportDoorsState
    if previous then previous.running = false end
    local state = {running = true, enabled = false}
    env._FairwellSpamTeleportDoorsState = state

    local box
    local ok, result = pcall(function()
        return tab:AddLeftGroupbox("Movement", "move")
    end)
    if ok then box = result end
    if not box then
        pcall(function() box = tab:AddRightGroupbox("Movement", "move") end)
    end
    if not box then
        warn("[Fairwell Doors Teleport] Could not create the Movement groupbox.")
        return
    end

    local statusLabel
    local lastStatus = ""
    local function setStatus(message)
        if message == lastStatus then return end
        lastStatus = message
        if statusLabel then
            pcall(function() statusLabel:SetText("Status: " .. message) end)
        end
        print("[Fairwell Doors Teleport] " .. message)
    end

    pcall(function()
        statusLabel = box:AddLabel({Text = "Status: off", DoesWrap = true})
        box:AddLabel({
            Text = 'Repeatedly teleports to physical parts belonging to Workspace instances named exactly "Doors". No Creak targets or guessed names.',
            DoesWrap = true
        })
    end)

    local function getPivot(instance)
        if not instance or not instance.Parent then return nil end
        if instance:IsA("BasePart") then return instance.CFrame end
        if instance:IsA("Model") then
            local success, pivot = pcall(function() return instance:GetPivot() end)
            if success then return pivot end
        end
        return nil
    end

    local function collectTargets()
        local targets, seen = {}, {}
        local function addPart(part)
            if part and part:IsA("BasePart") and part.Parent and not seen[part] then
                seen[part] = true
                targets[#targets + 1] = part
            end
        end

        local success, descendants = pcall(function() return workspace:GetDescendants() end)
        if not success or type(descendants) ~= "table" then return targets end

        for _, instance in ipairs(descendants) do
            if instance.Name == "Doors" then
                if instance:IsA("BasePart") then
                    addPart(instance)
                else
                    -- Physical parts are more reliable landing points than a model's pivot.
                    local childOk, children = pcall(function() return instance:GetDescendants() end)
                    if childOk and type(children) == "table" then
                        for _, child in ipairs(children) do
                            if child:IsA("BasePart") then addPart(child) end
                        end
                    end
                    -- A part-less Model may still have a meaningful pivot.
                    if #targets == 0 and instance:IsA("Model") and getPivot(instance) then
                        targets[#targets + 1] = instance
                    end
                end
            end
        end
        return targets
    end

    pcall(function()
        box:AddToggle("FairwellSpamTeleportDoors", {
            Text = "Spam Teleport to Doors",
            Tooltip = 'Repeatedly teleports to real parts inside instances named exactly "Doors".',
            Default = false,
            Callback = function(value)
                state.enabled = value == true
                setStatus(state.enabled and 'enabled; scanning for exact "Doors" targets' or "off")
            end
        })
    end)

    task.spawn(function()
        local cachedTargets = {}
        local lastScan = 0
        local targetIndex = 0
        while state.running do
            task.wait(0.12)
            if state.enabled then
                local character = localPlayer.Character
                local root = character and character:FindFirstChild("HumanoidRootPart")
                local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                if not character or not root or not humanoid or humanoid.Health <= 0 then
                    setStatus("waiting for a living character")
                else
                    if os.clock() - lastScan >= 0.5 then
                        cachedTargets = collectTargets()
                        lastScan = os.clock()
                        targetIndex = 0
                    end

                    if #cachedTargets == 0 then
                        setStatus('no usable parts found under an exact-name "Doors" instance in Workspace')
                    else
                        local target
                        for _ = 1, #cachedTargets do
                            targetIndex = (targetIndex % #cachedTargets) + 1
                            local candidate = cachedTargets[targetIndex]
                            if candidate and candidate.Parent then
                                target = candidate
                                break
                            end
                        end

                        local targetCFrame = target and getPivot(target)
                        if target and targetCFrame then
                            local destination = targetCFrame + Vector3.new(0, 3, 0)
                            local moved = pcall(function()
                                character:PivotTo(destination)
                            end)
                            if not moved then
                                moved = pcall(function() root.CFrame = destination end)
                            end
                            if moved then
                                setStatus("teleporting to Doors (" .. tostring(targetIndex) .. "/" .. tostring(#cachedTargets) .. ")")
                            else
                                setStatus("found Doors, but the character teleport failed")
                            end
                        else
                            cachedTargets = {}
                            lastScan = 0
                            setStatus('target disappeared; rescanning exact-name "Doors" instances')
                        end
                    end
                end
            end
        end
    end)
end
