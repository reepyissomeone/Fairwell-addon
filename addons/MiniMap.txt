-- MSDoors Mini Map addon
-- Standalone client-side room-layout map. No Fairwell dependency.
-- The map uses locally available CurrentRooms models and their world positions.
-- Only rooms currently present on the client can be shown.

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")

local player = Players.LocalPlayer

local enabled = false
local showRoomNumbers = true
local showWalls = true
local showStairs = true
local rememberExplored = true
local mapSize = 220
local gui
local updateConnection
local dragConnection
local roomButtons = {}
local roomFolderConnection
local descendantConnection
local discoveredRooms = {}

local COLORS = {
    panel = Color3.fromRGB(10, 13, 24),
    border = Color3.fromRGB(38, 147, 227),
    explored = Color3.fromRGB(42, 63, 86),
    current = Color3.fromRGB(38, 147, 227),
    text = Color3.fromRGB(235, 242, 255),
    muted = Color3.fromRGB(135, 153, 178),
    line = Color3.fromRGB(73, 100, 132),
    wall = Color3.fromRGB(172, 190, 210),
    stair = Color3.fromRGB(255, 203, 92),
}

local function getRooms()
    local folder = workspace:FindFirstChild("CurrentRooms")
    if not folder then
        return {}
    end

    for _, cached in pairs(discoveredRooms) do
        cached.active = false
        cached.model = nil
    end

    local result = {}
    for _, model in ipairs(folder:GetChildren()) do
        if model:IsA("Model") then
            local boxCFrame, boxSize = model:GetBoundingBox()
            local key = model.Name
            local room = discoveredRooms[key]
            if not room then
                room = { name = key, number = tonumber(key) }
                discoveredRooms[key] = room
            end
            room.model = model
            room.active = true
            room.position = boxCFrame.Position
            room.sizeX = math.max(boxSize.X, 12)
            room.sizeZ = math.max(boxSize.Z, 12)
            room.yaw = math.deg(math.atan2(-boxCFrame.LookVector.X, -boxCFrame.LookVector.Z))
            room.number = tonumber(key)
        end
    end

    for _, room in pairs(discoveredRooms) do
        if rememberExplored or room.active then
            table.insert(result, room)
        end
    end

    table.sort(result, function(a, b)
        if a.number and b.number then
            return a.number < b.number
        end
        return a.name < b.name
    end)
    return result
end

local function containsAny(text, words)
    text = string.lower(text or "")
    for _, word in ipairs(words) do
        if string.find(text, word, 1, true) then
            return true
        end
    end
    return false
end

local function mapPoint(position, centerX, centerZ, scale, canvasSize)
    return Vector2.new(
        canvasSize.X / 2 + (position.X - centerX) * scale,
        canvasSize.Y / 2 + (position.Z - centerZ) * scale
    )
end

local function drawLine(parent, name, p1, p2, color, thickness, zIndex)
    local delta = p2 - p1
    local line = Instance.new("Frame")
    line.Name = name
    line.AnchorPoint = Vector2.new(0.5, 0.5)
    line.Position = UDim2.fromOffset((p1.X + p2.X) / 2, (p1.Y + p2.Y) / 2)
    line.Size = UDim2.fromOffset(math.max(delta.Magnitude, 1), thickness)
    line.Rotation = math.deg(math.atan2(delta.Y, delta.X))
    line.BackgroundColor3 = color
    line.BorderSizePixel = 0
    line.ZIndex = zIndex or 1
    line.Parent = parent
    return line
end

local function partLooksLikeWall(part)
    if not part:IsA("BasePart") or part.Transparency >= 0.95 or not part.CanQuery then
        return false
    end
    local name = string.lower(part.Name)
    if containsAny(name, {"floor", "ceiling", "roof", "carpet", "ground", "trim", "stair", "step", "ladder", "rail", "rung", "door", "handle", "frame", "glass"}) then
        return false
    end
    if containsAny(name, {"wall", "partition", "barrier", "bulkhead", "divider"}) then
        return true
    end
    -- Geometry fallback for unnamed wall parts.
    local size = part.Size
    local horizontalSmall = math.min(size.X, size.Z)
    local horizontalLong = math.max(size.X, size.Z)
    return size.Y >= 5 and horizontalSmall <= 4 and horizontalLong >= 4
end

local function drawRaycastWallFootprint(canvas, part, centerX, centerZ, scale, canvasSize)
    local cf, size = part.CFrame, part.Size
    local halfX, halfZ = size.X / 2, size.Z / 2
    local corners = {
        cf:PointToWorldSpace(Vector3.new(-halfX, 0, -halfZ)),
        cf:PointToWorldSpace(Vector3.new(halfX, 0, -halfZ)),
        cf:PointToWorldSpace(Vector3.new(halfX, 0, halfZ)),
        cf:PointToWorldSpace(Vector3.new(-halfX, 0, halfZ)),
    }
    local mapped = {}
    for i, corner in ipairs(corners) do
        mapped[i] = mapPoint(corner, centerX, centerZ, scale, canvasSize)
    end
    for i = 1, 4 do
        local nextIndex = i % 4 + 1
        drawLine(canvas, "RaycastWall", mapped[i], mapped[nextIndex], COLORS.wall, 2, 3)
    end
end

local function destroyGui()
    if updateConnection then
        updateConnection:Disconnect()
        updateConnection = nil
    end
    if roomFolderConnection then
        roomFolderConnection:Disconnect()
        roomFolderConnection = nil
    end
    if descendantConnection then
        descendantConnection:Disconnect()
        descendantConnection = nil
    end
    if dragConnection then
        dragConnection:Disconnect()
        dragConnection = nil
    end
    if gui then
        gui:Destroy()
        gui = nil
    end
    table.clear(roomButtons)
end

local function createGui()
    if gui and gui.Parent then
        return
    end

    local playerGui = player:FindFirstChildOfClass("PlayerGui")
    if not playerGui then
        playerGui = player:WaitForChild("PlayerGui", 10)
    end
    if not playerGui then
        return
    end

    gui = Instance.new("ScreenGui")
    gui.Name = "MSDoorsMiniMap"
    gui.ResetOnSpawn = false
    gui.IgnoreGuiInset = false
    gui.DisplayOrder = 35
    gui.Parent = playerGui

    local panel = Instance.new("Frame")
    panel.Name = "MapPanel"
    panel.AnchorPoint = Vector2.new(1, 0)
    panel.Position = UDim2.new(1, -14, 0, 14)
    panel.Size = UDim2.fromOffset(mapSize, mapSize + 30)
    panel.BackgroundColor3 = COLORS.panel
    panel.BackgroundTransparency = 0.12
    panel.BorderSizePixel = 0
    panel.ClipsDescendants = true
    panel.Parent = gui

    local stroke = Instance.new("UIStroke")
    stroke.Color = COLORS.border
    stroke.Thickness = 1.5
    stroke.Parent = panel

    local title = Instance.new("TextLabel")
    title.Name = "Header"
    title.Size = UDim2.new(1, -8, 0, 24)
    title.Position = UDim2.fromOffset(4, 2)
    title.BackgroundTransparency = 1
    title.Font = Enum.Font.GothamBold
    title.Text = "MINI MAP"
    title.TextColor3 = COLORS.text
    title.TextSize = 12
    title.TextXAlignment = Enum.TextXAlignment.Left
    title.Parent = panel

    local close = Instance.new("TextButton")
    close.Name = "Close"
    close.Size = UDim2.fromOffset(22, 20)
    close.Position = UDim2.new(1, -25, 0, 3)
    close.BackgroundColor3 = Color3.fromRGB(25, 36, 54)
    close.BorderSizePixel = 0
    close.Font = Enum.Font.GothamBold
    close.Text = "×"
    close.TextSize = 16
    close.TextColor3 = COLORS.text
    close.Parent = panel
    close.Activated:Connect(function()
        enabled = false
        destroyGui()
    end)

    local map = Instance.new("Frame")
    map.Name = "RoomCanvas"
    map.Position = UDim2.fromOffset(4, 27)
    map.Size = UDim2.new(1, -8, 1, -31)
    map.BackgroundTransparency = 1
    map.ClipsDescendants = true
    map.Parent = panel

    local footer = Instance.new("TextLabel")
    footer.Name = "Hint"
    footer.AnchorPoint = Vector2.new(0.5, 1)
    footer.Position = UDim2.new(0.5, 0, 1, -2)
    footer.Size = UDim2.new(1, -8, 0, 14)
    footer.BackgroundTransparency = 1
    footer.Font = Enum.Font.Gotham
    footer.Text = "BLUE YOU  •  YELLOW HEADING"
    footer.TextColor3 = COLORS.muted
    footer.TextSize = 8
    footer.Parent = panel

    local compass = Instance.new("TextLabel")
    compass.Name = "Compass"
    compass.Position = UDim2.fromOffset(5, 29)
    compass.Size = UDim2.fromOffset(18, 14)
    compass.BackgroundColor3 = Color3.fromRGB(10, 13, 24)
    compass.BackgroundTransparency = 0.2
    compass.BorderSizePixel = 0
    compass.Font = Enum.Font.GothamBold
    compass.Text = "N"
    compass.TextColor3 = Color3.fromRGB(255, 242, 150)
    compass.TextSize = 10
    compass.ZIndex = 5
    compass.Parent = panel

    -- Drag by the header; works with mouse and touch.
    local dragging = false
    local dragInput
    local dragStart
    local startPosition
    title.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPosition = panel.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    title.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    dragConnection = UserInputService.InputChanged:Connect(function(input)
        if dragging and (input == dragInput or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            panel.Position = UDim2.new(
                startPosition.X.Scale,
                startPosition.X.Offset + delta.X,
                startPosition.Y.Scale,
                startPosition.Y.Offset + delta.Y
            )
        end
    end)

    gui:SetAttribute("MapSize", mapSize)
end

local function drawMap()
    if not enabled or not gui or not gui.Parent then
        return
    end

    local panel = gui:FindFirstChild("MapPanel")
    local canvas = panel and panel:FindFirstChild("RoomCanvas")
    if not canvas then
        return
    end

    for _, child in ipairs(canvas:GetChildren()) do
        child:Destroy()
    end
    table.clear(roomButtons)

    local rooms = getRooms()
    local header = panel:FindFirstChild("Header")
    if header then
        header.Text = "MINI MAP  •  " .. tostring(#rooms) .. " ROOMS"
    end
    if #rooms == 0 then
        local label = Instance.new("TextLabel")
        label.Size = UDim2.fromScale(1, 1)
        label.BackgroundTransparency = 1
        label.Font = Enum.Font.Gotham
        label.Text = "Waiting for rooms..."
        label.TextColor3 = COLORS.muted
        label.TextSize = 11
        label.Parent = canvas
        return
    end

    local character = player.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local currentRoom
    if root then
        local bestDistance = math.huge
        for _, room in ipairs(rooms) do
            local distance = (room.position - root.Position).Magnitude
            if distance < bestDistance then
                bestDistance = distance
                currentRoom = room
            end
        end
    end
    currentRoom = currentRoom or rooms[#rooms]

    -- Include the room bounds in the fit so walls and side rooms stay inside the map.
    local minX, maxX, minZ, maxZ = math.huge, -math.huge, math.huge, -math.huge
    for _, room in ipairs(rooms) do
        local halfX = (room.sizeX or 24) / 2
        local halfZ = (room.sizeZ or 24) / 2
        minX = math.min(minX, room.position.X - halfX)
        maxX = math.max(maxX, room.position.X + halfX)
        minZ = math.min(minZ, room.position.Z - halfZ)
        maxZ = math.max(maxZ, room.position.Z + halfZ)
    end

    local width = math.max(maxX - minX, 1)
    local depth = math.max(maxZ - minZ, 1)
    local canvasSize = canvas.AbsoluteSize
    local padding = 16
    local scale = math.min(
        (canvasSize.X - padding * 2) / width,
        (canvasSize.Y - padding * 2) / depth
    )
    scale = math.clamp(scale, 0.035, 0.65)

    local centerX = (minX + maxX) / 2
    local centerZ = (minZ + maxZ) / 2
    local defaultBoxW = 24
    local defaultBoxH = 18

    local positions = {}
    for _, room in ipairs(rooms) do
        positions[room] = mapPoint(room.position, centerX, centerZ, scale, canvasSize)
    end

    -- Trace room outlines by joining raycast hit points in angular order.
    if showWalls then
        local rayParams = RaycastParams.new()
        rayParams.FilterType = Enum.RaycastFilterType.Include
        rayParams.IgnoreWater = true
        local rayBudget = 300
        local scanOrder = {}
        if currentRoom and currentRoom.active and currentRoom.model then
            table.insert(scanOrder, currentRoom)
        end
        for _, room in ipairs(rooms) do
            if room ~= currentRoom and room.active and room.model then
                table.insert(scanOrder, room)
            end
        end

        for _, room in ipairs(scanOrder) do
            if rayBudget < 24 then break end
            rayParams.FilterDescendantsInstances = {room.model}
            local hits = {}
            local seen = {}
            local rayCount = math.min(64, rayBudget)
            local rayLength = math.clamp(math.max(room.sizeX or 40, room.sizeZ or 40) * 1.2, 35, 110)
            for i = 0, rayCount - 1 do
                local angle = (i / rayCount) * math.pi * 2
                local direction = Vector3.new(math.cos(angle), 0, math.sin(angle))
                local origin = Vector3.new(room.position.X, room.position.Y, room.position.Z)
                local hit = workspace:Raycast(origin, direction * rayLength, rayParams)
                rayBudget -= 1
                local tries = 0
                while hit and not partLooksLikeWall(hit.Instance) and tries < 3 and rayBudget > 0 do
                    local advance = (hit.Position - origin).Magnitude + 0.15
                    origin = Vector3.new(room.position.X, room.position.Y, room.position.Z) + direction * advance
                    local remaining = rayLength - advance
                    if remaining <= 1 then break end
                    hit = workspace:Raycast(origin, direction * remaining, rayParams)
                    rayBudget -= 1
                    tries += 1
                end
                if hit and partLooksLikeWall(hit.Instance) then
                    local point = Vector3.new(hit.Position.X, 0, hit.Position.Z)
                    local key = tostring(math.floor(point.X * 2)) .. ":" .. tostring(math.floor(point.Z * 2))
                    if not seen[key] then
                        seen[key] = true
                        table.insert(hits, {angle = math.atan2(point.Z - room.position.Z, point.X - room.position.X), point = point})
                    end
                end
                if rayBudget <= 0 then break end
            end
            if #hits >= 3 then
                table.sort(hits, function(a, b) return a.angle < b.angle end)
                for i = 1, #hits do
                    local nextIndex = i % #hits + 1
                    local p1 = mapPoint(hits[i].point, centerX, centerZ, scale, canvasSize)
                    local p2 = mapPoint(hits[nextIndex].point, centerX, centerZ, scale, canvasSize)
                    drawLine(canvas, "RoomWallContour", p1, p2, COLORS.wall, 2, 3)
                end
            end
        end
    end

    -- Stairs are inferred from stair/step/ladder names and drawn as a yellow stair glyph.
    -- The vertical range of the detected parts is used to show whether the route changes floors.
    if showStairs then
        local stairCount = 0
        for _, room in ipairs(rooms) do
            if room.active and room.model and stairCount < 35 then
                local stairParts = {}
                local lowY, highY = math.huge, -math.huge
                local minSX, maxSX, minSZ, maxSZ = math.huge, -math.huge, math.huge, -math.huge
                local stairContainers = {}
                for _, descendant in ipairs(room.model:GetDescendants()) do
                    if (descendant:IsA("Model") or descendant:IsA("Folder")) and containsAny(descendant.Name, {"stair", "steps", "ladder"}) then
                        stairContainers[descendant] = true
                    end
                end
                for _, part in ipairs(room.model:GetDescendants()) do
                    if part:IsA("BasePart") then
                        local namedPart = containsAny(part.Name, {"stair", "steps", "step", "ladder", "rung"})
                        local inStairContainer = false
                        local ancestor = part.Parent
                        while ancestor and ancestor ~= room.model do
                            if stairContainers[ancestor] then
                                inStairContainer = true
                                break
                            end
                            ancestor = ancestor.Parent
                        end
                        if namedPart or inStairContainer then
                            table.insert(stairParts, part)
                            lowY = math.min(lowY, part.Position.Y - part.Size.Y / 2)
                            highY = math.max(highY, part.Position.Y + part.Size.Y / 2)
                            minSX = math.min(minSX, part.Position.X - part.Size.X / 2)
                            maxSX = math.max(maxSX, part.Position.X + part.Size.X / 2)
                            minSZ = math.min(minSZ, part.Position.Z - part.Size.Z / 2)
                            maxSZ = math.max(maxSZ, part.Position.Z + part.Size.Z / 2)
                        end
                    end
                end
                if #stairParts > 0 then
                    local mid = mapPoint(Vector3.new((minSX + maxSX) / 2, 0, (minSZ + maxSZ) / 2), centerX, centerZ, scale, canvasSize)
                    local glyph = Instance.new("TextLabel")
                    glyph.Name = "Stairs"
                    glyph.AnchorPoint = Vector2.new(0.5, 0.5)
                    glyph.Position = UDim2.fromOffset(mid.X, mid.Y)
                    glyph.Size = UDim2.fromOffset(28, 24)
                    glyph.BackgroundColor3 = Color3.fromRGB(22, 24, 30)
                    glyph.BackgroundTransparency = 0.12
                    glyph.BorderSizePixel = 0
                    glyph.Font = Enum.Font.GothamBlack
                    glyph.Text = (highY - lowY > 8) and "⇧" or "≋"
                    glyph.TextColor3 = COLORS.stair
                    glyph.TextSize = 18
                    glyph.ZIndex = 5
                    glyph.Parent = canvas
                    local outline = Instance.new("UIStroke")
                    outline.Color = COLORS.stair
                    outline.Thickness = 1
                    outline.Parent = glyph
                    stairCount += 1
                end
            end
        end
    end

    for _, room in ipairs(rooms) do
        local point = positions[room]
        local isCurrent = room == currentRoom
        local node = Instance.new("Frame")
        node.Name = "Room_" .. room.name
        node.AnchorPoint = Vector2.new(0.5, 0.5)
        node.Position = UDim2.fromOffset(point.X, point.Y)
        local roomW = room.sizeX and math.clamp(room.sizeX * scale, 12, 52) or defaultBoxW
        local roomH = room.sizeZ and math.clamp(room.sizeZ * scale, 12, 52) or defaultBoxH
        node.Size = UDim2.fromOffset(roomW, roomH)
        node.Rotation = room.yaw or 0
        node.BackgroundColor3 = isCurrent and COLORS.current or COLORS.explored
        node.BackgroundTransparency = room.active and 0 or 0.28
        node.BorderSizePixel = 0
        node.ZIndex = 2
        node.Parent = canvas

        local outline = Instance.new("UIStroke")
        outline.Color = isCurrent and Color3.fromRGB(130, 205, 255) or (room.active and COLORS.line or Color3.fromRGB(83, 91, 108))
        outline.Thickness = isCurrent and 2 or 1
        outline.Parent = node

        if showRoomNumbers then
            local text = Instance.new("TextLabel")
            text.Size = UDim2.fromScale(1, 1)
            text.BackgroundTransparency = 1
            text.Font = Enum.Font.GothamBold
            text.Text = room.number and tostring(room.number) or room.name
            text.TextColor3 = COLORS.text
            text.TextSize = 9
            text.TextTruncate = Enum.TextTruncate.AtEnd
            text.ZIndex = 3
            text.Parent = node
        end
    end

    -- Plot the player's actual position, not the room center, so the marker moves while walking.
    if root then
        local p = mapPoint(root.Position, centerX, centerZ, scale, canvasSize)
        local marker = Instance.new("TextLabel")
        marker.Name = "PlayerMarker"
        marker.AnchorPoint = Vector2.new(0.5, 0.5)
        marker.Position = UDim2.fromOffset(p.X, p.Y)
        marker.Size = UDim2.fromOffset(18, 18)
        marker.BackgroundColor3 = Color3.fromRGB(10, 13, 24)
        marker.BackgroundTransparency = 0.15
        marker.BorderSizePixel = 0
        marker.Font = Enum.Font.GothamBlack
        marker.Text = "▲"
        marker.TextColor3 = Color3.fromRGB(255, 242, 150)
        marker.TextSize = 13
        marker.Rotation = -math.deg(math.atan2(root.CFrame.LookVector.Z, root.CFrame.LookVector.X)) + 90
        marker.ZIndex = 6
        marker.Parent = canvas
    end
end

local function startMap()
    enabled = true
    createGui()
    if not gui then
        return
    end

    if updateConnection then
        updateConnection:Disconnect()
    end
    local elapsed = 0
    updateConnection = RunService.Heartbeat:Connect(function(dt)
        if not enabled then
            return
        end
        elapsed += dt
        if elapsed >= 0.35 then
            elapsed = 0
            drawMap()
        end
    end)
end

local function stopMap()
    enabled = false
    destroyGui()
end
