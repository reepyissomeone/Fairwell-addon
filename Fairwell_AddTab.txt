-- Fairwell Heaven: add a tab to the existing MSDoors window
-- This addon does NOT create a separate window.
-- Remote content lives in Fairwell_Content.txt so it can be updated independently.

local env = (getgenv and getgenv()) or _G
local CONTENT_URL = "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/Fairwell_Content.txt"

local Library = env.Library or env.Obsidian or env.UI_Library
local Window = env.Window or env.MainWindow or env.MSDoorsWindow

if not Window and Library then
    Window = Library.Window
end

if not Window or type(Window.AddTab) ~= "function" then
    warn("[Fairwell] Couldn't find the existing MSDoors window.")
    warn("[Fairwell] Make sure MSDoors has loaded before running this addon.")
    warn("[Fairwell] No new window was created.")
    return
end

local ok, tab = pcall(function()
    return Window:AddTab("Fairwell", "heart")
end)

if not ok or not tab then
    warn("[Fairwell] Could not add tab:", tab)
    return
end

local function loadRemoteContent()
    if type(game.HttpGet) ~= "function" then
        warn("[Fairwell] HttpGet is unavailable; remote content could not load.")
        return false
    end

    local fetched, source = pcall(function()
        return game:HttpGet(CONTENT_URL)
    end)

    if not fetched or type(source) ~= "string" or source == "" then
        warn("[Fairwell] Failed to fetch remote content:", source)
        return false
    end

    if type(loadstring) ~= "function" then
        warn("[Fairwell] loadstring is unavailable; remote content could not run.")
        return false
    end

    local compiled, chunk = pcall(loadstring, source)
    if not compiled or type(chunk) ~= "function" then
        warn("[Fairwell] Remote content did not compile:", chunk)
        return false
    end

    local ran, contentBuilder = pcall(chunk)
    if not ran or type(contentBuilder) ~= "function" then
        warn("[Fairwell] Remote content must return a function:", contentBuilder)
        return false
    end

    local built, buildError = pcall(contentBuilder, tab)
    if not built then
        warn("[Fairwell] Remote content failed while building the tab:", buildError)
        return false
    end

    print("[Fairwell] Remote content loaded successfully.")
    return true
end

if not loadRemoteContent() then
    -- Fallback so the tab is still visible if GitHub is temporarily unavailable.
    local fallbackOk, fallback = pcall(function()
        return tab:AddLeftGroupbox("Fairwell Heaven", "heart")
    end)
    if fallbackOk and fallback then
        pcall(function()
            fallback:AddLabel({
                Text = "Fairwell tab loaded, but remote content failed. Check the console and try running the addon again.",
                DoesWrap = true
            })
        end)
    end
end

print("[Fairwell] Tab added to the existing MSDoors window.")
