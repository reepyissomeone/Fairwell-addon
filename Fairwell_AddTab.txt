-- Fairwell Heaven: add a tab to the existing MSDoors window
-- This addon does NOT create a separate window.
-- Remote content lives in Fairwell_Content.txt so it can be updated independently.

local env = (getgenv and getgenv()) or _G
local CONTENT_URL = "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/Fairwell_Content.txt"
local ACCENT = Color3.fromRGB(27, 147, 227)
local ACCENT_DARK = Color3.fromRGB(16, 13, 105)

local Library = env.Library or env.Obsidian or env.UI_Library
local Window = env.Window or env.MainWindow or env.MSDoorsWindow
if not Window and Library then Window = Library.Window end

if not Window or type(Window.AddTab) ~= "function" then
    warn("[Fairwell] Couldn't find the existing MSDoors window.")
    warn("[Fairwell] Make sure MSDoors has loaded before running this addon.")
    return
end

local ok, tab = pcall(function()
    return Window:AddTab("Fairwell", "heart")
end)
if not ok or not tab then
    warn("[Fairwell] Could not add tab:", tab)
    return
end

-- Tint only the Fairwell tab button, leaving other MSDoors tabs untouched.
local function tintFairwellTab()
    local roots = {}
    pcall(function() table.insert(roots, game:GetService("CoreGui")) end)
    if type(gethui) == "function" then
        pcall(function() table.insert(roots, gethui()) end)
    end
    for _, root in ipairs(roots) do
        pcall(function()
            for _, obj in ipairs(root:GetDescendants()) do
                if (obj:IsA("TextButton") or obj:IsA("ImageButton")) then
                    local text = ""
                    pcall(function() text = obj.Text end)
                    if string.lower(tostring(text)) == "fairwell" then
                        obj.BackgroundColor3 = ACCENT_DARK
                        pcall(function() obj.TextColor3 = Color3.fromRGB(255, 255, 255) end)
                        pcall(function() obj.ImageColor3 = ACCENT end)
                        pcall(function() obj.BorderColor3 = ACCENT end)
                        pcall(function()
                            for _, child in ipairs(obj:GetDescendants()) do
                                if child:IsA("TextLabel") then
                                    child.TextColor3 = Color3.fromRGB(255, 255, 255)
                                elseif child:IsA("ImageLabel") then
                                    child.ImageColor3 = ACCENT
                                end
                            end
                        end)
                    end
                end
            end
        end)
    end
end

local function loadRemoteContent()
    local fetched, source = pcall(function() return game:HttpGet(CONTENT_URL) end)
    if not fetched or type(source) ~= "string" or source == "" then
        warn("[Fairwell] Failed to fetch remote content:", source)
        return false
    end
    if type(loadstring) ~= "function" then
        warn("[Fairwell] loadstring is unavailable.")
        return false
    end
    local compiled, chunk = pcall(loadstring, source)
    if not compiled or type(chunk) ~= "function" then
        warn("[Fairwell] Remote content did not compile:", chunk)
        return false
    end
    local ran, contentBuilder = pcall(chunk)
    if not ran or type(contentBuilder) ~= "function" then
        warn("[Fairwell] Remote content must return a function:", contentBuilder)
        return false
    end
    local built, buildError = pcall(contentBuilder, tab)
    if not built then
        warn("[Fairwell] Remote content failed while building the tab:", buildError)
        return false
    end
    print("[Fairwell] Remote content loaded successfully.")
    return true
end

if not loadRemoteContent() then
    local fallbackOk, fallback = pcall(function()
        return tab:AddLeftGroupbox("Fairwell Heaven", "heart")
    end)
    if fallbackOk and fallback then
        pcall(function()
            fallback:AddLabel({Text = "Remote content failed to load. Check the console and run the addon again.", DoesWrap = true})
        end)
    end
end

-- Run after the tab has been created and content has populated the UI.
tintFairwellTab()
task.defer(tintFairwellTab)
task.delay(1, tintFairwellTab)
print("[Fairwell] Tab added to MSDoors with its own blue accent.")
