-- Fairwell Heaven native MSDoors addon entrypoint.
-- MSDoors' addon loader expects a returned table with Name and Elements.
-- The button starts our custom Obsidian tab loader inside the existing MSDoors window.

return {
    Name = "Fairwell",
    Title = "Fairwell Heaven",
    Description = "Adds the Fairwell feature tab to the existing MSDoors Obsidian window.",
    Author = "Fairwell",
    Version = "1.1",
    Side = "Left",
    Elements = {
        {
            Type = "Label",
            Arguments = {
                Text = "Open Fairwell Heaven in the existing MSDoors window. This uses the Obsidian tab and groupbox layout.",
                DoesWrap = true
            }
        },
        {
            Type = "Button",
            Arguments = {
                Text = "Open Fairwell Tab",
                DoubleClick = false,
                Func = function()
                    local env = (getgenv and getgenv()) or _G
                    local httpGet = function(url)
                        return game:HttpGet(url)
                    end
                    local compile = (type(loadstring) == "function" and loadstring)
                        or env.loadstring
                    if type(compile) ~= "function" then
                        warn("[Fairwell] Cannot start: loadstring is unavailable.")
                        return
                    end

                    local url = "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/Fairwell_Loader.txt?cache=obsidian-entry-20261018-1"
                    local fetched, source = pcall(httpGet, url)
                    if not fetched or type(source) ~= "string" or source == "" then
                        warn("[Fairwell] Could not download the Obsidian tab loader:", source)
                        return
                    end

                    local compiled, chunk, compileError = pcall(compile, source)
                    if not compiled or type(chunk) ~= "function" then
                        warn("[Fairwell] Tab loader compilation failed:", compileError or chunk)
                        return
                    end

                    local ran, runtimeError = pcall(chunk)
                    if not ran then
                        warn("[Fairwell] Tab loader failed:", runtimeError)
                        return
                    end
                    print("[Fairwell] Obsidian tab loader started.")
                end
            }
        }
    }
}
