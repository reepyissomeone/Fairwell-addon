-- Fairwell Heaven: direct-running script for the existing MSDoors window
-- This script does NOT create a separate window.
-- No global theme/color changes are applied to MSDoors.
-- Remote content lives in Fairwell_Content.txt so it can be updated independently.

local CONTENT_URL = "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/Fairwell_Content.txt?cache=compiler-call-fix-20261018-4"

local function isObsidianWindow(candidate)
    if type(candidate) ~= "table" then return false end
    local ok, valid = pcall(function()
        return type(candidate.AddTab) == "function" and candidate.MainFrame ~= nil
    end)
    return ok and valid
end

local function findMSDoorsWindow()
    local env = (getgenv and getgenv()) or _G
    local Library = env.Library or env.Obsidian or env.UI_Library
        or _G.Library or _G.Obsidian or _G.UI_Library
    local Window = env.Window or env.MainWindow or env.MSDoorsWindow
        or _G.Window or _G.MainWindow or _G.MSDoorsWindow

    if not isObsidianWindow(Window) and type(Library) == "table" then
        local ok, candidate = pcall(function() return Library.Window end)
        if ok and isObsidianWindow(candidate) then Window = candidate end
    end

    -- MSDoors may keep Obsidian's Library/Window local instead of publishing globals.
    -- Executor getgc can still find the existing window; never create a second one.
    if not isObsidianWindow(Window) and type(getgc) == "function" then
        local ok, objects = pcall(getgc, true)
        if ok and type(objects) == "table" then
            local fallbackWindow, fallbackLibrary
            for _, object in ipairs(objects) do
                if type(object) == "table" then
                    local foundLibraryWindow
                    local libraryOk, candidate = pcall(function() return object.Window end)
                    if libraryOk and isObsidianWindow(candidate) then
                        foundLibraryWindow = candidate
                        fallbackLibrary = object
                    end
                    if foundLibraryWindow then
                        Window, Library = foundLibraryWindow, fallbackLibrary
                        break
                    elseif isObsidianWindow(object) and not fallbackWindow then
                        fallbackWindow = object
                    end
                end
            end
            if not isObsidianWindow(Window) and fallbackWindow then
                Window = fallbackWindow
            end
        end
    end
    if not isObsidianWindow(Window) then return nil, Library end
    return Window, Library
end

-- MSDoors may finish creating its UI shortly after an addon/script starts.
local Window, Library
for attempt = 1, 40 do
    Window, Library = findMSDoorsWindow()
    if Window and type(Window.AddTab) == "function" then
        break
    end
    if task and type(task.wait) == "function" then
        task.wait(0.25)
    else
        break
    end
end

if not Window or type(Window.AddTab) ~= "function" then
    warn("[Fairwell] Couldn't find the existing MSDoors window after waiting.")
    warn("[Fairwell] Make sure MSDoors has loaded before running this addon.")
    return
end

-- Obsidian keeps its tab registry private in some builds, so remember our own tab.
local env = (getgenv and getgenv()) or _G
if env._FairwellTabWindow == Window and env._FairwellTab then
    warn("[Fairwell] The Fairwell tab is already loaded.")
    return
end

-- Use the library's own Obsidian tab API; no separate window or global theme changes.
local ok, tab = pcall(function()
    return Window:AddTab("Fairwell", "sparkles")
end)
if not ok or not tab then
    warn("[Fairwell] Could not add tab:", tab)
    return
end
env._FairwellTabWindow = Window
env._FairwellTab = tab

-- Keep a bounded, copyable diagnostic log in the UI as well as the executor console.
local ERROR_LOG = {}
local MAX_LOG_ENTRIES = 60
local function logMessage(message)
    local line = tostring(message)
    table.insert(ERROR_LOG, line)
    while #ERROR_LOG > MAX_LOG_ENTRIES do table.remove(ERROR_LOG, 1) end
    warn(line)
end

local function loadRemoteContent()
    local fetched, source = pcall(function() return game:HttpGet(CONTENT_URL) end)
    if not fetched or type(source) ~= "string" or source == "" then
        logMessage("[Fairwell] Failed to fetch remote content: " .. tostring(source))
        return false
    end
    local compile = (type(loadstring) == "function" and loadstring)
        or ((getgenv and getgenv()) or _G).loadstring
    if type(compile) ~= "function" then
        logMessage("[Fairwell] loadstring is unavailable.")
        return false
    end
    local compiled, chunk, compileError = pcall(function()
        return compile(tostring(source))
    end)
    if not compiled or type(chunk) ~= "function" then
        logMessage("[Fairwell] Remote content did not compile: " .. tostring(compileError or chunk or "unknown compiler error"))
        return false
    end
    local ran, contentBuilder = pcall(chunk)
    if not ran or type(contentBuilder) ~= "function" then
        logMessage("[Fairwell] Remote content must return a function: " .. tostring(contentBuilder))
        return false
    end
    local built, buildError = xpcall(function()
        contentBuilder(tab, loadRemoteContent)
    end, function(err)
        return debug and debug.traceback and debug.traceback(tostring(err), 2) or tostring(err)
    end)
    if not built then
        logMessage("[Fairwell] Remote content failed while building the tab:\n" .. tostring(buildError))
        return false
    end

    logMessage("[Fairwell] Remote content loaded successfully.")
    return true
end
local contentLoaded = loadRemoteContent()
if not contentLoaded then
    local fallbackOk, fallback = pcall(function()
        return tab:AddLeftGroupbox("Fairwell Loader", "alert-circle")
    end)
    if fallbackOk and fallback then
        local statusLabel
        pcall(function()
            statusLabel = fallback:AddLabel({
                Text = "Fairwell features did not finish loading. Check the executor console, then retry below.",
                DoesWrap = true
            })
        end)
        pcall(function()
            fallback:AddButton({
                Text = "Retry Loading Fairwell",
                Func = function()
                    local retried = loadRemoteContent()
                    if retried and statusLabel and type(statusLabel.SetText) == "function" then
                        pcall(function() statusLabel:SetText("Fairwell features loaded successfully.") end)
                    end
                end
            })
        end)
    end
    warn("[Fairwell] The tab was created, but its feature content did not load.")
else
    print("[Fairwell] Fairwell tab and feature content loaded successfully.")
end

-- Copyable diagnostics groupbox, kept on the existing Fairwell tab.
local diagnosticOk, diagnosticBox = pcall(function()
    return tab:AddLeftGroupbox("Fairwell Diagnostics", "clipboard")
end)
if diagnosticOk and diagnosticBox then
    pcall(function()
        diagnosticBox:AddLabel({
            Text = "Keeps the latest Fairwell loader errors. Press Copy Log to copy them for troubleshooting.",
            DoesWrap = true
        })
    end)
    pcall(function()
        diagnosticBox:AddButton({
            Text = "Copy Error Log",
            Func = function()
                local textToCopy = #ERROR_LOG > 0 and table.concat(ERROR_LOG, "\n") or "[Fairwell] No errors recorded."
                local envNow = (getgenv and getgenv()) or _G
                local copier = (type(setclipboard) == "function" and setclipboard)
                    or (type(toclipboard) == "function" and toclipboard)
                    or envNow.setclipboard or envNow.toclipboard
                if type(copier) == "function" then
                    local ok, err = pcall(copier, textToCopy)
                    if ok then
                        print("[Fairwell] Error log copied to clipboard.")
                    else
                        warn("[Fairwell] Clipboard copy failed: " .. tostring(err))
                    end
                else
                    warn("[Fairwell] This executor does not expose a clipboard function.")
                end
            end
        })
    end)
    pcall(function()
        diagnosticBox:AddButton({
            Text = "Clear Error Log",
            Func = function()
                table.clear(ERROR_LOG)
                print("[Fairwell] Diagnostic log cleared.")
            end
        })
    end)
end
