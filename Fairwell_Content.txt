-- Fairwell Heaven DOORS features for MSDoors
-- Remote content file. Only gameplay features are shown in this tab.

return function(tab, reloadContent, hotelTab)
    if not tab then return end

    -- Developer-only Fairwell Communicator grant panel. Kept near the top so
    -- it still loads even if a later, unrelated feature has an error.
    do
        local Players = game:GetService("Players")
        local TweenService = game:GetService("TweenService")
        local localPlayer = Players.LocalPlayer
        local allowedName = "FIXROBLOX_015"
        local communicatorBox
        pcall(function()
            communicatorBox = tab:AddLeftGroupbox("Fairwell Communicator", "message-circle")
        end)

        local statusLabel
        local function setCommunicatorStatus(message)
            print("[Fairwell Communicator] " .. message)
            if statusLabel then
                pcall(function() statusLabel:SetText(message) end)
            end
        end

        local function grantCommunicator()
            if not localPlayer then
                setCommunicatorStatus("Could not find LocalPlayer.")
                return
            end
            if string.upper(localPlayer.Name) ~= allowedName then
                setCommunicatorStatus("Developer-only: username does not match.")
                return

            local backpack = localPlayer:FindFirstChildOfClass("Backpack") or localPlayer:WaitForChild("Backpack", 10)
            if not backpack then
                setCommunicatorStatus("Backpack not found. Try again.")
                return
            end

            local character = localPlayer.Character
            local oldTool = backpack:FindFirstChild("Fairwell Communicator")
            if oldTool then oldTool:Destroy() end
            oldTool = character and character:FindFirstChild("Fairwell Communicator")
            if oldTool then oldTool:Destroy() end

            local tool = Instance.new("Tool")
            tool.Name = "Fairwell Communicator"
            tool.ToolTip = "Equip to open Fairwell's communicator"
            tool.RequiresHandle = true
            tool.CanBeDropped = false
            tool.Grip = CFrame.new(0, -0.1, -0.35) * CFrame.Angles(math.rad(-8), 0, math.rad(-5))

            local handle = Instance.new("Part")
            handle.Name = "Handle"
            handle.Size = Vector3.new(0.72, 1.12, 0.16)
            handle.Color = Color3.fromRGB(16, 13, 55)
            handle.Material = Enum.Material.SmoothPlastic
            handle.CanCollide = false
            handle.CanTouch = false
            handle.Massless = true
            handle.Parent = tool

                part.CanTouch = false

                part.Parent = tool

                weld.Part1 = part
                weld.Parent = part
                return part

            local screen = detail("CyanScreen", Vector3.new(0.46, 0.62, 0.018), CFrame.new(0, 0.12, -0.122), Color3.fromRGB(35, 185, 255), Enum.Material.Neon)

            detail("BottomButton", Vector3.new(0.18, 0.065, 0.03), CFrame.new(0, -0.43, -0.105), Color3.fromRGB(27, 147, 227), Enum.Material.Neon)

            detail("RightButton", Vector3.new(0.055, 0.11, 0.03), CFrame.new(0.26, -0.39, -0.105), Color3.fromRGB(70, 110, 230), Enum.Material.Neon)

            local gui

                    gui.Enabled = not gui.Enabled
                    return

                if not playerGui then return end

                panel.BackgroundColor3 = Color3.fromRGB(6, 4, 43)

                panel.BorderSizePixel = 2
                panel.Parent = gui

                local title = Instance.new("TextLabel")
                title.Size = UDim2.new(1, -42, 0, 34)
                title.Position = UDim2.fromOffset(8, 4)
                title.BackgroundTransparency = 1
                title.Text = "FAIRWELL COMMUNICATOR"

                title.Font = Enum.Font.GothamBold
                title.TextSize = 15
                title.TextXAlignment = Enum.TextXAlignment.Left

                transcript.BackgroundColor3 = Color3.fromRGB(10, 9, 28)

                input.Position = UDim2.new(0, 10, 1, -42)

                input.BorderColor3 = Color3.fromRGB(27, 147, 227)

                        reply = "Commands: /help, /status, /ping. This chat does not execute code."

                        reply = "Online as " .. localPlayer.Name .. "."

                    end
                    transcript.Text = transcript.Text .. "\nYOU: " .. message .. "\nFAIRWELL: " .. reply

                send.Activated:Connect(submit)
                input.FocusLost:Connect(function(enterPressed) if enterPressed then submit() end end)

            local originalGrip = tool.Grip
            tool.Equipped:Connect(function()
                local tilted = originalGrip * CFrame.Angles(math.rad(-8), math.rad(4), math.rad(-5))

                TweenService:Create(tool, TweenInfo.new(0.22, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Grip = originalGrip}):Play()
            end)
            tool.Activated:Connect(function()
                task.spawn(function()
                    local bright = TweenService:Create(screen, TweenInfo.new(0.12), {Color = Color3.fromRGB(170, 245, 255)})
                    bright:Play()
                    bright.Completed:Wait()

                    TweenService:Create(statusLight, TweenInfo.new(0.18, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, 1, true), {Color = Color3.fromRGB(27, 147, 227)}):Play()

                openCommunicator()
            end)

            tool.Parent = backpack
            setCommunicatorStatus("Granted! Check your Backpack / inventory.")
        end

        if communicatorBox then
            pcall(function()
                statusLabel = communicatorBox:AddLabel({Text = "Only available to FIXROBLOX_015", DoesWrap = true})
            end)
            pcall(function()
                communicatorBox:AddButton({
                    Text = "Give Fairwell Communicator",
                    Func = grantCommunicator
                })
            end)
        end
        setCommunicatorStatus("Grant button ready.")
    end

    local stairwellBox
    pcall(function()
        stairwellBox = tab:AddLeftGroupbox("Stairwell", "eye")
    end)

    local espOk, espBox = pcall(function()
        return stairwellBox
    end)

    if espOk and espBox then
        local espEnabled = false
        local espFolder = Instance.new("Folder")
        espFolder.Name = "FairwellStemESP"
        espFolder.Parent = game:GetService("CoreGui")

        local function clearESP()
            for _, child in ipairs(espFolder:GetChildren()) do
                child:Destroy()
            end
        end

        local function stemRelated(instance)
            local node = instance
            for _ = 1, 5 do
                if not node then break end
                if string.find(string.lower(node.Name), "stem", 1, true) then
                    return true
                end
                node = node.Parent
            end
            return false
        end

        local function scan()
            clearESP()
            local seen = {}
            for _, instance in ipairs(workspace:GetDescendants()) do
                if stemRelated(instance) then
                    local target = instance:IsA("Model") and instance or instance:FindFirstAncestorOfClass("Model")
                    target = target or (instance:IsA("BasePart") and instance or nil)
                    if target and not seen[target] then
                        seen[target] = true
                        local holder = Instance.new("Folder")
                        holder.Name = "StemESPItem"
                        holder.Parent = espFolder

                        local highlight = Instance.new("Highlight")
                        highlight.Adornee = target
                        highlight.FillColor = Color3.fromRGB(255, 70, 70)
                        highlight.OutlineColor = Color3.fromRGB(255, 230, 70)
                        highlight.FillTransparency = 0.55
                        highlight.OutlineTransparency = 0
                        highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        highlight.Parent = holder

                        local part = target:IsA("BasePart") and target or target:FindFirstChildWhichIsA("BasePart", true)
                        if part then
                            local labelGui = Instance.new("BillboardGui")
                            labelGui.Name = "StemLabel"
                            labelGui.Adornee = part
                            labelGui.AlwaysOnTop = true
                            labelGui.Size = UDim2.fromOffset(110, 30)
                            labelGui.StudsOffset = Vector3.new(0, 2.5, 0)
                            labelGui.Parent = holder

                            local label = Instance.new("TextLabel")
                            label.BackgroundTransparency = 1
                            label.Size = UDim2.fromScale(1, 1)
                            label.Text = "STEM"
                            label.TextColor3 = Color3.fromRGB(255, 230, 70)
                            label.TextStrokeTransparency = 0
                            label.TextScaled = true
                            label.Font = Enum.Font.GothamBold
                            label.Parent = labelGui
                        end
                    end
                end
            end
        end

        pcall(function()
            espBox:AddToggle("FairwellStemESPToggle", {
                Text = "STEM ESP",
                Default = false,
                Callback = function(enabled)
                    espEnabled = enabled == true
                    clearESP()
                    if espEnabled then scan() end
                end
            })
        end)

        task.spawn(function()
            while espFolder.Parent do
                if espEnabled then scan() end
                task.wait(2)
            end
        end)
    end

    local stemOk, stemBox = pcall(function()
        return stairwellBox
    end)

    if stemOk and stemBox then
        local Players = game:GetService("Players")
        local ReplicatedStorage = game:GetService("ReplicatedStorage")
        local player = Players.LocalPlayer
        local env = (getgenv and getgenv()) or _G
        env._FairwellAutoKillStem = false
        local statusLabel

        local function setStatus(message)
            print("[Fairwell Auto kill Stem] " .. message)
            if statusLabel then pcall(function() statusLabel:SetText("Status: " .. message) end) end
        end
        pcall(function() statusLabel = stemBox:AddLabel({Text = "Status: idle", DoesWrap = true}) end)

        local function findHeldStem()
            local character = player.Character
            if not character then return nil end
            local tool = character:FindFirstChild("Stem")
            if not tool or not tool:IsA("Tool") then return nil end
            local handle = tool:FindFirstChild("Handle")
            if not handle or not handle:IsA("BasePart") then return tool, nil, nil end
            for _, item in ipairs(handle:GetDescendants()) do
                if item.Name:match("^Stem_") then return tool, handle, item.Name end
            end
            return tool, handle, nil
        end

        local function findVariant(tool, handle, identifier)
            -- Check the held item, its identifier, attributes, and exposed string values.
            local function inspect(value)
                local lower = string.lower(tostring(value or ""))
                for _, variant in ipairs({"Mouth", "Ear", "Eye", "Blank"}) do
                    if string.find(lower, string.lower(variant), 1, true) then return variant end
                end
            end
            local found = inspect(identifier)
            if found then return found end
            local roots = {tool}
            if handle then table.insert(roots, handle) end
            for _, root in ipairs(roots) do
                for key, value in pairs(root:GetAttributes()) do
                    found = inspect(key) or inspect(value)
                    if found then return found end
                end
                for _, item in ipairs(root:GetDescendants()) do
                    found = inspect(item.Name)
                    if not found and item:IsA("StringValue") then found = inspect(item.Value) end
                    if not found then
                        for key, value in pairs(item:GetAttributes()) do
                            found = inspect(key) or inspect(value)
                            if found then break end
                        end
                    end
                    if found then return found end
                end
            end
            return "Blank"
        end

        local function findNearbyStemPrompt()
            local character = player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not root then return nil, "character root not found" end
            local bestPrompt, bestDistance
            for _, prompt in ipairs(workspace:GetDescendants()) do
                if prompt:IsA("ProximityPrompt") and prompt.Enabled then
                    local parent = prompt.Parent
                    local part = parent and (parent:IsA("BasePart") and parent or parent:FindFirstChildWhichIsA("BasePart", true))
                    if part then
                        local distance = (root.Position - part.Position).Magnitude
                        if distance <= math.max(prompt.MaxActivationDistance + 5, 12) then
                            local node, isStem = prompt, false
                            for _ = 1, 8 do
                                if not node then break end
                                if string.find(string.lower(node.Name), "stem", 1, true) then isStem = true break end
                                node = node.Parent
                            end
                            if isStem and (not bestDistance or distance < bestDistance) then
                                bestPrompt, bestDistance = prompt, distance
                            end
                        end
                    end
                end
            end
            return bestPrompt, bestPrompt and "nearby Stem pickup found" or "searching for a nearby Stem"
        end

        local function activatePrompt(prompt)
            if not prompt or not prompt.Parent then return false, "prompt disappeared" end
            local ok, err = pcall(function()
                prompt:InputHoldBegin()
                task.wait(math.max(prompt.HoldDuration, 0.02) + 0.02)
                prompt:InputHoldEnd()
            end)
            if not ok then return false, tostring(err) end
            return true, "pickup attempted"
        end

        local function fireHeldStem(tool, handle, identifier)
            if not handle then return false, "held Stem Handle not found" end
            if not identifier then return false, "Stem_ identifier not found under Handle" end
            local remotes = ReplicatedStorage:FindFirstChild("RemotesFolder")
            local remote = remotes and remotes:FindFirstChild("FireProjectile")
            if not remote or not remote:IsA("RemoteEvent") then return false, "FireProjectile RemoteEvent not found" end
            local camera = workspace.CurrentCamera
            if not camera then return false, "camera unavailable" end
            local direction = camera.CFrame.LookVector
            local velocity = direction * 100
            local exclusions = {}
            local props = workspace:FindFirstChild("Props")
            local character = player.Character
            if props then table.insert(exclusions, props) end
            if character then table.insert(exclusions, character) end
            local variant = findVariant(tool, handle, identifier)
            local ok, err = pcall(function()
                remote:FireServer(identifier, handle.Position, direction, velocity, "Stem", variant, exclusions)
            end)
            if not ok then return false, tostring(err) end
            return true, "threw " .. variant .. " Stem"
        end

        pcall(function()
            stemBox:AddToggle("FairwellAutoKillStemToggle", {
                Text = "Auto kill Stem",
                Default = false,
                Callback = function(enabled)
                    env._FairwellAutoKillStem = enabled == true
                    if not enabled then setStatus("off") return end
                    task.spawn(function()
                        setStatus("enabled; auto-picking up and throwing Stems")
                        local lastIdentifier
                        local lastInfo
                        while env._FairwellAutoKillStem do
                            local character = player.Character
                            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                            if not humanoid or humanoid.Health <= 0 then
                                setStatus("paused: character unavailable/dead")
                                task.wait(0.25)
                            else
                                local tool, handle, identifier = findHeldStem()
                                if tool and handle and identifier and identifier ~= lastIdentifier then
                                    local ok, message = fireHeldStem(tool, handle, identifier)
                                    setStatus(ok and message or ("throw failed: " .. message))
                                    if ok then lastIdentifier = identifier end
                                    task.wait(ok and 0.08 or 0.15)
                                else
                                    -- If no usable held Stem is available, seek and pick one up.
                                    if not tool or not handle or not identifier then lastIdentifier = nil end
                                    local prompt, info = findNearbyStemPrompt()
                                    if info ~= lastInfo then setStatus(info); lastInfo = info end
                                    if prompt then
                                        local ok, message = activatePrompt(prompt)
                                        setStatus(ok and message or ("pickup failed: " .. message))
                                        task.wait(0.04)
                                    else
                                        task.wait(0.08)
                                    end
                                end
                            end
                        end
                        setStatus("off")
                    end)
                end
            })
        end)
    end

    -- Anti-Creak Survival Assist:
    -- Creak behavior documented by the community is gaze/proximity-sensitive.
    -- This is local detection and coaching only; it does not modify shared/server AI.
    local antiCreakEnv = (getgenv and getgenv()) or _G
    if antiCreakEnv._FairwellAntiCreakState then
        antiCreakEnv._FairwellAntiCreakState.running = false
        local oldHighlight = antiCreakEnv._FairwellAntiCreakState.highlight
        if oldHighlight then pcall(function() oldHighlight:Destroy() end) end
    end

    local antiCreakState = {
        running = true,
        enabled = false,
        highlight = nil
    }
    antiCreakEnv._FairwellAntiCreakState = antiCreakState

    local antiCreakBox
    pcall(function()
        antiCreakBox = stairwellBox
    end)

    if antiCreakBox then
        local Players = game:GetService("Players")
        local RunService = game:GetService("RunService")
        local localPlayer = Players.LocalPlayer
        local statusLabel
        local lastStatusText = ""
        local lastNotifiedText = ""
        local lastNotifyTime = 0
        local trackedModel = nil
        local trackedPart = nil
        local previousDistance = nil
        local previousSampleAt = 0
        local gazeStartedAt = nil
        local lastScanAt = 0
        local rayParams = RaycastParams.new()
        rayParams.FilterType = Enum.RaycastFilterType.Exclude

        pcall(function()
            statusLabel = antiCreakBox:AddLabel({
                Text = "Status: off",
                DoesWrap = true
            })
        end)

        pcall(function()
            antiCreakBox:AddLabel({
                Text = "Local survival assist: highlights Creak, warns about close range, prolonged eye contact, and a fast approach. It does not freeze or alter server AI.",
                DoesWrap = true
            })
        end)

        local function setAntiCreakStatus(message)
            if message == lastStatusText then return end
            lastStatusText = message
            if statusLabel then
                pcall(function() statusLabel:SetText("Status: " .. message) end)
            end
        end

        local function findCreakPart(model)
            if not model or not model:IsA("Model") then return nil end
            return model.PrimaryPart
                or model:FindFirstChild("HumanoidRootPart", true)
                or model:FindFirstChild("Root", true)
                or model:FindFirstChildWhichIsA("BasePart", true)
        end

        local function findNearestCreak(root)
            local bestModel, bestPart, bestDistance
            for _, obj in ipairs(workspace:GetDescendants()) do
                if obj:IsA("Model") and string.find(string.lower(obj.Name), "creak", 1, true) then
                    local part = findCreakPart(obj)
                    if part and part:IsDescendantOf(workspace) then
                        local distance = (part.Position - root.Position).Magnitude
                        if distance <= 300 and (not bestDistance or distance < bestDistance) then
                            bestModel, bestPart, bestDistance = obj, part, distance
                        end
                    end
                end
            end
            return bestModel, bestPart, bestDistance
        end

        local highlightedModel = nil

        local function clearCreakHighlight()
            if antiCreakState.highlight then
                pcall(function() antiCreakState.highlight:Destroy() end)
                antiCreakState.highlight = nil
            end
            highlightedModel = nil
        end

        local function updateCreakHighlight(model, danger)
            if highlightedModel ~= model then
                clearCreakHighlight()
                highlightedModel = model
                if model then
                    local highlight = Instance.new("Highlight")
                    highlight.Name = "FairwellAntiCreakHighlight"
                    highlight.Adornee = model
                    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                    highlight.FillTransparency = 0.78
                    highlight.OutlineTransparency = 0
                    highlight.Parent = game:GetService("CoreGui")
                    antiCreakState.highlight = highlight
                end
            end
            local highlight = antiCreakState.highlight
            if highlight then
                pcall(function()
                    highlight.FillColor = danger and Color3.fromRGB(255, 55, 55) or Color3.fromRGB(80, 210, 255)
                    highlight.OutlineColor = danger and Color3.fromRGB(255, 55, 55) or Color3.fromRGB(230, 245, 255)
                    highlight.Enabled = model ~= nil
                end)
            end
        end

        local function notifyCreak(message, urgent)
            local now = os.clock()
            if message == lastNotifiedText and now - lastNotifyTime < 6 then return end
            if now - lastNotifyTime < (urgent and 2.5 or 5) then return end
            lastNotifiedText = message
            lastNotifyTime = now
            pcall(function()
                if tab and tab.Notify then
                    tab:Notify({
                        Title = urgent and "ANTI-CREAK: DANGER" or "ANTI-CREAK",
                        Description = message,
                        Time = urgent and 5 or 4
                    })
                end
            end)
        end

        pcall(function()
            antiCreakBox:AddToggle("FairwellAntiCreakSurvival", {
                Text = "Anti-Creak Survival Assist",
                Default = false,
                Callback = function(enabled)
                    antiCreakState.enabled = enabled == true
                    gazeStartedAt = nil
                    previousDistance = nil
                    previousSampleAt = 0
                    lastNotifiedText = ""
                    if not antiCreakState.enabled then
                        trackedModel = nil
                        trackedPart = nil
                        clearCreakHighlight()
                        setAntiCreakStatus("off")
                    else
                        setAntiCreakStatus("scanning for Creak")
                    end
                end
            })
        end)

        task.spawn(function()
            while antiCreakState.running do
                task.wait(0.2)
                if antiCreakState.enabled then
                    local character = localPlayer.Character
                    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                    local root = character and character:FindFirstChild("HumanoidRootPart")
                    local camera = workspace.CurrentCamera

                    if not character or not root or not humanoid or humanoid.Health <= 0 or not camera then
                        trackedModel = nil
                        trackedPart = nil
                        gazeStartedAt = nil
                        previousDistance = nil
                        clearCreakHighlight()
                        setAntiCreakStatus("waiting for a living character")
                    else
                        rayParams.FilterDescendantsInstances = {character}
                        local now = os.clock()
                        if now - lastScanAt >= 0.8 or (trackedModel and not trackedModel.Parent) then
                            trackedModel, trackedPart = findNearestCreak(root)
                            lastScanAt = now
                        end

                        if not trackedModel or not trackedPart or not trackedPart.Parent then
                            trackedModel = nil
                            trackedPart = nil
                            gazeStartedAt = nil
                            previousDistance = nil
                            clearCreakHighlight()
                            setAntiCreakStatus("enabled; Creak not detected within 300 studs")
                        else
                            local distance = (trackedPart.Position - root.Position).Magnitude
                            local cameraOffset = trackedPart.Position - camera.CFrame.Position
                            local lookingAtCreak = false
                            if cameraOffset.Magnitude > 0 then
                                local dot = camera.CFrame.LookVector:Dot(cameraOffset.Unit)
                                if dot >= 0.94 then
                                    local result = workspace:Raycast(camera.CFrame.Position, cameraOffset, rayParams)
                                    lookingAtCreak = result ~= nil and result.Instance ~= nil
                                        and (result.Instance == trackedModel or result.Instance:IsDescendantOf(trackedModel))
                                end
                            end

                            if lookingAtCreak then
                                gazeStartedAt = gazeStartedAt or now
                            else
                                gazeStartedAt = nil
                            end

                            local dt = previousSampleAt > 0 and (now - previousSampleAt) or 0
                            local closingSpeed = 0
                            if previousDistance and dt > 0 then
                                closingSpeed = (previousDistance - distance) / dt
                            end
                            previousDistance = distance
                            previousSampleAt = now

                            local danger = false
                            local status = string.format("Creak detected - %.0f studs away", distance)
                            if distance <= 12 then
                                danger = true
                                status = string.format("DANGER: Creak is very close (%.0f studs); create distance", distance)
                                notifyCreak("Creak is dangerously close. Create distance and do not chase it.", true)
                            elseif closingSpeed >= 11 and distance <= 45 then
                                danger = true
                                status = string.format("DANGER: Creak closing quickly (%.0f studs; %.0f studs/s)", distance, closingSpeed)
                                notifyCreak("Creak is closing quickly. Be ready to turn and face it if it rushes or sneaks up behind you.", true)
                            elseif gazeStartedAt and now - gazeStartedAt >= 2.5 then
                                danger = true
                                status = string.format("prolonged eye contact (%.1fs) - break the stare", now - gazeStartedAt)
                                notifyCreak("You've been staring at Creak. Break eye contact briefly; don't chase it or stay too close.", false)
                            elseif distance <= 24 then
                                status = string.format("Creak nearby (%.0f studs) - avoid crowding it", distance)
                            elseif lookingAtCreak then
                                status = string.format("Creak visible (%.0f studs) - keep glances brief", distance)
                            end

                            updateCreakHighlight(trackedModel, danger)
                            setAntiCreakStatus(status)
                        end
                    end
                end
            end
            clearCreakHighlight()
        end)
    end

        local function stopHover()

        local function startHover()
            stopHover()
            local character = player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            if not root or not humanoid then
                status("waiting for character")
                return
            end

            local rayParams = RaycastParams.new()
            rayParams.FilterType = Enum.RaycastFilterType.Exclude
            rayParams.FilterDescendantsInstances = {character}
            local hoverHeight = 3.0
            local baseY = root.Position.Y
            local horizontalVelocity = Vector3.zero
            local lastGroundY

            hoverConnection = RunService.Heartbeat:Connect(function(dt)
                if not env._FairwellAntiGroundskeeper then stopHover(); return end
                if not character.Parent or not root.Parent or not humanoid.Parent or humanoid.Health <= 0 then
                    status("waiting for character")
                    return
                end

                -- CFrame movement bypasses the need to jump to gain clearance.
                -- Raycast downward to maintain a consistent gap above the surface.
                local result = workspace:Raycast(root.Position, Vector3.new(0, -22, 0), rayParams)
                if not result then
                    status("no ground detected")
                    return
                end

                local material = result.Material and result.Material.Name or ""
                local node = result.Instance
                local grassName = false
                for _ = 1, 5 do
                    if not node then break end
                    if string.find(string.lower(node.Name), "grass", 1, true) then
                        grassName = true
                        break
                    end
                    node = node.Parent
                end

                -- Also check nearby grass parts for maps with separate grass visuals.
                local isGrass = material == "Grass" or material == "LeafyGrass" or grassName
                if not isGrass then
                    for _, instance in ipairs(workspace:GetDescendants()) do
                        if instance:IsA("BasePart") and string.find(string.lower(instance.Name), "grass", 1, true)
                            and (instance.Position - root.Position).Magnitude <= 9
                            and math.abs(instance.Position.Y - result.Position.Y) <= 5 then
                            isGrass = true
                            break
                        end
                    end
                end

                if isGrass then
                    local targetY = result.Position.Y + humanoid.HipHeight + (root.Size.Y * 0.5) + hoverHeight
                    local current = root.Position
                    local alpha = math.clamp(dt * 12, 0, 1)
                    local newY = current.Y + (targetY - current.Y) * alpha
                    -- Preserve the character's current facing and horizontal position.
                    character:PivotTo(CFrame.new(current.X, newY, current.Z) * (root.CFrame - root.Position))
                    -- Cancel vertical momentum so the character doesn't fall back onto the grass.
                    local velocity = root.AssemblyLinearVelocity
                    root.AssemblyLinearVelocity = Vector3.new(velocity.X, 0, velocity.Z)
                    status("CFrame hovering above grass")
                    lastGroundY = result.Position.Y
                else
                    status("ground detected; grass not identified")
                end
            end)
        end

        pcall(function()
            hoverBox:AddToggle("FairwellAntiGroundskeeperToggle", {
                Text = "Anti Groundskeeper",
                Default = false,
                Callback = function(enabled)
                    env._FairwellAntiGroundskeeper = enabled == true
                    if enabled then startHover() else stopHover() end

        local function threatKind(name)
            local lower = string.lower(name or "")
            -- Rush-style threats that can be survived by entering a hiding spot.
            -- Ambush, Blitz, and A-120 may rebound, so their monitor waits for them to disappear.
            if string.find(lower, "ambush", 1, true) then return "Ambush" end
            if string.find(lower, "blitz", 1, true) then return "Blitz" end
            -- Archives threats: Scribbles is handled conservatively like a lingering/rebound threat.
            if string.find(lower, "scribble", 1, true) then return "Scribble" end
            if string.find(lower, "bash", 1, true) then return "Bash" end
            if string.find(lower, "a%-?120", 1) or string.find(lower, "a120", 1, true) then return "A-120" end
            if string.find(lower, "a%-?60", 1) or string.find(lower, "a60", 1, true) then return "A-60" end
            if string.find(lower, "rush", 1, true) then return "Rush" end
            if string.find(lower, "surge", 1, true) then return "Surge" end
            return nil
        end

        local function findThreat(root)
            local best, bestDistance, bestKind
            local seen = {}
            for _, item in ipairs(workspace:GetDescendants()) do
                if item:IsA("Model") or item:IsA("BasePart") then
                    local kind = threatKind(item.Name)
                    -- Avoid counting every named part inside the same entity as a separate threat.
                    local model = item:IsA("Model") and item or item:FindFirstAncestorOfClass("Model")
                    if kind and (not model or not seen[model]) then
                        local target = model or item
                        local targetKind = threatKind(target.Name) or kind
                        local part = target:IsA("BasePart") and target or target.PrimaryPart
                            or target:FindFirstChildWhichIsA("BasePart", true)
                        if part then
                            local distance = (root.Position - part.Position).Magnitude
                            if model then seen[model] = true end
                            if distance <= 220 and (not bestDistance or distance < bestDistance) then
                                best, bestDistance, bestKind = target, distance, targetKind
                            end
                        end
                    end
                end
            end
            return best, bestDistance, bestKind
        end

        local function promptPosition(prompt)
            local node = prompt.Parent
            while node and node ~= workspace do
                if node:IsA("BasePart") then return node.Position, node end
                if node:IsA("Model") then
                    local part = node.PrimaryPart or node:FindFirstChildWhichIsA("BasePart", true)
                    if part then return part.Position, part end
                end
                node = node.Parent
            end
            return nil, nil
        end

        local function isHidingPrompt(prompt)
            local words = string.lower(tostring(prompt.Name) .. " "
                .. tostring(prompt.ActionText) .. " " .. tostring(prompt.ObjectText))
            if string.find(words, "hide", 1, true)
                or string.find(words, "closet", 1, true)
                or string.find(words, "wardrobe", 1, true)
                or string.find(words, "locker", 1, true) then
                return true
            end
            local node = prompt.Parent
            for _ = 1, 8 do
                if not node then break end
                local name = string.lower(node.Name)
                if string.find(name, "closet", 1, true)
                    or string.find(name, "wardrobe", 1, true)
                    or string.find(name, "locker", 1, true)
                    or string.find(name, "hiding", 1, true) then
                    return true
                end
                node = node.Parent
            end
            return false
        end

        local function findClosestCloset(root)
            local bestPrompt, bestPosition, bestDistance
            for _, prompt in ipairs(workspace:GetDescendants()) do
                if prompt:IsA("ProximityPrompt") and prompt.Enabled and isHidingPrompt(prompt) then
                    local position = promptPosition(prompt)
                    if position then
                        local distance = (root.Position - position).Magnitude
                        if distance <= 125 and (not bestDistance or distance < bestDistance) then
                            bestPrompt, bestPosition, bestDistance = prompt, position, distance
                        end
                    end
                end
            end
            return bestPrompt, bestPosition, bestDistance
        end

        local function triggerGetOutOfHiding()

            if not getOutEvent then return false end

        local busy = false
        local lastAction = 0
        local function tryHide()
            if busy or not env._FairwellAutoCloset or os.clock() - lastAction < 1.5 then return end
            busy = true
            lastAction = os.clock()

            local character, root, humanoid = getCharacter()
            if not character or not root or not humanoid or humanoid.Health <= 0 then
                closetStatus("waiting for character")
                busy = false
                return

            local threat, threatDistance, kind = findThreat(root)
            if not threat then
                closetStatus("searching for rush-style threats")
                busy = false
                return

            local prompt, position, distance = findClosestCloset(root)
            if not prompt then
                closetStatus(kind .. " detected; no hiding ProximityPrompt found")
                busy = false
                return
            end

            closetStatus(kind .. " detected; approaching hiding prompt")
            if distance > math.max(prompt.MaxActivationDistance - 1, 4) then
                humanoid:MoveTo(position)
                local deadline = os.clock() + math.min(6, distance / math.max(humanoid.WalkSpeed, 8) + 1.5)
                repeat
                    task.wait(0.12)
                    character, root, humanoid = getCharacter()
                    if not env._FairwellAutoCloset or not root or not humanoid or humanoid.Health <= 0 then break end
                    if not prompt.Parent or not prompt.Enabled then break end
                    position = promptPosition(prompt)
                    if not position then break end
                    distance = (root.Position - position).Magnitude
                until distance <= math.max(prompt.MaxActivationDistance - 1, 4) or os.clock() >= deadline
            end

            if env._FairwellAutoCloset and prompt.Parent and prompt.Enabled then
                local ok = pcall(function()
                    prompt:InputHoldBegin()
                    task.wait(math.max(prompt.HoldDuration, 0.05) + 0.08)
                    prompt:InputHoldEnd()
                end)
                if ok then
                    closetStatus("used hiding ProximityPrompt")
                    task.spawn(function()
                        local started = os.clock()
                        local lastSeen = os.clock()
                        local exited = false
                        local finished = false
                        local maxDuration = (kind == "Ambush" or kind == "Blitz" or kind == "A-120" or kind == "Scribble") and 180 or 60
                        local previousDistance
                        local increasingSamples = 0
                        local closestDistance = math.huge
                        local isReboundThreat = kind == "Ambush" or kind == "Blitz" or kind == "A-120" or kind == "Scribble"

                        while env._FairwellAutoCloset and (os.clock() - started) < maxDuration do
                            task.wait(0.2)

                            local currentCharacter = player.Character
                            local currentRoot = currentCharacter and currentCharacter:FindFirstChild("HumanoidRootPart")
                            local currentHumanoid = currentCharacter and currentCharacter:FindFirstChildOfClass("Humanoid")
                            if not currentRoot or not currentHumanoid or currentHumanoid.Health <= 0 then
                                break
                            end

                            local liveThreat, liveDistance, liveKind = findThreat(currentRoot)
                            if liveThreat and liveKind == kind then
                                lastSeen = os.clock()
                                if liveDistance then
                                    closestDistance = math.min(closestDistance, liveDistance)
                                    if previousDistance and liveDistance > previousDistance + 2 then
                                        increasingSamples = increasingSamples + 1
                                    else
                                        increasingSamples = 0
                                    end
                                    previousDistance = liveDistance
                                end

                                if isReboundThreat then
                                    -- Never exit just because the entity is far away: it may rebound.
                                    closetStatus(kind .. " active; staying hidden until it disappears")
                                elseif not exited and liveDistance and closestDistance <= 28
                                    and liveDistance >= 38 and increasingSamples >= 3 then
                                    -- Confirm it passed: it got close, then moved away across multiple samples.
                                    if triggerGetOutOfHiding() then
                                        exited = true
                                        closetStatus(kind .. " passed; leaving hiding spot")
                                    else
                                        closetStatus(kind .. " passed; exit signal unavailable")
                                    end
                                elseif exited then
                                    closetStatus(kind .. " passed; encounter monitored")
                                else
                                    closetStatus(kind .. " approaching/passing; staying hidden")
                                end
                            else
                                local missingFor = os.clock() - lastSeen
                                local grace = (kind == "Ambush" or kind == "Blitz" or kind == "A-120" or kind == "Scribble") and 5 or 1.2
                                if missingFor >= grace then
                                    if not exited then
                                        triggerGetOutOfHiding()
                                    end
                                    closetStatus(kind .. " despawned; encounter monitor stopped")
                                    finished = true
                                    break
                                else
                                    closetStatus(kind .. " temporarily absent; monitoring")
                                end
                            end
                        end

                        if not finished and env._FairwellAutoCloset then
                            closetStatus(kind .. " monitoring ended; toggle Auto Closet off/on to retry")
                        end
                    end)
                else
                    closetStatus("prompt interaction failed")
                end
            end
            busy = false
        end

        pcall(function()
            closetBox:AddToggle("FairwellAutoClosetToggle", {
                Text = "Auto Closet",
                Default = false,

        -- Detect current floor/subfloor using active workspace names and room structure.
        -- Order matters: more specific subfloors must win over generic room folders.
        local function currentFloorInfo()
            local allNames = {}
            local function addName(obj)
                if obj then allNames[string.lower(obj.Name)] = true end
            end
            for _, obj in ipairs(workspace:GetChildren()) do addName(obj) end

            -- Inspect only names, not arbitrary text or item descriptions.
            -- Include nested room containers because some MSDoors versions place
            -- the active floor marker below Workspace rather than at its root.
            local activeRooms = workspace:FindFirstChild("CurrentRooms")
                or workspace:FindFirstChild("Rooms")
            local activeRoomNumber, activeRoom
            if activeRooms then
                for _, room in ipairs(activeRooms:GetChildren()) do
                    local number = tonumber(room.Name:match("%d+"))
                    if number and (not activeRoomNumber or number > activeRoomNumber) then
                        activeRoomNumber, activeRoom = number, room
                    end
                end
            end

            local roomNames = {}
            if activeRoom then
                table.insert(roomNames, string.lower(activeRoom.Name))
                for _, obj in ipairs(activeRoom:GetDescendants()) do
                    table.insert(roomNames, string.lower(obj.Name))
                end
            end

            local function anyNameContains(needles, includeRoomNames)
                for name in pairs(allNames) do
                    for _, needle in ipairs(needles) do
                        if string.find(name, needle, 1, true) then return true end
                    end
                end
                if includeRoomNames then
                    for _, name in ipairs(roomNames) do
                        for _, needle in ipairs(needles) do
                            if string.find(name, needle, 1, true) then return true end
                        end
                    end
                end
                return false
            end

            local floor, subfloor = "Unknown", "Unknown"
            local roomLabel = activeRoomNumber and string.format("Room %03d", activeRoomNumber) or nil

            if anyNameContains({"stairwell", "stairwellrooms"}, false) then
                floor, subfloor = "The Stairwell", "The Stairwell"
            elseif anyNameContains({"archives", "sectora", "sectorz", "a-000", "a000"}, true) then
                floor, subfloor = "The Archives", "The Archives"
            elseif anyNameContains({"backdoor", "backdoors"}, false) then
                floor, subfloor = "The Backdoor", "The Backdoor"
            elseif anyNameContains({"outdoors", "groundskeeper", "bramble", "eyestalk"}, false) then
                floor, subfloor = "The Outdoors", "The Outdoors"
            elseif anyNameContains({"mines", "minecart", "the nest", "nestroom", "damroom"}, false) then
                floor, subfloor = "The Mines", "The Mines"
            elseif anyNameContains({"hotel-", "hotelminus", "hotelminusvision"}, false) then
                floor, subfloor = "Hotel-", "Hotel- Vision"
            elseif anyNameContains({"lobby", "lobbyfloor", "lobbyroom", "lobbyrooms"}, false) then
                -- Lobby markers must be checked before the generic CurrentRooms fallback;
                -- MSDoors can keep a CurrentRooms folder present while the player is in the lobby.
                floor, subfloor = "Lobby", "Lobby"
            elseif anyNameContains({"hotel-", "hotelminus", "hotelminusvision"}, false) then
                floor, subfloor = "Hotel-", "Hotel- Vision"
            elseif anyNameContains({"hotel"}, false) then
                floor, subfloor = "The Hotel", "The Hotel"
            elseif anyNameContains({"rooms", "a-000", "a000"}, false) then
                floor, subfloor = "The Rooms", "The Rooms Vision"
            elseif activeRooms then
                floor, subfloor = "The Hotel", "The Hotel"
            end

            -- The Mines contains named regions; only label a subfloor when an
            -- actual region marker is found in the active room's descendants.
            if floor == "The Mines" then
                if anyNameContains({"stairwell"}, true) then
                    subfloor = "The Stairwell"
                elseif anyNameContains({"nest", "grumble"}, true) then
                    subfloor = "The Mines — The Nest"
                elseif anyNameContains({"dam"}, true) then
                    subfloor = "The Mines — The Dam"
                end
            end

            -- Report the active room number when the current folder uses numeric names.
            return floor, subfloor, roomLabel
        end

            for _ = 1, 8 do
                if not node then break end
                local ancestorName = norm(node.Name)
                if string.find(ancestorName, "timelever", 1, true)
                    or string.find(ancestorName, "candlemedium", 1, true)
                    or string.find(ancestorName, "rippedwall", 1, true) then
                    return false
                end
                node = node.Parent
            end

            -- Exact names only. Prefix/suffix matching and generic "item"/"pickup"
            -- prompt heuristics caused scenery and room mechanisms to trigger alerts.
            local n = norm(obj.Name)
            if not known[n] and not itemAliases[n] then return false end
            return obj:IsA("Model") or obj:IsA("Tool")
        end
        pcall(function()
            settingsBox:AddDropdown("FairwellItemNotificationChoices", {
                Text = "Notify for these items",
                Values = itemChoices,
                Default = (function()
                    local defaults = {}
                    for _, choice in ipairs(itemChoices) do
                        if importantItems[norm(choice)] then table.insert(defaults, choice) end
                    end
                    return defaults
                end)(),
                Multi = true,
                AllowNull = true,
                Callback = function(values)
                    selectedItemNames = {}
                    if type(values) == "table" then
                        for key, value in pairs(values) do
                            if type(key) == "number" and type(value) == "string" then
                                selectedItemNames[value] = true
                            elseif type(key) == "string" and value == true then
                                selectedItemNames[key] = true
                            end
                        end
                    end
                    setStatus("selection updated: " .. tostring((function()
                        local n = 0
                        for _ in pairs(selectedItemNames) do n += 1 end
                        return n
                    end)()) .. " item types")
                end
            })
        end)
        pcall(function()
            settingsBox:AddToggle("FairwellAllFloorItemNotifications", {
                Text="Item Notifications", Default=false,
                Callback=function(enabled)
                    itemNotifyState.enabled = enabled == true
                    seen = {}
                    startedAt = os.clock()
                    setStatus(itemNotifyState.enabled and "scanning all floors" or "off")
                end
            })
        end)
        task.spawn(function()
            while itemNotifyState.running do
                if itemNotifyState.enabled then
                    local count = 0
                    for _, obj in ipairs(workspace:GetDescendants()) do
                        if not itemNotifyState.enabled then break end
                        if (obj:IsA("Model") or obj:IsA("Tool")) and candidate(obj) then
                            count += 1
                            local key = obj:GetFullName()
                            if not seen[key] then
                                seen[key] = true
                                local rawDetectedName = obj.Name:gsub("_", " ")
                                local detectedName = itemAliases[norm(rawDetectedName)] or rawDetectedName
                                local detectedNorm = norm(detectedName)
                                local wantsNotification = false
                                for choice in pairs(selectedItemNames) do
                                    local choiceNorm = norm(choice)
                                    if detectedNorm == choiceNorm then
                                        wantsNotification = true
                                        break
                                    end
                                end
                                local floor = currentFloor()
                                local requiredFloor
                                for choice, onlyFloor in pairs(itemFloor) do
                                    if norm(choice) == detectedNorm then requiredFloor = onlyFloor break end
                                end
                                local correctFloor = not requiredFloor or floor == requiredFloor
                                local isImportant = importantItems[detectedNorm] == true
                                if wantsNotification and correctFloor and os.clock() - startedAt > 2 then
                                    local detectedObject = obj
                                    local waitBeforePopup = popupSettings.delay
                                    task.delay(waitBeforePopup, function()
                                        if itemNotifyState.running and itemNotifyState.enabled
                                            and detectedObject and detectedObject.Parent
                                            and type(showItemPopup) == "function" then
                                            showItemPopup(detectedName)
                                        end
                                    end)
                                end
                            end
                        end
                    end
                    local floor, subfloor, roomLabel = currentFloorInfo()
                    setStatus("active: " .. floor .. " / " .. subfloor .. (roomLabel and (" / " .. roomLabel) or "") .. "; " .. tostring(count) .. " item candidates")
                end
                -- Keep scanning responsive without tying the scan loop to popup delay.
                task.wait(0.5)
            end
        end)
    end

    -- Auto Giggle Glowstick: equips and throws a real Glowstick toward a nearby Giggle.
    -- This uses the normal Tool activation path; it does not fire arbitrary remotes.
    local gigglePlayers = game:GetService("Players")
    local giggleRunService = game:GetService("RunService")
    local gigglePlayer = gigglePlayers.LocalPlayer
    local giggleEnv = (getgenv and getgenv()) or _G
    if giggleEnv._FairwellAutoGiggleGlowstickState then
        giggleEnv._FairwellAutoGiggleGlowstickState.running = false
    end
    local giggleState = {running = true, enabled = false}
    giggleEnv._FairwellAutoGiggleGlowstickState = giggleState

    local giggleBox
    pcall(function()
        giggleBox = tab:AddRightGroupbox("The Mines", "target")
    end)

    if giggleBox then
        local giggleStatus
        local lastGiggleThrow = 0
        local busy = false
        local statusText = ""

        local function setGiggleStatus(message)
            if message == statusText then return end
            statusText = message
            if giggleStatus then pcall(function() giggleStatus:SetText("Status: " .. message) end) end
        end

        pcall(function()
            giggleStatus = giggleBox:AddLabel({Text = "Status: off", DoesWrap = true})
        end)
        pcall(function()
            giggleBox:AddLabel({
                Text = "Locks the camera onto a nearby Giggle and rapidly throws Glowsticks. Stops when inventory is empty, target disappears, or the safety limit is reached.",
                DoesWrap = true
            })
        end)

        local function isGiggleModel(model)
            if not model or not model:IsA("Model") then return false end
            local name = string.lower(model.Name)
            return string.find(name, "giggle", 1, true) ~= nil
                and string.find(name, "dead", 1, true) == nil
                and string.find(name, "corpse", 1, true) == nil
        end

        local function modelAimPart(model)
            if not model then return nil end
            return model.PrimaryPart
                or model:FindFirstChild("Head", true)
                or model:FindFirstChild("HumanoidRootPart", true)
                or model:FindFirstChildWhichIsA("BasePart", true)
        end

        local function findNearestGiggle(root)
            local bestModel, bestPart, bestDistance
            for _, obj in ipairs(workspace:GetDescendants()) do
                if isGiggleModel(obj) then
                    local part = modelAimPart(obj)
                    if part and part:IsDescendantOf(workspace) then
                        local distance = (part.Position - root.Position).Magnitude
                        if distance <= 65 and (not bestDistance or distance < bestDistance) then
                            bestModel, bestPart, bestDistance = obj, part, distance
                        end
                    end
                end
            end
            return bestModel, bestPart, bestDistance
        end

        local function findGlowstick(character, humanoid)
            local backpack = gigglePlayer:FindFirstChildOfClass("Backpack")
            if character then
                for _, item in ipairs(character:GetChildren()) do
                    if item:IsA("Tool") and string.find(string.lower(item.Name), "glowstick", 1, true) then
                        return item
                    end
                end
            end
            if backpack then
                for _, item in ipairs(backpack:GetChildren()) do
                    if item:IsA("Tool") and string.find(string.lower(item.Name), "glowstick", 1, true) then
                        return item
                    end
                end
            end
            return nil
        end

        local function countGlowsticks(character)
            local count = 0
            local backpack = gigglePlayer:FindFirstChildOfClass("Backpack")
            for _, container in ipairs({character, backpack}) do
                if container then
                    for _, item in ipairs(container:GetChildren()) do
                        if item:IsA("Tool") and string.find(string.lower(item.Name), "glowstick", 1, true) then count += 1 end
                    end
                end
            end
            return count
        end

        local function throwAtGiggle(tool, character, humanoid, targetPart)
            if busy or not tool or not targetPart or not targetPart.Parent then return false end
            busy = true
            local camera = workspace.CurrentCamera
            local root = character and character:FindFirstChild("HumanoidRootPart")
            local oldCameraCFrame = camera and camera.CFrame
            local oldCameraType = camera and camera.CameraType
            local oldAutoRotate = humanoid and humanoid.AutoRotate
            local lockConnection
            local ok, err = pcall(function()
                if tool.Parent ~= character then humanoid:EquipTool(tool) end
                task.wait(0.12)
                if tool.Parent ~= character then error("Glowstick did not equip") end
                local folder = character:FindFirstChild("Glowsticks")
                local remote = folder and folder:FindFirstChild("Remote")
                if not (remote and remote:IsA("RemoteEvent")) then remote = tool:FindFirstChild("Remote", true) end
                if not (remote and remote:IsA("RemoteEvent")) then error("Glowsticks Remote not found") end
                if not root or not root:IsA("BasePart") then error("HumanoidRootPart not found") end

                -- Face the target with the complete character pivot before sending the normal
                -- Glowsticks request. This is the strongest client-side rotation attempt, but
                -- the game server may still override it if character rotation is server-owned.
                humanoid.AutoRotate = false
                local lastFacing = nil
                local function faceTarget()
                    if not targetPart.Parent or not root.Parent or not character.Parent then return end
                    local pos = root.Position
                    local target = targetPart.Position
                    local flatTarget = Vector3.new(target.X, pos.Y, target.Z)
                    local delta = flatTarget - pos
                    if delta.Magnitude > 0.05 then
                        local facing = CFrame.lookAt(pos, flatTarget)
                        local currentPivot = character:GetPivot()
                        local rootToPivot = root.CFrame:ToObjectSpace(currentPivot)
                        character:PivotTo(facing * rootToPivot)
                        lastFacing = facing
                    end
                    if camera then
                        camera.CameraType = Enum.CameraType.Scriptable
                        local lookFrom = root.Position - root.CFrame.LookVector * 8 + Vector3.new(0, 3.5, 0)
                        camera.CFrame = CFrame.lookAt(lookFrom, target)
                    end
                end
                faceTarget()
                lockConnection = giggleRunService.RenderStepped:Connect(faceTarget)

                -- Let the rotation replicate for a brief moment before using the game's normal throw.
                task.wait(0.18)
                local started, throws = os.clock(), 0
                while giggleState.enabled and targetPart.Parent and countGlowsticks(character) > 0
                    and throws < 40 and os.clock() - started < 5 do
                    faceTarget()
                    remote:FireServer()
                    throws += 1
                    task.wait(0.09)
                end
                setGiggleStatus("faced target locally; sent " .. throws .. " throw requests. Server-visible facing depends on DOORS replication.")
            end)
            if lockConnection then pcall(function() lockConnection:Disconnect() end) end
            if humanoid and humanoid.Parent and oldAutoRotate ~= nil then
                pcall(function() humanoid.AutoRotate = oldAutoRotate end)
            end
            if camera and oldCameraCFrame and camera.Parent then
                pcall(function()
                    if oldCameraType then camera.CameraType = oldCameraType end
                    camera.CFrame = oldCameraCFrame
                end)
            end
            busy = false
            if not ok then setGiggleStatus("throw attempt failed: " .. tostring(err)); return false end
            lastGiggleThrow = os.clock()
            return true
        end

        pcall(function()
            giggleBox:AddToggle("FairwellAutoGiggleGlowstick", {
                Text = "Auto-throw Glowsticks at Giggles",
                Default = false,
                Callback = function(enabled)
                    giggleState.enabled = enabled == true
                    setGiggleStatus(giggleState.enabled and "searching for Giggle" or "off")
                end
            })
        end)

        task.spawn(function()
            while giggleState.running do
                task.wait(0.25)
                if giggleState.enabled and not busy then
                    local character = gigglePlayer.Character
                    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                    local root = character and character:FindFirstChild("HumanoidRootPart")
                    if not character or not humanoid or humanoid.Health <= 0 or not root then
                        setGiggleStatus("waiting for living character")
                    elseif os.clock() - lastGiggleThrow < 1.8 then
                        -- Prevent repeated activation while the throw animation/item state settles.
                    else
                        local model, part, distance = findNearestGiggle(root)
                        if not model or not part then
                            setGiggleStatus("enabled; no Giggle detected within 65 studs")
                        else
                            local tool = findGlowstick(character, humanoid)
                            if not tool then
                                setGiggleStatus(string.format("Giggle %.0f studs away; no Glowstick in inventory", distance))
                            else
                                throwAtGiggle(tool, character, humanoid, part)
                            end
                        end
                    end
                end
            end
        end)
    end
