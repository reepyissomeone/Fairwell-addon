-- Fairwell Heaven DOORS feature loader for MSDoors
-- Source is split into readable files under src/features and src/config.
-- The fragments are concatenated in their original order before compilation,
-- preserving the original shared Lua scope and runtime behavior.

local MODULE_URLS = {
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/refactor/organize-feature-files/src/features/01-communicator-and-stem.txt",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/refactor/organize-feature-files/src/features/02-anti-creak.txt",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/refactor/organize-feature-files/src/features/03-anti-groundskeeper.txt",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/refactor/organize-feature-files/src/features/04-auto-closet.txt",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/refactor/organize-feature-files/src/features/05-delete-gate.txt",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/refactor/organize-feature-files/src/features/06-friendly-entities.txt",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/refactor/organize-feature-files/src/features/07-item-notifications-and-floor-detection.txt",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/refactor/organize-feature-files/src/features/08-auto-giggle-glowstick.txt",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/refactor/organize-feature-files/src/config/config-manager.txt",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/refactor/organize-feature-files/src/features/10-remaining-content.txt",
}

local pieces = {}
for index, url in ipairs(MODULE_URLS) do
    local ok, source = pcall(function()
        return game:HttpGet(url)
    end)
    if not ok or type(source) ~= "string" or source == "" then
        error("[Fairwell] Failed to download source module " .. tostring(index) .. ": " .. url)
    end
    pieces[index] = source
end

local source = table.concat(pieces, "\n")
local compileOk, chunk = pcall(loadstring, source)
if not compileOk or type(chunk) ~= "function" then
    error("[Fairwell] Combined feature source failed to compile: " .. tostring(chunk))
end

local runOk, builder = pcall(chunk)
if not runOk or type(builder) ~= "function" then
    error("[Fairwell] Combined feature source did not return a builder: " .. tostring(builder))
end

return builder
