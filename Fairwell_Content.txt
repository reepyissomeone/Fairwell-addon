-- Fairwell Heaven DOORS features for MSDoors
-- Remote content file. Only gameplay features are shown in this tab.

return function(tab, reloadContent)
    if not tab then return end

    local stairwellBox
    pcall(function()
        stairwellBox = tab:AddLeftGroupbox("Stairwell", "eye")
    end)

    local espOk, espBox = pcall(function()
        return stairwellBox
    end)

    if espOk and espBox then
        local espEnabled = false
        local espFolder = Instance.new("Folder")
        espFolder.Name = "FairwellStemESP"
        espFolder.Parent = game:GetService("CoreGui")

        local function clearESP()
            for _, child in ipairs(espFolder:GetChildren()) do
                child:Destroy()
            end
        end

        local function stemRelated(instance)
            local node = instance
            for _ = 1, 5 do
                if not node then break end
                if string.find(string.lower(node.Name), "stem", 1, true) then
                    return true
                end
                node = node.Parent
            end
            return false
        end

        local function scan()
            clearESP()
            local seen = {}
            for _, instance in ipairs(workspace:GetDescendants()) do
                if stemRelated(instance) then
                    local target = instance:IsA("Model") and instance or instance:FindFirstAncestorOfClass("Model")
                    target = target or (instance:IsA("BasePart") and instance or nil)
                    if target and not seen[target] then
                        seen[target] = true
                        local holder = Instance.new("Folder")
                        holder.Name = "StemESPItem"
                        holder.Parent = espFolder

                        local highlight = Instance.new("Highlight")
                        highlight.Adornee = target
                        highlight.FillColor = Color3.fromRGB(255, 70, 70)
                        highlight.OutlineColor = Color3.fromRGB(255, 230, 70)
                        highlight.FillTransparency = 0.55
                        highlight.OutlineTransparency = 0
                        highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        highlight.Parent = holder

                        local part = target:IsA("BasePart") and target or target:FindFirstChildWhichIsA("BasePart", true)
                        if part then
                            local labelGui = Instance.new("BillboardGui")
                            labelGui.Name = "StemLabel"
                            labelGui.Adornee = part
                            labelGui.AlwaysOnTop = true
                            labelGui.Size = UDim2.fromOffset(110, 30)
                            labelGui.StudsOffset = Vector3.new(0, 2.5, 0)
                            labelGui.Parent = holder

                            local label = Instance.new("TextLabel")
                            label.BackgroundTransparency = 1
                            label.Size = UDim2.fromScale(1, 1)
                            label.Text = "STEM"
                            label.TextColor3 = Color3.fromRGB(255, 230, 70)
                            label.TextStrokeTransparency = 0
                            label.TextScaled = true
                            label.Font = Enum.Font.GothamBold
                            label.Parent = labelGui
                        end
                    end
                end
            end
        end

        pcall(function()
            espBox:AddToggle("FairwellStemESPToggle", {
                Text = "STEM ESP",
                Default = false,
                Callback = function(enabled)
                    espEnabled = enabled == true
                    clearESP()
                    if espEnabled then scan() end
                end
            })
        end)

        task.spawn(function()
            while espFolder.Parent do
                if espEnabled then scan() end
                task.wait(2)
            end
        end)
    end

    local stemOk, stemBox = pcall(function()
        return stairwellBox
    end)

    if stemOk and stemBox then
        local Players = game:GetService("Players")
        local ReplicatedStorage = game:GetService("ReplicatedStorage")
        local player = Players.LocalPlayer
        local env = (getgenv and getgenv()) or _G
        env._FairwellAutoKillStem = false
        local statusLabel

        local function setStatus(message)
            print("[Fairwell Auto kill Stem] " .. message)
            if statusLabel then pcall(function() statusLabel:SetText("Status: " .. message) end) end
        end
        pcall(function() statusLabel = stemBox:AddLabel({Text = "Status: idle", DoesWrap = true}) end)

        local function findHeldStem()
            local character = player.Character
            if not character then return nil end
            local tool = character:FindFirstChild("Stem")
            if not tool or not tool:IsA("Tool") then return nil end
            local handle = tool:FindFirstChild("Handle")
            if not handle or not handle:IsA("BasePart") then return tool, nil, nil end
            for _, item in ipairs(handle:GetDescendants()) do
                if item.Name:match("^Stem_") then return tool, handle, item.Name end
            end
            return tool, handle, nil
        end

        local function findVariant(tool, handle, identifier)
            -- Check the held item, its identifier, attributes, and exposed string values.
            local function inspect(value)
                local lower = string.lower(tostring(value or ""))
                for _, variant in ipairs({"Mouth", "Ear", "Eye", "Blank"}) do
                    if string.find(lower, string.lower(variant), 1, true) then return variant end
                end
            end
            local found = inspect(identifier)
            if found then return found end
            local roots = {tool}
            if handle then table.insert(roots, handle) end
            for _, root in ipairs(roots) do
                for key, value in pairs(root:GetAttributes()) do
                    found = inspect(key) or inspect(value)
                    if found then return found end
                end
                for _, item in ipairs(root:GetDescendants()) do
                    found = inspect(item.Name)
                    if not found and item:IsA("StringValue") then found = inspect(item.Value) end
                    if not found then
                        for key, value in pairs(item:GetAttributes()) do
                            found = inspect(key) or inspect(value)
                            if found then break end
                        end
                    end
                    if found then return found end
                end
            end
            return "Blank"
        end

        local function findNearbyStemPrompt()
            local character = player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not root then return nil, "character root not found" end
            local bestPrompt, bestDistance
            for _, prompt in ipairs(workspace:GetDescendants()) do
                if prompt:IsA("ProximityPrompt") and prompt.Enabled then
                    local parent = prompt.Parent
                    local part = parent and (parent:IsA("BasePart") and parent or parent:FindFirstChildWhichIsA("BasePart", true))
                    if part then
                        local distance = (root.Position - part.Position).Magnitude
                        if distance <= math.max(prompt.MaxActivationDistance + 5, 12) then
                            local node, isStem = prompt, false
                            for _ = 1, 8 do
                                if not node then break end
                                if string.find(string.lower(node.Name), "stem", 1, true) then isStem = true break end
                                node = node.Parent
                            end
                            if isStem and (not bestDistance or distance < bestDistance) then
                                bestPrompt, bestDistance = prompt, distance
                            end
                        end
                    end
                end
            end
            return bestPrompt, bestPrompt and "nearby Stem pickup found" or "searching for a nearby Stem"
        end

        local function activatePrompt(prompt)
            if not prompt or not prompt.Parent then return false, "prompt disappeared" end
            local ok, err = pcall(function()
                prompt:InputHoldBegin()
                task.wait(math.max(prompt.HoldDuration, 0.02) + 0.02)
                prompt:InputHoldEnd()
            end)
            if not ok then return false, tostring(err) end
            return true, "pickup attempted"
        end

        local function fireHeldStem(tool, handle, identifier)
            if not handle then return false, "held Stem Handle not found" end
            if not identifier then return false, "Stem_ identifier not found under Handle" end
            local remotes = ReplicatedStorage:FindFirstChild("RemotesFolder")
            local remote = remotes and remotes:FindFirstChild("FireProjectile")
            if not remote or not remote:IsA("RemoteEvent") then return false, "FireProjectile RemoteEvent not found" end
            local camera = workspace.CurrentCamera
            if not camera then return false, "camera unavailable" end
            local direction = camera.CFrame.LookVector
            local velocity = direction * 100
            local exclusions = {}
            local props = workspace:FindFirstChild("Props")
            local character = player.Character
            if props then table.insert(exclusions, props) end
            if character then table.insert(exclusions, character) end
            local variant = findVariant(tool, handle, identifier)
            local ok, err = pcall(function()
                remote:FireServer(identifier, handle.Position, direction, velocity, "Stem", variant, exclusions)
            end)
            if not ok then return false, tostring(err) end
            return true, "threw " .. variant .. " Stem"
        end

        pcall(function()
            stemBox:AddToggle("FairwellAutoKillStemToggle", {
                Text = "Auto kill Stem",
                Default = false,
                Callback = function(enabled)
                    env._FairwellAutoKillStem = enabled == true
                    if not enabled then setStatus("off") return end
                    task.spawn(function()
                        setStatus("enabled; auto-picking up and throwing Stems")
                        local lastIdentifier
                        local lastInfo
                        while env._FairwellAutoKillStem do
                            local character = player.Character
                            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                            if not humanoid or humanoid.Health <= 0 then
                                setStatus("paused: character unavailable/dead")
                                task.wait(0.25)
                            else
                                local tool, handle, identifier = findHeldStem()
                                if tool and handle and identifier and identifier ~= lastIdentifier then
                                    local ok, message = fireHeldStem(tool, handle, identifier)
                                    setStatus(ok and message or ("throw failed: " .. message))
                                    if ok then lastIdentifier = identifier end
                                    task.wait(ok and 0.08 or 0.15)
                                else
                                    -- If no usable held Stem is available, seek and pick one up.
                                    if not tool or not handle or not identifier then lastIdentifier = nil end
                                    local prompt, info = findNearbyStemPrompt()
                                    if info ~= lastInfo then setStatus(info); lastInfo = info end
                                    if prompt then
                                        local ok, message = activatePrompt(prompt)
                                        setStatus(ok and message or ("pickup failed: " .. message))
                                        task.wait(0.04)
                                    else
                                        task.wait(0.08)
                                    end
                                end
                            end
                        end
                        setStatus("off")
                    end)
                end
            })
        end)
    end

    -- Anti Groundskeeper: CFrame hover while enabled; automatically stays above detected ground.
    local hoverOk, hoverBox = pcall(function()
        return tab:AddRightGroupbox("Outdoors", "wind")
    end)

    if hoverOk and hoverBox then
        local Players = game:GetService("Players")
        local RunService = game:GetService("RunService")
        local player = Players.LocalPlayer
        local env = (getgenv and getgenv()) or _G
        env._FairwellAntiGroundskeeper = false
        local hoverConnection
        local characterAddedConnection
        local statusLabel
        local lastStatus

        pcall(function()
            statusLabel = hoverBox:AddLabel({Text = "Status: off", DoesWrap = true})
        end)

        local function status(message)
            if message == lastStatus then return end
            lastStatus = message
            if statusLabel then
                pcall(function() statusLabel:SetText("Status: " .. message) end)
            end
        end

        local function stopHover()
            if hoverConnection then hoverConnection:Disconnect(); hoverConnection = nil end
            status("off")
        end

        local function startHover()
            stopHover()
            local character = player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            if not root or not humanoid then
                status("waiting for character")
                return
            end

            local rayParams = RaycastParams.new()
            rayParams.FilterType = Enum.RaycastFilterType.Exclude
            rayParams.FilterDescendantsInstances = {character}
            local hoverHeight = 3.0
            local baseY = root.Position.Y
            local horizontalVelocity = Vector3.zero
            local lastGroundY

            hoverConnection = RunService.Heartbeat:Connect(function(dt)
                if not env._FairwellAntiGroundskeeper then stopHover(); return end
                if not character.Parent or not root.Parent or not humanoid.Parent or humanoid.Health <= 0 then
                    status("waiting for character")
                    return
                end

                -- CFrame movement bypasses the need to jump to gain clearance.
                -- Raycast downward to maintain a consistent gap above the surface.
                local result = workspace:Raycast(root.Position, Vector3.new(0, -22, 0), rayParams)
                if not result then
                    status("no ground detected")
                    return
                end

                local material = result.Material and result.Material.Name or ""
                local node = result.Instance
                local grassName = false
                for _ = 1, 5 do
                    if not node then break end
                    if string.find(string.lower(node.Name), "grass", 1, true) then
                        grassName = true
                        break
                    end
                    node = node.Parent
                end

                -- Also check nearby grass parts for maps with separate grass visuals.
                local isGrass = material == "Grass" or material == "LeafyGrass" or grassName
                if not isGrass then
                    for _, instance in ipairs(workspace:GetDescendants()) do
                        if instance:IsA("BasePart") and string.find(string.lower(instance.Name), "grass", 1, true)
                            and (instance.Position - root.Position).Magnitude <= 9
                            and math.abs(instance.Position.Y - result.Position.Y) <= 5 then
                            isGrass = true
                            break
                        end
                    end
                end

                if isGrass then
                    local targetY = result.Position.Y + humanoid.HipHeight + (root.Size.Y * 0.5) + hoverHeight
                    local current = root.Position
                    local alpha = math.clamp(dt * 12, 0, 1)
                    local newY = current.Y + (targetY - current.Y) * alpha
                    -- Preserve the character's current facing and horizontal position.
                    character:PivotTo(CFrame.new(current.X, newY, current.Z) * (root.CFrame - root.Position))
                    -- Cancel vertical momentum so the character doesn't fall back onto the grass.
                    local velocity = root.AssemblyLinearVelocity
                    root.AssemblyLinearVelocity = Vector3.new(velocity.X, 0, velocity.Z)
                    status("CFrame hovering above grass")
                    lastGroundY = result.Position.Y
                else
                    status("ground detected; grass not identified")
                end
            end)
        end

        pcall(function()
            hoverBox:AddToggle("FairwellAntiGroundskeeperToggle", {
                Text = "Anti Groundskeeper",
                Default = false,
                Callback = function(enabled)
                    env._FairwellAntiGroundskeeper = enabled == true
                    if enabled then startHover() else stopHover() end
                end
            })
        end)

        characterAddedConnection = player.CharacterAdded:Connect(function()
            if env._FairwellAntiGroundskeeper then
                task.wait(1)
                startHover()
            end
        end)
    end


    -- Auto Closet: detect Rush/Ambush models and use hiding ProximityPrompts.
    local closetOk, closetBox = pcall(function()
        return tab:AddRightGroupbox("Any Floor", "door-open")
    end)

    if closetOk and closetBox then
        local Players = game:GetService("Players")
        local player = Players.LocalPlayer
        local ReplicatedStorage = game:GetService("ReplicatedStorage")
        local env = (getgenv and getgenv()) or _G
        env._FairwellAutoCloset = false
        local getOutEvent
        pcall(function()
            local remotes = ReplicatedStorage:FindFirstChild("RemotesFolder")
            local event = remotes and remotes:FindFirstChild("GetOutOfHiding")
            if event and event:IsA("RemoteEvent") then getOutEvent = event end
        end)
        local statusLabel
        local function closetStatus(message)
            if statusLabel then
                pcall(function() statusLabel:SetText("Status: " .. message) end)
            end
        end
        pcall(function()
            statusLabel = closetBox:AddLabel({Text = "Status: off", DoesWrap = true})
        end)

        local function getCharacter()
            local character = player.Character
            return character, character and character:FindFirstChild("HumanoidRootPart"),
                character and character:FindFirstChildOfClass("Humanoid")
        end

        local function threatKind(name)
            local lower = string.lower(name or "")
            -- Rush-style threats that can be survived by entering a hiding spot.
            -- Ambush, Blitz, and A-120 may rebound, so their monitor waits for them to disappear.
            if string.find(lower, "ambush", 1, true) then return "Ambush" end
            if string.find(lower, "blitz", 1, true) then return "Blitz" end
            if string.find(lower, "a%-?120", 1) or string.find(lower, "a120", 1, true) then return "A-120" end
            if string.find(lower, "a%-?60", 1) or string.find(lower, "a60", 1, true) then return "A-60" end
            if string.find(lower, "rush", 1, true) then return "Rush" end
            if string.find(lower, "surge", 1, true) then return "Surge" end
            return nil
        end

        local function findThreat(root)
            local best, bestDistance, bestKind
            local seen = {}
            for _, item in ipairs(workspace:GetDescendants()) do
                if item:IsA("Model") or item:IsA("BasePart") then
                    local kind = threatKind(item.Name)
                    -- Avoid counting every named part inside the same entity as a separate threat.
                    local model = item:IsA("Model") and item or item:FindFirstAncestorOfClass("Model")
                    if kind and (not model or not seen[model]) then
                        local target = model or item
                        local targetKind = threatKind(target.Name) or kind
                        local part = target:IsA("BasePart") and target or target.PrimaryPart
                            or target:FindFirstChildWhichIsA("BasePart", true)
                        if part then
                            local distance = (root.Position - part.Position).Magnitude
                            if model then seen[model] = true end
                            if distance <= 220 and (not bestDistance or distance < bestDistance) then
                                best, bestDistance, bestKind = target, distance, targetKind
                            end
                        end
                    end
                end
            end
            return best, bestDistance, bestKind
        end

        local function promptPosition(prompt)
            local node = prompt.Parent
            while node and node ~= workspace do
                if node:IsA("BasePart") then return node.Position, node end
                if node:IsA("Model") then
                    local part = node.PrimaryPart or node:FindFirstChildWhichIsA("BasePart", true)
                    if part then return part.Position, part end
                end
                node = node.Parent
            end
            return nil, nil
        end

        local function isHidingPrompt(prompt)
            local words = string.lower(tostring(prompt.Name) .. " "
                .. tostring(prompt.ActionText) .. " " .. tostring(prompt.ObjectText))
            if string.find(words, "hide", 1, true)
                or string.find(words, "closet", 1, true)
                or string.find(words, "wardrobe", 1, true)
                or string.find(words, "locker", 1, true) then
                return true
            end
            local node = prompt.Parent
            for _ = 1, 8 do
                if not node then break end
                local name = string.lower(node.Name)
                if string.find(name, "closet", 1, true)
                    or string.find(name, "wardrobe", 1, true)
                    or string.find(name, "locker", 1, true)
                    or string.find(name, "hiding", 1, true) then
                    return true
                end
                node = node.Parent
            end
            return false
        end

        local function findClosestCloset(root)
            local bestPrompt, bestPosition, bestDistance
            for _, prompt in ipairs(workspace:GetDescendants()) do
                if prompt:IsA("ProximityPrompt") and prompt.Enabled and isHidingPrompt(prompt) then
                    local position = promptPosition(prompt)
                    if position then
                        local distance = (root.Position - position).Magnitude
                        if distance <= 125 and (not bestDistance or distance < bestDistance) then
                            bestPrompt, bestPosition, bestDistance = prompt, position, distance
                        end
                    end
                end
            end
            return bestPrompt, bestPosition, bestDistance
        end

        local function triggerGetOutOfHiding()
            -- Fire the local GetOutOfHiding listeners if the executor exposes firesignal.
            if not getOutEvent then return false end
            local ok = pcall(function()
                if type(firesignal) ~= "function" then error("firesignal unavailable") end
                firesignal(getOutEvent.OnClientEvent)
            end)
            return ok
        end

        local busy = false
        local lastAction = 0
        local function tryHide()
            if busy or not env._FairwellAutoCloset or os.clock() - lastAction < 1.5 then return end
            busy = true
            lastAction = os.clock()

            local character, root, humanoid = getCharacter()
            if not character or not root or not humanoid or humanoid.Health <= 0 then
                closetStatus("waiting for character")
                busy = false
                return
            end

            local threat, threatDistance, kind = findThreat(root)
            if not threat then
                closetStatus("searching for Rush / Ambush")
                busy = false
                return
            end

            local prompt, position, distance = findClosestCloset(root)
            if not prompt then
                closetStatus(kind .. " detected; no hiding ProximityPrompt found")
                busy = false
                return
            end

            closetStatus(kind .. " detected; approaching hiding prompt")
            if distance > math.max(prompt.MaxActivationDistance - 1, 4) then
                humanoid:MoveTo(position)
                local deadline = os.clock() + math.min(6, distance / math.max(humanoid.WalkSpeed, 8) + 1.5)
                repeat
                    task.wait(0.12)
                    character, root, humanoid = getCharacter()
                    if not env._FairwellAutoCloset or not root or not humanoid or humanoid.Health <= 0 then break end
                    if not prompt.Parent or not prompt.Enabled then break end
                    position = promptPosition(prompt)
                    if not position then break end
                    distance = (root.Position - position).Magnitude
                until distance <= math.max(prompt.MaxActivationDistance - 1, 4) or os.clock() >= deadline
            end

            if env._FairwellAutoCloset and prompt.Parent and prompt.Enabled then
                local ok = pcall(function()
                    prompt:InputHoldBegin()
                    task.wait(math.max(prompt.HoldDuration, 0.05) + 0.08)
                    prompt:InputHoldEnd()
                end)
                if ok then
                    closetStatus("used hiding ProximityPrompt")
                    task.spawn(function()
                        local started = os.clock()
                        local lastSeen = os.clock()
                        local exited = false
                        local finished = false
                        local maxDuration = (kind == "Ambush" or kind == "Blitz" or kind == "A-120") and 180 or 60
                        local previousDistance
                        local increasingSamples = 0
                        local closestDistance = math.huge
                        local isReboundThreat = kind == "Ambush" or kind == "Blitz" or kind == "A-120"

                        while env._FairwellAutoCloset and (os.clock() - started) < maxDuration do
                            task.wait(0.2)

                            local currentCharacter = player.Character
                            local currentRoot = currentCharacter and currentCharacter:FindFirstChild("HumanoidRootPart")
                            local currentHumanoid = currentCharacter and currentCharacter:FindFirstChildOfClass("Humanoid")
                            if not currentRoot or not currentHumanoid or currentHumanoid.Health <= 0 then
                                break
                            end

                            local liveThreat, liveDistance, liveKind = findThreat(currentRoot)
                            if liveThreat and liveKind == kind then
                                lastSeen = os.clock()
                                if liveDistance then
                                    closestDistance = math.min(closestDistance, liveDistance)
                                    if previousDistance and liveDistance > previousDistance + 2 then
                                        increasingSamples = increasingSamples + 1
                                    else
                                        increasingSamples = 0
                                    end
                                    previousDistance = liveDistance
                                end

                                if isReboundThreat then
                                    -- Never exit just because the entity is far away: it may rebound.
                                    closetStatus(kind .. " active; staying hidden until it disappears")
                                elseif not exited and liveDistance and closestDistance <= 28
                                    and liveDistance >= 38 and increasingSamples >= 3 then
                                    -- Confirm it passed: it got close, then moved away across multiple samples.
                                    if triggerGetOutOfHiding() then
                                        exited = true
                                        closetStatus(kind .. " passed; leaving hiding spot")
                                    else
                                        closetStatus(kind .. " passed; exit signal unavailable")
                                    end
                                elseif exited then
                                    closetStatus(kind .. " passed; encounter monitored")
                                else
                                    closetStatus(kind .. " approaching/passing; staying hidden")
                                end
                            else
                                local missingFor = os.clock() - lastSeen
                                local grace = (kind == "Ambush" or kind == "Blitz" or kind == "A-120") and 5 or 1.2
                                if missingFor >= grace then
                                    if not exited then
                                        triggerGetOutOfHiding()
                                    end
                                    closetStatus(kind .. " despawned; encounter monitor stopped")
                                    finished = true
                                    break
                                else
                                    closetStatus(kind .. " temporarily absent; monitoring")
                                end
                            end
                        end

                        if not finished and env._FairwellAutoCloset then
                            closetStatus(kind .. " monitoring ended; toggle Auto Closet off/on to retry")
                        end
                    end)
                else
                    closetStatus("prompt interaction failed")
                end
            end
            busy = false
        end

        pcall(function()
            closetBox:AddToggle("FairwellAutoClosetToggle", {
                Text = "Auto Closet",
                Default = false,
                Callback = function(enabled)
                    env._FairwellAutoCloset = enabled == true
                    closetStatus(enabled and "searching for Rush / Ambush" or "off")
                end
            })
        end)

        task.spawn(function()
            while true do
                if env._FairwellAutoCloset then
                    local _, root, humanoid = getCharacter()
                    if root and humanoid and humanoid.Health > 0 then
                        local threat, distance, kind = findThreat(root)
                        if threat and distance then
                            tryHide()
                        else
                            closetStatus("searching for Rush / Ambush")
                        end
                    else
                        closetStatus("waiting for character")
                    end
                end
                task.wait(0.3)
            end
        end)
    end


end
