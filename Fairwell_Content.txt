-- Fairwell Heaven DOORS feature loader for MSDoors
-- Feature source is organized under src/features and src/config.
-- The source fragments are concatenated in their original order before compilation,
-- preserving the original shared Lua scope and behavior.

local MODULE_URLS = {
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/src/features/01-communicator-and-stem.txt?v=20261018",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/src/features/02-anti-creak.txt?v=20261018",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/src/features/03-anti-groundskeeper.txt?v=20261018",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/src/features/04-auto-closet.txt?v=20261018",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/src/features/05-delete-gate.txt?v=20261018-2",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/src/features/06-friendly-entities.txt?v=20261018",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/src/features/07-item-notifications-and-floor-detection.txt?v=20261018-3",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/src/features/08-auto-giggle-glowstick.txt?v=20261018",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/src/features/09-auto-open-locked-doors.txt?v=20261018-11",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/src/features/10-feature-connector.txt?v=3",
    "https://raw.githubusercontent.com/reepyissomeone/Fairwell-addon/main/src/config/config-manager.txt?v=20261018-2",
}

local pieces = {}
for index, url in ipairs(MODULE_URLS) do
    local ok, source = pcall(function()
        return game:HttpGet(url)
    end)
    if not ok or type(source) ~= "string" or source == "" then
        error("[Fairwell] Failed to download source module " .. tostring(index) .. ": " .. url)
    end
    pieces[index] = source
end

-- Add an explicit newline between remote fragments so adjacent source files cannot join tokens or comments.
local combinedSource = table.concat(pieces, "\n")
local env = (getgenv and getgenv()) or _G
local compile = (type(loadstring) == "function" and loadstring) or env.loadstring
if type(compile) ~= "function" then
    error("[Fairwell] No loadstring compiler is available in this executor.")
end
-- Call the executor compiler through a closure so wrappers with custom call
-- conventions receive the source as an explicit first argument.
local compileOk, chunk, compileError = pcall(function()
    return compile(tostring(combinedSource))
end)
if not compileOk or type(chunk) ~= "function" then
    error("[Fairwell] Combined feature source failed to compile: "
        .. tostring(compileError or chunk or "unknown compiler error"))
end

local runOk, builder = pcall(chunk)
if not runOk or type(builder) ~= "function" then
    error("[Fairwell] Combined feature source did not return a builder: " .. tostring(builder))
end

return builder
