-- Fairwell Heaven DOORS features for MSDoors
-- Remote content file. Only gameplay features are shown in this tab.

return function(tab, reloadContent)
    if not tab then return end

    local espOk, espBox = pcall(function()
        return tab:AddLeftGroupbox("STEM ESP", "eye")
    end)

    if espOk and espBox then
        local espEnabled = false
        local espFolder = Instance.new("Folder")
        espFolder.Name = "FairwellStemESP"
        espFolder.Parent = game:GetService("CoreGui")

        local function clearESP()
            for _, child in ipairs(espFolder:GetChildren()) do
                child:Destroy()
            end
        end

        local function stemRelated(instance)
            local node = instance
            for _ = 1, 5 do
                if not node then break end
                if string.find(string.lower(node.Name), "stem", 1, true) then
                    return true
                end
                node = node.Parent
            end
            return false
        end

        local function scan()
            clearESP()
            local seen = {}
            for _, instance in ipairs(workspace:GetDescendants()) do
                if stemRelated(instance) then
                    local target = instance:IsA("Model") and instance or instance:FindFirstAncestorOfClass("Model")
                    target = target or (instance:IsA("BasePart") and instance or nil)
                    if target and not seen[target] then
                        seen[target] = true
                        local holder = Instance.new("Folder")
                        holder.Name = "StemESPItem"
                        holder.Parent = espFolder

                        local highlight = Instance.new("Highlight")
                        highlight.Adornee = target
                        highlight.FillColor = Color3.fromRGB(255, 70, 70)
                        highlight.OutlineColor = Color3.fromRGB(255, 230, 70)
                        highlight.FillTransparency = 0.55
                        highlight.OutlineTransparency = 0
                        highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        highlight.Parent = holder

                        local part = target:IsA("BasePart") and target or target:FindFirstChildWhichIsA("BasePart", true)
                        if part then
                            local labelGui = Instance.new("BillboardGui")
                            labelGui.Name = "StemLabel"
                            labelGui.Adornee = part
                            labelGui.AlwaysOnTop = true
                            labelGui.Size = UDim2.fromOffset(110, 30)
                            labelGui.StudsOffset = Vector3.new(0, 2.5, 0)
                            labelGui.Parent = holder

                            local label = Instance.new("TextLabel")
                            label.BackgroundTransparency = 1
                            label.Size = UDim2.fromScale(1, 1)
                            label.Text = "STEM"
                            label.TextColor3 = Color3.fromRGB(255, 230, 70)
                            label.TextStrokeTransparency = 0
                            label.TextScaled = true
                            label.Font = Enum.Font.GothamBold
                            label.Parent = labelGui
                        end
                    end
                end
            end
        end

        pcall(function()
            espBox:AddToggle("FairwellStemESPToggle", {
                Text = "STEM ESP",
                Default = false,
                Callback = function(enabled)
                    espEnabled = enabled == true
                    clearESP()
                    if espEnabled then scan() end
                end
            })
        end)

        task.spawn(function()
            while espFolder.Parent do
                if espEnabled then scan() end
                task.wait(2)
            end
        end)
    end

    local stemOk, stemBox = pcall(function()
        return tab:AddLeftGroupbox("STEM Auto-Throw", "target")
    end)

    if stemOk and stemBox then
        local Players = game:GetService("Players")
        local player = Players.LocalPlayer
        local env = (getgenv and getgenv()) or _G
        env._FairwellStemAutoThrow = false
        local statusLabel

        local function setStatus(message)
            print("[Fairwell STEM] " .. message)
            if statusLabel then
                pcall(function() statusLabel:SetText("Status: " .. message) end)
            end
        end

        pcall(function()
            statusLabel = stemBox:AddLabel({Text = "Status: idle", DoesWrap = true})
        end)

        local function nameHasStem(instance)
            local node = instance
            for _ = 1, 8 do
                if not node then break end
                if string.find(string.lower(node.Name), "stem", 1, true) then
                    return true
                end
                node = node.Parent
            end
            return false
        end

        local function getRoot()
            local character = player and player.Character
            return character and character:FindFirstChild("HumanoidRootPart")
        end

        -- Find Stem-like objects first. The prompt itself often has a generic name,
        -- so don't require the word "Stem" to appear in the prompt's ancestry.
        local function findStemTargets()
            local targets, seen = {}, {}
            for _, item in ipairs(workspace:GetDescendants()) do
                if nameHasStem(item) then
                    local model = item:IsA("Model") and item or item:FindFirstAncestorOfClass("Model")
                    local part = item:IsA("BasePart") and item or
                        (model and model:FindFirstChildWhichIsA("BasePart", true)) or
                        item:FindFirstChildWhichIsA("BasePart", true)
                    local target = model or part
                    if target and part and not seen[target] then
                        seen[target] = true
                        table.insert(targets, {object = target, part = part})
                    end
                end
            end
            return targets
        end

        local function findNearbyStemPrompt()
            local root = getRoot()
            if not root then return nil, "character root not found" end

            local targets = findStemTargets()
            if #targets == 0 then
                return nil, "no Stem-named object found"
            end

            local bestPrompt, bestDistance
            local allPrompts = {}
            for _, item in ipairs(workspace:GetDescendants()) do
                if item:IsA("ProximityPrompt") and item.Enabled then
                    local parent = item.Parent
                    local part = parent and (parent:IsA("BasePart") and parent or
                        parent:FindFirstChildWhichIsA("BasePart", true))
                    if part then
                        local distance = (root.Position - part.Position).Magnitude
                        if distance <= math.max(item.MaxActivationDistance + 5, 12) then
                            table.insert(allPrompts, {prompt = item, part = part, distance = distance})
                        end
                    end
                end
            end

            -- Prefer prompts inside a Stem model, but also allow a nearby generic
            -- prompt when the Stem model uses a separate prompt/interaction object.
            for _, entry in ipairs(allPrompts) do
                if nameHasStem(entry.prompt) then
                    if not bestDistance or entry.distance < bestDistance then
                        bestPrompt, bestDistance = entry.prompt, entry.distance
                    end
                end
            end

            if not bestPrompt then
                local bestTargetDistance = math.huge
                for _, target in ipairs(targets) do
                    local targetDistance = (root.Position - target.part.Position).Magnitude
                    if targetDistance < bestTargetDistance then
                        bestTargetDistance = targetDistance
                    end
                end
                for _, entry in ipairs(allPrompts) do
                    local closestTargetDistance = math.huge
                    for _, target in ipairs(targets) do
                        local d = (target.part.Position - entry.part.Position).Magnitude
                        if d < closestTargetDistance then closestTargetDistance = d end
                    end
                    if closestTargetDistance <= 18 and
                        (not bestDistance or entry.distance < bestDistance) then
                        bestPrompt, bestDistance = entry.prompt, entry.distance
                    end
                end
            end

            if bestPrompt then
                return bestPrompt, "prompt found: " .. bestPrompt.Name
            end
            return nil, "Stem found, but no nearby enabled prompt"
        end

        local function pressG()
            local vim = game:GetService("VirtualInputManager")
            local ok = pcall(function()
                vim:SendKeyEvent(true, Enum.KeyCode.G, false, game)
                task.wait(0.08)
                vim:SendKeyEvent(false, Enum.KeyCode.G, false, game)
            end)
            if not ok and type(keypress) == "function" and type(keyrelease) == "function" then
                pcall(function()
                    keypress(0x47)
                    task.wait(0.08)
                    keyrelease(0x47)
                end)
            end
            return ok
        end

        local function activatePrompt(prompt)
            if not prompt or not prompt.Parent then return false, "prompt disappeared" end
            local ok, err = pcall(function()
                prompt:InputHoldBegin()
                task.wait(math.max(prompt.HoldDuration, 0.1) + 0.1)
                prompt:InputHoldEnd()
            end)
            if not ok then return false, tostring(err) end
            return true, "prompt activation attempted"
        end

        local function runAutoThrow()
            task.spawn(function()
                setStatus("enabled; searching for Stem")
                local lastStatus = ""
                local lastStatusAt = 0
                while env._FairwellStemAutoThrow do
                    local character = player and player.Character
                    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                    if not humanoid or humanoid.Health <= 0 then
                        env._FairwellStemAutoThrow = false
                        setStatus("stopped: character unavailable/dead")
                        break
                    end

                    local prompt, info = findNearbyStemPrompt()
                    if info ~= lastStatus or os.clock() - lastStatusAt > 5 then
                        setStatus(info)
                        lastStatus, lastStatusAt = info, os.clock()
                    end

                    if prompt then
                        local ok, result = activatePrompt(prompt)
                        setStatus(result)
                        task.wait(0.4)
                        local sent = pressG()
                        if sent then
                            setStatus("pickup attempted; G input sent (game may ignore simulated input)")
                        else
                            setStatus("pickup attempted; G input unavailable")
                        end
                        task.wait(0.6)
                    else
                        task.wait(0.5)
                    end
                end
                if not env._FairwellStemAutoThrow then
                    setStatus("off")
                end
            end)
        end

        pcall(function()
            stemBox:AddToggle("FairwellStemAutoThrowToggle", {
                Text = "STEM Auto-Throw",
                Default = false,
                Callback = function(enabled)
                    env._FairwellStemAutoThrow = enabled == true
                    if env._FairwellStemAutoThrow then
                        runAutoThrow()
                    else
                        setStatus("off")
                    end
                end
            })
        end)
    end
end
