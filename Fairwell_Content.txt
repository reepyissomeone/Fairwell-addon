-- Fairwell Heaven: remotely changeable tab content
-- This file is loaded by Fairwell_AddTab.txt. It does not create a window.

return function(tab, reloadContent)
    if not tab then
        warn("[Fairwell] Remote content received no tab.")
        return
    end

    local function addLabel(box, text)
        local ok, err = pcall(function()
            box:AddLabel({ Text = text, DoesWrap = true })
        end)
        if not ok then
            warn("[Fairwell] Could not add label:", err)
        end
    end

    local function addButton(box, text, callback)
        local ok = pcall(function()
            box:AddButton({
                Text = text,
                Func = callback
            })
        end)
        if not ok then
            pcall(function()
                box:AddButton(text, callback)
            end)
        end
    end

    local infoOk, info = pcall(function()
        return tab:AddLeftGroupbox("Fairwell Heaven - Info", "info")
    end)
    if infoOk and info then
        addLabel(info, "Fairwell Heaven Addon")
        addLabel(info, "Version: 1.1.0")
        addLabel(info, "Purpose: extra Fairwell features for MSDoors.")
        addLabel(info, "This addon adds to the existing MSDoors window; it does not create a second window.")
        addLabel(info, "Credits: Reepy Walson — concept, Fairwell character, and project direction.")
        addLabel(info, "Addon code and integration: Fairwell Heaven project.")
        addLabel(info, "Repository: github.com/reepyissomeone/Fairwell-addon")
        addLabel(info, "Remote content is fetched from the public repository when this addon starts or is reloaded.")
    end

    local remoteOk, remote = pcall(function()
        return tab:AddLeftGroupbox("Fairwell - Remote Content", "refresh-cw")
    end)
    if remoteOk and remote then
        addLabel(remote, "Content source: Fairwell_Content.txt on GitHub.")
        addLabel(remote, "Reloading rebuilds this tab's content. Changes appear after a successful reload.")
        if type(reloadContent) == "function" then
            addButton(remote, "Reload Fairwell Content", function()
                reloadContent()
            end)
        else
            addLabel(remote, "Reload callback unavailable. Re-run the Fairwell addon to fetch updates.")
        end
    end

    local toolsOk, toolsBox = pcall(function()
        return tab:AddLeftGroupbox("Fairwell - Addon Tools", "wrench")
    end)
    if toolsOk and toolsBox then
        addLabel(toolsBox, "These are Fairwell addon controls, not replacements for MSDoors features.")
        addButton(toolsBox, "Print Addon Status", function()
            print("[Fairwell] Addon is running. Remote content version: 1.1.0")
        end)
        addButton(toolsBox, "Copy Repository URL", function()
            local url = "https://github.com/reepyissomeone/Fairwell-addon"
            if type(setclipboard) == "function" then
                setclipboard(url)
                print("[Fairwell] Repository URL copied.")
            else
                print("[Fairwell] Repository: " .. url)
            end
        end)

    local stemOk, stemBox = pcall(function()
        return tab:AddLeftGroupbox("STEM Auto-Throw", "target")
    end)
    if stemOk and stemBox then
        addLabel(stemBox, "Automatically picks up nearby Stem prompts and presses the mobile Throw button. Stem usually needs several throws to be destroyed.")
        addLabel(stemBox, "Stay near the Stem and keep the game visible. Toggle off if it gets stuck.")
        
        local Players = game:GetService("Players")
        local ProximityPromptService = game:GetService("ProximityPromptService")
        local LocalPlayer = Players.LocalPlayer
        local env = (getgenv and getgenv()) or _G
        env._FairwellStemAutoThrow = env._FairwellStemAutoThrow or false

        local function isStemPrompt(prompt)
            if not prompt or not prompt:IsA("ProximityPrompt") then return false end
            local text = string.lower(tostring(prompt.Name) .. " " .. tostring(prompt.ActionText) .. " " .. tostring(prompt.ObjectText))
            local node = prompt.Parent
            for _ = 1, 5 do
                if not node then break end
                text = text .. " " .. string.lower(tostring(node.Name))
                node = node.Parent
            end
            return string.find(text, "stem", 1, true) ~= nil
        end

        local function findThrowButton()
            local pg = LocalPlayer and LocalPlayer:FindFirstChildOfClass("PlayerGui")
            if not pg then return nil end
            for _, item in ipairs(pg:GetDescendants()) do
                if item:IsA("GuiButton") and item.Visible then
                    local label = string.lower(tostring(item.Name))
                    if item:IsA("TextButton") then label = label .. " " .. string.lower(item.Text) end
                    if string.find(label, "throw", 1, true) then
                        return item
                    end
                end
            end
            return nil
        end

        local function findNearbyStemPrompt()
            local character = LocalPlayer and LocalPlayer.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not root then return nil end
            local best, bestDistance = nil, math.huge
            for _, item in ipairs(workspace:GetDescendants()) do
                if item:IsA("ProximityPrompt") and item.Enabled and isStemPrompt(item) then
                    local parent = item.Parent
                    local part = parent and (parent:IsA("BasePart") and parent or parent:FindFirstChildWhichIsA("BasePart", true))
                    if part then
                        local distance = (root.Position - part.Position).Magnitude
                        if distance <= math.max(item.MaxActivationDistance + 3, 10) and distance < bestDistance then
                            best, bestDistance = item, distance
                        end
                    end
                end
            end
            return best
        end

        local function autoThrowLoop()
            task.spawn(function()
                print("[Fairwell STEM] Auto-Throw enabled. Looking for Stem prompts.")
                local throwsWithoutProgress = 0
                while env._FairwellStemAutoThrow do
                    local character = LocalPlayer and LocalPlayer.Character
                    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                    if not humanoid or humanoid.Health <= 0 then
                        print("[Fairwell STEM] Stopped because the player is dead.")
                        env._FairwellStemAutoThrow = false
                        break
                    end

                    local prompt = findNearbyStemPrompt()
                    if prompt then
                        print("[Fairwell STEM] Picking up: " .. tostring(prompt.ObjectText ~= "" and prompt.ObjectText or prompt.Parent.Name))
                        pcall(function()
                            prompt:InputHoldBegin()
                            task.wait(math.max(prompt.HoldDuration, 0.15))
                            prompt:InputHoldEnd()
                        end)
                        task.wait(0.7)

                        local throwButton = findThrowButton()
                        if throwButton then
                            pcall(function() throwButton:Activate() end)
                            print("[Fairwell STEM] Pressed the on-screen Throw button.")
                            task.wait(1.1)
                            throwsWithoutProgress = 0
                        else
                            throwsWithoutProgress += 1
                            if throwsWithoutProgress >= 4 then
                                print("[Fairwell STEM] Couldn't find a visible Throw button. Toggle off/on after checking the mobile UI.")
                                env._FairwellStemAutoThrow = false
                                break
                            end
                            task.wait(0.6)
                        end
                    else
                        task.wait(0.6)
                    end
                end
                print("[Fairwell STEM] Auto-Throw stopped.")
            end)
        end

        addButton(stemBox, "Toggle STEM Auto-Throw", function()
            env._FairwellStemAutoThrow = not env._FairwellStemAutoThrow
            if env._FairwellStemAutoThrow then
                autoThrowLoop()
            else
                print("[Fairwell STEM] Auto-Throw disabled.")
            end
        end)
    end
    end
end
